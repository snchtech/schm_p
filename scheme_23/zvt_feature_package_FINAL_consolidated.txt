export const ZVT_BRANCH_DEFAULT_GRID = 10;

export function snapValue(
  value,
  gridSize = ZVT_BRANCH_DEFAULT_GRID
) {
  if (!gridSize || gridSize <= 1) {
    return value;
  }

  return Math.round(value / gridSize) * gridSize;
}

export function snapPoint(
  point,
  gridSize = ZVT_BRANCH_DEFAULT_GRID
) {
  return {
    x: snapValue(point.x, gridSize),
    y: snapValue(point.y, gridSize),
  };
}

function samePoint(a, b) {
  return (
    a &&
    b &&
    a.x === b.x &&
    a.y === b.y
  );
}

function isHorizontal(a, b) {
  return a.y === b.y;
}

function isVertical(a, b) {
  return a.x === b.x;
}

function isOrthogonal(a, b) {
  return (
    isHorizontal(a, b) ||
    isVertical(a, b)
  );
}

function isCollinear(a, b, c) {
  return (
    (
      a.x === b.x &&
      b.x === c.x
    ) ||
    (
      a.y === b.y &&
      b.y === c.y
    )
  );
}

export function dedupePoints(points) {
  const result = [];

  for (const point of points || []) {
    if (
      !result.length ||
      !samePoint(
        result[result.length - 1],
        point
      )
    ) {
      result.push(point);
    }
  }

  return result;
}

export function simplifyOrthogonalPoints(
  points
) {
  const list =
    dedupePoints(points);

  if (list.length <= 2) {
    return list;
  }

  const result = [list[0]];

  for (
    let i = 1;
    i < list.length - 1;
    i++
  ) {
    const prev =
      result[result.length - 1];

    const current =
      list[i];

    const next =
      list[i + 1];

    if (
      isCollinear(
        prev,
        current,
        next
      )
    ) {
      continue;
    }

    result.push(current);
  }

  result.push(
    list[list.length - 1]
  );

  return result;
}

/**
 * Ensure orthogonal route.
 *
 * For non-orthogonal pair insert one bend.
 * Used for horizontal/vertical/elbow route parts.
 *
 * free branch first root segment is handled separately and can remain diagonal.
 */
export function orthogonalizePoints(
  points,
  {
    prefer = "horizontal-first",
  } = {}
) {
  if (!points?.length) {
    return [];
  }

  const result = [
    { ...points[0] },
  ];

  for (
    let i = 1;
    i < points.length;
    i++
  ) {
    const prev =
      result[result.length - 1];

    const next = {
      ...points[i],
    };

    if (
      isOrthogonal(
        prev,
        next
      )
    ) {
      result.push(next);
      continue;
    }

    const bend =
      prefer === "vertical-first"
        ? {
            x: prev.x,
            y: next.y,
          }
        : {
            x: next.x,
            y: prev.y,
          };

    result.push(
      bend,
      next
    );
  }

  return simplifyOrthogonalPoints(
    result
  );
}

/**
 * Insert a local dog-leg into selected segment.
 *
 * horizontal:
 *
 * -----+          +-----
 *      |          |
 *      +----------+
 *
 * vertical works analogously.
 */
export function insertDogLeg(
  points,
  segmentIndex,
  pointer,
  {
    gridSize =
      ZVT_BRANCH_DEFAULT_GRID,

    offset =
      ZVT_BRANCH_DEFAULT_GRID * 3,
  } = {}
) {
  if (
    segmentIndex < 0 ||
    segmentIndex >=
      points.length - 1
  ) {
    return points;
  }

  const a =
    points[segmentIndex];

  const b =
    points[segmentIndex + 1];

  const p =
    snapPoint(
      pointer,
      gridSize
    );

  const result =
    points.map(
      (point) => ({
        ...point,
      })
    );

  if (
    isHorizontal(a, b)
  ) {
    const minX =
      Math.min(a.x, b.x);

    const maxX =
      Math.max(a.x, b.x);

    const x =
      Math.max(
        minX,
        Math.min(
          maxX,
          p.x
        )
      );

    let y = p.y;

    if (y === a.y) {
      y =
        snapValue(
          a.y + offset,
          gridSize
        );
    }

    const x2 =
      Math.max(
        minX,
        Math.min(
          maxX,
          x + offset
        )
      );

    result.splice(
      segmentIndex + 1,
      0,
      {
        x,
        y: a.y,
      },
      {
        x,
        y,
      },
      {
        x: x2,
        y,
      },
      {
        x: x2,
        y: a.y,
      }
    );
  } else if (
    isVertical(a, b)
  ) {
    const minY =
      Math.min(a.y, b.y);

    const maxY =
      Math.max(a.y, b.y);

    const y =
      Math.max(
        minY,
        Math.min(
          maxY,
          p.y
        )
      );

    let x = p.x;

    if (x === a.x) {
      x =
        snapValue(
          a.x + offset,
          gridSize
        );
    }

    const y2 =
      Math.max(
        minY,
        Math.min(
          maxY,
          y + offset
        )
      );

    result.splice(
      segmentIndex + 1,
      0,
      {
        x: a.x,
        y,
      },
      {
        x,
        y,
      },
      {
        x,
        y: y2,
      },
      {
        x: a.x,
        y: y2,
      }
    );
  } else {
    /**
     * Diagonal segment can occur in FREE branch.
     * For such segment we do NOT force dog-leg automatically.
     * Insert one control point at pointer.
     */
    result.splice(
      segmentIndex + 1,
      0,
      p
    );
  }

  return simplifyOrthogonalPoints(
    result
  );
}

/**
 * Move one bend handle.
 *
 * Interior orthogonal vertex:
 * adjust neighbors to preserve 90° turns.
 */
export function moveBendOrthogonal(
  points,
  index,
  target,
  {
    gridSize =
      ZVT_BRANCH_DEFAULT_GRID,
  } = {}
) {
  if (
    index <= 0 ||
    index >=
      points.length - 1
  ) {
    return points;
  }

  const result =
    points.map(
      (point) => ({
        ...point,
      })
    );

  const p =
    snapPoint(
      target,
      gridSize
    );

  const prev =
    result[index - 1];

  const current =
    result[index];

  const next =
    result[index + 1];

  const prevH =
    isHorizontal(
      prev,
      current
    );

  const nextH =
    isHorizontal(
      current,
      next
    );

  if (
    prevH &&
    !nextH
  ) {
    result[index] = p;

    result[index - 1] = {
      ...prev,
      y: p.y,
    };

    result[index + 1] = {
      ...next,
      x: p.x,
    };
  } else if (
    !prevH &&
    nextH
  ) {
    result[index] = p;

    result[index - 1] = {
      ...prev,
      x: p.x,
    };

    result[index + 1] = {
      ...next,
      y: p.y,
    };
  } else if (
    prevH &&
    nextH
  ) {
    result[index] = {
      x: current.x,
      y: p.y,
    };

    result[index - 1] = {
      ...prev,
      y: p.y,
    };

    result[index + 1] = {
      ...next,
      y: p.y,
    };
  } else {
    result[index] = {
      x: p.x,
      y: current.y,
    };

    result[index - 1] = {
      ...prev,
      x: p.x,
    };

    result[index + 1] = {
      ...next,
      x: p.x,
    };
  }

  return simplifyOrthogonalPoints(
    result
  );
}

/**
 * Shift whole segment parallel to itself.
 *
 * This is the UX requested for:
 * "трохи зміщувати branch при необхідності"
 *
 * horizontal segment:
 *   drag up/down => move Y of both endpoints.
 *
 * vertical segment:
 *   drag left/right => move X of both endpoints.
 *
 * Neighboring points are updated automatically.
 */
export function shiftSegmentParallel(
  points,
  segmentIndex,
  pointer,
  {
    gridSize =
      ZVT_BRANCH_DEFAULT_GRID,
  } = {}
) {
  if (
    segmentIndex < 0 ||
    segmentIndex >=
      points.length - 1
  ) {
    return points;
  }

  const result =
    points.map(
      (point) => ({
        ...point,
      })
    );

  const a =
    result[segmentIndex];

  const b =
    result[segmentIndex + 1];

  const p =
    snapPoint(
      pointer,
      gridSize
    );

  if (
    isHorizontal(a, b)
  ) {
    const nextY =
      p.y;

    result[segmentIndex] = {
      ...a,
      y: nextY,
    };

    result[segmentIndex + 1] = {
      ...b,
      y: nextY,
    };

    if (
      segmentIndex > 0
    ) {
      result[segmentIndex - 1] = {
        ...result[
          segmentIndex - 1
        ],
        x:
          result[segmentIndex].x,
      };
    }

    if (
      segmentIndex + 2 <
      result.length
    ) {
      result[segmentIndex + 2] = {
        ...result[
          segmentIndex + 2
        ],
        x:
          result[
            segmentIndex + 1
          ].x,
      };
    }
  } else if (
    isVertical(a, b)
  ) {
    const nextX =
      p.x;

    result[segmentIndex] = {
      ...a,
      x: nextX,
    };

    result[segmentIndex + 1] = {
      ...b,
      x: nextX,
    };

    if (
      segmentIndex > 0
    ) {
      result[segmentIndex - 1] = {
        ...result[
          segmentIndex - 1
        ],
        y:
          result[segmentIndex].y,
      };
    }

    if (
      segmentIndex + 2 <
      result.length
    ) {
      result[segmentIndex + 2] = {
        ...result[
          segmentIndex + 2
        ],
        y:
          result[
            segmentIndex + 1
          ].y,
      };
    }
  } else {
    // FREE diagonal segment:
    // do not shift orthogonally.
    return result;
  }

  return simplifyOrthogonalPoints(
    result
  );
}

/**
 * Delete selected interior bend.
 */
export function deleteBend(
  points,
  index
) {
  if (
    index <= 0 ||
    index >=
      points.length - 1
  ) {
    return points;
  }

  const result =
    points.filter(
      (_, i) =>
        i !== index
    );

  return simplifyOrthogonalPoints(
    orthogonalizePoints(
      result
    )
  );
}

import {
  buildBaseBranchRoute,
} from "@/lib/zvt/zvtBranchRouteBuilder";

import {
  resolveZvtBranchRoot,
} from "@/lib/zvt/zvtBranchRootGeometry";

import {
  simplifyOrthogonalPoints,
} from "@/lib/zvt/zvtBranchEditGeometry";


export function buildEditableBranchPoints(
  branch,
  nodes,
  edges
) {
  const root =
    resolveZvtBranchRoot(
      branch,
      nodes,
      edges
    );

  const base =
    buildBaseBranchRoute(
      branch,
      root
    );

  const manual =
    branch.route?.manualVertices || [];

  if (!manual.length) {
    return base;
  }

  return simplifyOrthogonalPoints(
    [
      base[0],
      ...manual,
      base[
        base.length - 1
      ],
    ]
  );
}


export function editablePointsToRoute(
  branch,
  points
) {
  if (
    !points ||
    points.length < 2
  ) {
    return branch.route;
  }

  /**
   * Root is dynamic and not persisted.
   * End stays terminalPoint.
   * Interior points become manualVertices.
   */
  const middle =
    points.slice(
      1,
      -1
    );

  const end =
    points[
      points.length - 1
    ];

  return {
    ...branch.route,

    manualVertices:
      middle.map(
        (
          point,
          index
        ) => ({
          id:
            point.id ||
            branch.route
              ?.manualVertices?.[
                index
              ]?.id ||
            crypto.randomUUID(),

          x: point.x,
          y: point.y,
        })
      ),

    terminalPoint: {
      x: end.x,
      y: end.y,
    },
  };
}

insertZvtBranchBendAt(
  branchId,
  segmentIndex,
  pointer
) {
  const store =
    useCanvasStore.getState();

  if (
    (store.accessMode || "editor") ===
    "reader"
  ) {
    return;
  }

  const branch =
    (store.zvtBranches || [])
      .find(
        (item) =>
          item.id === branchId
      );

  if (!branch) {
    return;
  }

  const points =
    buildEditableBranchPoints(
      branch,
      store.nodes || [],
      store.edges || []
    );

  const next =
    insertDogLeg(
      points,
      segmentIndex,
      pointer,
      {
        gridSize:
          store.gridSize || 10,
      }
    );

  store.replaceZvtBranch?.(
    branchId,
    {
      ...branch,

      route:
        editablePointsToRoute(
          branch,
          next
        ),

      updatedAt:
        Date.now(),
    }
  );
},

function projectToSegment(
  point,
  a,
  b
) {
  const dx =
    b.x - a.x;

  const dy =
    b.y - a.y;

  const lenSq =
    dx * dx + dy * dy;

  if (!lenSq) {
    return {
      point: { ...a },
      distanceSq:
        (
          point.x - a.x
        ) ** 2 +
        (
          point.y - a.y
        ) ** 2,
    };
  }

  let t =
    (
      (point.x - a.x) *
        dx +
      (point.y - a.y) *
        dy
    ) /
    lenSq;

  t =
    Math.max(
      0,
      Math.min(1, t)
    );

  const projected = {
    x:
      a.x + dx * t,
    y:
      a.y + dy * t,
  };

  return {
    point: projected,
    distanceSq:
      (
        point.x -
        projected.x
      ) ** 2 +
      (
        point.y -
        projected.y
      ) ** 2,
  };
}


export function findNearestSegmentIndex(
  points,
  pointer
) {
  let bestIndex = -1;
  let bestDistance = Infinity;

  for (
    let i = 0;
    i < points.length - 1;
    i++
  ) {
    const candidate =
      projectToSegment(
        pointer,
        points[i],
        points[i + 1]
      );

    if (
      candidate.distanceSq <
      bestDistance
    ) {
      bestDistance =
        candidate.distanceSq;

      bestIndex = i;
    }
  }

  return bestIndex;
}

"use client";

import React from "react";

function buildParameterLabel(zvt) {
  const parameter = zvt.data?.parameter || "";
  const suffix = zvt.data?.parameterSuffix || "";
  return `${parameter}${suffix}`;
}

function buildFunctionLabel(zvt) {
  const signs = Array.isArray(zvt.data?.functionSigns)
    ? zvt.data.functionSigns.join("")
    : "";
  const suffix = zvt.data?.functionSuffix || "";
  return `${signs}${suffix}`;
}

function buildRecipientBaseLabel(zvt) {
  const mode = zvt.data?.recipientLabelMode || "M";

  if (mode === "custom") {
    return (zvt.data?.recipientCustomText || "").trim();
  }

  const recipients = Array.isArray(zvt.data?.recipients)
    ? zvt.data.recipients
    : [];

  if (!recipients.length) return "M";
  return `M${recipients.join(",")}`;
}

function renderParameterGraphicMarkSvg(zvt, {
  x,
  y,
  stroke = "black",
  strokeWidth = 1.6,
} = {}) {
  const parameter = zvt.data?.parameter || "F";
  const scale = zvt.style?.parameterGraphicScale ?? 1;

  const map = {
    F: "flow",
    P: "pressure",
    L: "level",
    T: "temperature",
    ae: "conductivity",
    pH: "conductivity",
  };

  const mark = map[parameter] || "pressure";
  const r = 7 * scale;
  const stemLength = 18 * scale;
  const baseHalf = 11 * scale;
  const circleBottom = y + r;
  const stemBottom = circleBottom + stemLength;
  const sw = strokeWidth * scale;

  let svg = `<circle cx="${x}" cy="${y}" r="${r}" fill="none" stroke="${stroke}" stroke-width="${sw}" />`;

  if (mark === "level") {
    const sx = x + r;
    const ex = sx + 16 * scale;

    svg += `<line x1="${sx}" y1="${y}" x2="${ex}" y2="${y}" stroke="${stroke}" stroke-width="${sw}" />`;
    svg += `<polyline points="${ex},${y} ${ex - 6 * scale},${y - 4 * scale} ${ex - 6 * scale},${y + 4 * scale} ${ex},${y}" fill="none" stroke="${stroke}" stroke-width="${sw}" />`;
    return svg;
  }

  svg += `<line x1="${x}" y1="${circleBottom}" x2="${x}" y2="${stemBottom}" stroke="${stroke}" stroke-width="${sw}" />`;

  if (mark === "flow") {
    svg += `<line x1="${x - baseHalf}" y1="${stemBottom - 2 * scale}" x2="${x + baseHalf}" y2="${stemBottom - 2 * scale}" stroke="${stroke}" stroke-width="${sw}" />`;
    svg += `<line x1="${x - 4 * scale}" y1="${stemBottom - 6 * scale}" x2="${x - 4 * scale}" y2="${stemBottom + 3 * scale}" stroke="${stroke}" stroke-width="${sw}" />`;
    svg += `<line x1="${x + 4 * scale}" y1="${stemBottom - 6 * scale}" x2="${x + 4 * scale}" y2="${stemBottom + 3 * scale}" stroke="${stroke}" stroke-width="${sw}" />`;
  }

  if (mark === "pressure") {
    svg += `<line x1="${x - baseHalf}" y1="${stemBottom}" x2="${x + baseHalf}" y2="${stemBottom}" stroke="${stroke}" stroke-width="${sw}" />`;
  }

  if (mark === "temperature") {
    svg += `<line x1="${x - baseHalf}" y1="${stemBottom - 3 * scale}" x2="${x + baseHalf}" y2="${stemBottom - 3 * scale}" stroke="${stroke}" stroke-width="${sw}" />`;
    svg += `<line x1="${x - baseHalf}" y1="${stemBottom + 2 * scale}" x2="${x + baseHalf}" y2="${stemBottom + 2 * scale}" stroke="${stroke}" stroke-width="${sw}" />`;
  }

  if (mark === "conductivity") {
    svg += `<line x1="${x - 13 * scale}" y1="${stemBottom}" x2="${x + 13 * scale}" y2="${stemBottom}" stroke="${stroke}" stroke-width="${sw}" />`;
  }

  return svg;
}

function buildSignalLabel(zvt) {
  const base = buildRecipientBaseLabel(zvt);
  const actionSign = zvt.data?.actionSign || "none";
  const position = zvt.data?.actionSignPosition || "before";

  if (!base) return "";
  if (actionSign === "none") return base;

  return position === "after"
    ? `${base} ${actionSign}`
    : `${actionSign} ${base}`;
}

function formatRealtimeValue(zvt, runtime) {
  if (!runtime) return "";

  const telemetry = zvt.integration?.telemetry;
  if (!telemetry?.enabled || !telemetry?.showOnCanvas) {
    return "";
  }

  if (runtime.connectionState === "error") {
    return "ERR";
  }

  if (runtime.value === null || runtime.value === undefined) {
    return runtime.connectionState === "connecting"
      ? "..."
      : "";
  }

  const precision = Number.isFinite(telemetry.precision)
    ? telemetry.precision
    : 2;

  const numeric = Number(runtime.value);
  const valueText = Number.isFinite(numeric)
    ? numeric.toFixed(precision)
    : String(runtime.value);

  const unit = runtime.unit || telemetry.unit || "";

  return unit
    ? `${valueText} ${unit}`
    : valueText;
}

function isRealtimeStale(zvt, runtime) {
  const receivedAt = runtime?.receivedAt;
  if (!receivedAt) return false;

  const staleAfterMs =
    zvt.integration?.telemetry?.staleAfterMs ?? 15000;

  return Date.now() - receivedAt > staleAfterMs;
}

  const showParameterGraphicMark = zvt.data?.showParameterGraphicMark ?? true;

  const realtimeOffsetY =

            y={-radius + 3}
            width={radius * 2}
            height={radius - 3}
            text={functionLabel}
            fontFamily={fontFamily}
            fontSize={fontSize}
            fontStyle="bold"
            align="center"
            verticalAlign="middle"
            fill="black"
          />

          <Text
            x={-radius}
            y={1}
            width={radius * 2}
            height={radius - 2}
            text={parameterLabel}
            fontFamily={fontFamily}
            fontSize={fontSize}
            fontStyle="bold"
            align="center"
            verticalAlign="middle"
            fill="black"
          />
        </>
      ) : (
        <>
          {/* "За місцем": обидва написи є, але горизонтальної лінії немає. */}
          {hasFunctionLabel && (
            <Text
              x={-radius}
              y={hasParameterLabel ? -radius + 5 : -fontSize / 2}
              width={radius * 2}
              height={fontSize + 4}
              text={functionLabel}
              fontFamily={fontFamily}
              fontSize={fontSize}
              fontStyle="bold"
              align="center"
              verticalAlign="middle"
              fill="black"
            />
          )}

          {hasParameterLabel && (
            <Text
              x={-radius}
              y={hasFunctionLabel ? 2 : -fontSize / 2}
              width={radius * 2}
              height={fontSize + 4}
              text={parameterLabel}
              fontFamily={fontFamily}
              fontSize={fontSize}
              fontStyle="bold"
              align="center"
              verticalAlign="middle"
              fill="black"
            />
          )}
        </>
      )}

      {realtimeText && (
        <Text
          x={-radius * 2}
          y={radius + realtimeOffsetY}
          width={radius * 4}
          text={
            realtimeStale
              ? `${realtimeText} • stale`
              : realtimeText
          }
          fontFamily={fontFamily}
          fontSize={realtimeFontSize}
          fontStyle="bold"
          align="center"
          fill="black"
          opacity={realtimeStale ? 0.55 : 1}
          listening={false}
        />
      )}

      {signalGeometry && (
        <>
          <Line
            points={signalGeometry.line.map((value, index) => (index % 2 === 0 ? value - x : value - y))}
            stroke="black"
            strokeWidth={2}
            dash={[18, 8]}
          />

          <Line
            points={signalGeometry.arrowPoints.map((value, index) => (index % 2 === 0 ? value - x : value - y))}

            fill="black"
            stroke="black"
            strokeWidth={1}
          />

          <Text
            x={signalGeometry.labelX - x}
            y={signalGeometry.labelY - y}
            width={signalGeometry.labelWidth}
            text={signalLabel}
            fontFamily={fontFamily}
            fontSize={smallFontSize + 1}
            fontStyle="bold"
            fill="black"
            align={zvt.data?.signal?.direction === "left" ? "left" : "left"}
          />
        </>
      )}
    </Group>
  );
}

 *
 * Він НЕ знає про React/Konva.
 * Його задача:
 * - connect;
 * - subscribe sensorIds;
 * - normalize incoming messages;
 * - передати normalized telemetry у callback.
 *
 * Очікуваний payload за замовчуванням:
 *
 * {
 *   "sensorId": "SENSOR-PT-101",
 *   "value": 12.34,
 *   "timestamp": "2026-09-07T16:40:00Z",
 *   "quality": "good"
 * }
 *
 * Якщо backend матиме інший формат —
 * змінюємо тільки normalizeMessage/subscribeMessageFactory.
 */

export class ZvtRealtimeClient {
  constructor({
    url,
    onTelemetry,
    onStateChange,
    subscribeMessageFactory,
  }) {
    this.url = url;
    this.onTelemetry = onTelemetry;
    this.onStateChange = onStateChange;

    this.subscribeMessageFactory =
      subscribeMessageFactory ||
      ((sensorIds) => ({
        type: "subscribe",
        sensorIds,
      }));

    this.socket = null;
    this.sensorIds = [];
    this.reconnectTimer = null;
    this.closedManually = false;
  }

  connect(sensorIds = []) {
    this.sensorIds = [
      ...new Set(sensorIds.filter(Boolean)),
    ];

    if (!this.url || this.sensorIds.length === 0) {
      return;
    }

    this.closedManually = false;
    this.onStateChange?.("connecting");

    this.socket = new WebSocket(this.url);

    this.socket.onopen = () => {
      this.onStateChange?.("connected");
      this.sendSubscription();
    };

    this.socket.onmessage = (event) => {
      try {
        const raw = JSON.parse(event.data);
        const normalized = this.normalizeMessage(raw);

        if (normalized) {
          this.onTelemetry?.(normalized);
        }
      } catch (error) {
        console.warn(
          "[ZVT realtime] invalid message",
          error
        );
      }
    };

    this.socket.onerror = () => {
      this.onStateChange?.("error");
    };

    this.socket.onclose = () => {
      this.onStateChange?.("disconnected");

      if (!this.closedManually) {
        this.scheduleReconnect();
      }
    };
  }

  updateSubscriptions(sensorIds = []) {
    this.sensorIds = [
      ...new Set(sensorIds.filter(Boolean)),
    ];

    if (this.sensorIds.length === 0) {
      this.disconnect();
      return;
    }

    if (
      !this.socket ||
      this.socket.readyState === WebSocket.CLOSED
    ) {
      this.connect(this.sensorIds);
      return;
    }

    if (
      this.socket.readyState === WebSocket.OPEN
    ) {
      this.sendSubscription();
    }
  }

  sendSubscription() {
    if (
      !this.socket ||
      this.socket.readyState !== WebSocket.OPEN
    ) {
      return;
    }

    this.socket.send(
      JSON.stringify(
        this.subscribeMessageFactory(this.sensorIds)
      )
    );
  }

  normalizeMessage(raw) {
    if (!raw || !raw.sensorId) return null;

    return {
      sensorId: String(raw.sensorId),
      value: raw.value,
      timestamp:
        raw.timestamp || new Date().toISOString(),
      quality: raw.quality || "unknown",
      raw,
    };
  }

  scheduleReconnect() {
    clearTimeout(this.reconnectTimer);

    this.reconnectTimer = setTimeout(() => {
      this.connect(this.sensorIds);
    }, 3000);
  }

  disconnect() {
    this.closedManually = true;
    clearTimeout(this.reconnectTimer);

    if (this.socket) {
      this.socket.close();
      this.socket = null;
    }
  }
}

"use client";

import React, {
  useEffect,
  useMemo,
  useRef,
} from "react";

import { useCanvasStore } from "@/store/useCanvasStore";
import { ZvtRealtimeClient } from "@/services/zvt/zvtRealtimeClient";

/**
 * React bridge між persisted zvt.integration і runtime Zustand.
 *
 * Монтувати ОДИН раз поза <Stage>.
 *
 * НЕ створювати WebSocket у кожному ZvtSymbol.
 * Інакше 50 ЗВТ = 50 socket connections.
 */
export default function ZvtRealtimeBridge() {
  const zvts = useCanvasStore((s) => s.zvts || []);
  const setZvtRuntime = useCanvasStore(
    (s) => s.setZvtRuntime
  );

  const realtimeEnabled = useCanvasStore(
    (s) =>
      s.zvtSettings?.realtimeEnabled ?? true
  );

  const clientRef = useRef(null);

  const bindings = useMemo(() => {
    return (zvts || [])
      .filter(
        (zvt) =>
          zvt.integration?.telemetry?.enabled &&
          zvt.integration?.telemetry?.sensorId
      )
      .map((zvt) => ({
        zvtId: zvt.id,
        sensorId:
          zvt.integration.telemetry.sensorId,
        config: zvt.integration.telemetry,
      }));
  }, [zvts]);

  const bindingBySensorId = useMemo(() => {
    const map = new Map();

    for (const binding of bindings) {
      const list =
        map.get(binding.sensorId) || [];

      list.push(binding);
      map.set(binding.sensorId, list);
    }

    return map;
  }, [bindings]);

  const sensorIds = useMemo(
    () => bindings.map((item) => item.sensorId),
    [bindings]
  );

  useEffect(() => {
    const url =
      process.env.NEXT_PUBLIC_ZVT_WS_URL;

    if (!realtimeEnabled) {
      return;
    }

    if (!url) {
      // Поки backend не підключений — просто нічого не робимо.
      return;
    }

    const client = new ZvtRealtimeClient({
      url,

      onTelemetry: (message) => {
        const currentBindings =
          bindingBySensorId.get(message.sensorId) || [];

        for (const binding of currentBindings) {
          const config = binding.config;

          /**
           * Якщо backend надалі передаватиме raw payload
           * з іншим valueField, можна брати значення так:
           */
          const rawValue =
            message.raw?.[config.valueField] ??
            message.value;

          const timestamp =
            message.raw?.[config.timestampField] ??
            message.timestamp;

          const quality =
            message.raw?.[config.qualityField] ??
            message.quality;

          setZvtRuntime?.(
            binding.zvtId,
            {
              sensorId: message.sensorId,
              rawValue,
              value: rawValue,
              unit: config.unit || "",
              timestamp,
              quality,
              connectionState: "connected",
              receivedAt: Date.now(),
              error: null,
            }
          );
        }
      },

      onStateChange: (state) => {
        for (const binding of bindings) {
          setZvtRuntime?.(
            binding.zvtId,
            {
              sensorId: binding.sensorId,
              connectionState: state,
            }
          );
        }
      },
    });

    clientRef.current = client;
    client.connect(sensorIds);

    return () => {
      client.disconnect();
      clientRef.current = null;
    };
  }, [
    bindingBySensorId,
    bindings,
    sensorIds,
    setZvtRuntime,
    realtimeEnabled,
  ]);

  return null;
}

import {
  buildZvtBranchMemberLeader,
  buildZvtBranchPoints,
} from "@/lib/zvt/zvtBranchGeometry";

function flatPointsToSvg(
  flat
) {
  const pairs = [];

  for (
    let i = 0;
    i < flat.length;
    i += 2
  ) {
    pairs.push(
      `${flat[i]},${flat[i + 1]}`
    );
  }

  return pairs.join(" ");
}

function objectPointsToSvg(
  points
) {
  return points
    .map(
      (point) =>
        `${point.x},${point.y}`
    )
    .join(" ");
}

export function renderZvtBranchesSvg(
  scene
) {
  const branches =
    scene.zvtBranches || [];

  const zvts =
    scene.zvts || [];

  return branches
    .map((branch) => {
      /**
       * КЛЮЧОВО:
       * buildZvtBranchPoints()
       * вже:
       * - resolve dynamic root;
       * - orthogonalize;
       * - simplify.
       *
       * Тому SVG = Canvas geometry.
       */
      const branchPoints =
        buildZvtBranchPoints(
          branch,
          scene.nodes || [],
          scene.edges || []
        );

      const dash =
        branch.style?.dash?.length
          ? ` stroke-dasharray="${branch.style.dash.join(" ")}"`
          : "";

      let svg =
        `<polyline ` +
        `points="${objectPointsToSvg(branchPoints)}" ` +
        `fill="none" ` +
        `stroke="${branch.style?.stroke || "black"}" ` +
        `stroke-width="${branch.style?.strokeWidth ?? 2}" ` +
        `stroke-linecap="round" ` +
        `stroke-linejoin="round"` +
        `${dash} />`;

      const members =
        zvts.filter(
          (zvt) =>
            zvt.route?.branchId ===
            branch.id
        );

      for (
        const zvt of members
      ) {
        const leader =
          buildZvtBranchMemberLeader({
            branchPoints,
            zvt,
          });

        if (
          leader.length >= 4
        ) {
          svg +=
            `<polyline ` +
            `points="${flatPointsToSvg(leader)}" ` +
            `fill="none" ` +
            `stroke="${zvt.style?.stroke || "black"}" ` +
            `stroke-width="${zvt.style?.strokeWidth ?? 2}" ` +
            `stroke-linecap="round" ` +
            `stroke-linejoin="round" />`;
        }
      }

      return svg;
    })
    .join("\n");
}

import { getZvtInstallPlace } from "@/lib/zvt/zvtDictionaries";
import { buildZvtLeaderRoute, buildZvtSignalGeometry, resolveZvtAnchor } from "@/lib/zvt/zvtGeometry";

function esc(value) {
  return String(value ?? "")
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;")
    .replaceAll('"', "&quot;");
}

function pointsToSvg(points) {
  const pairs = [];
  for (let i = 0; i < points.length; i += 2) {
    pairs.push(`${points[i]},${points[i + 1]}`);
  }
  return pairs.join(" ");
}

function buildParameterLabel(zvt) {
  return `${zvt.data?.parameter || ""}${zvt.data?.parameterSuffix || ""}`;
}

function buildFunctionLabel(zvt) {
  const signs = Array.isArray(zvt.data?.functionSigns)
    ? zvt.data.functionSigns.join("")
    : "";
  return `${signs}${zvt.data?.functionSuffix || ""}`;
}

function buildSignalLabel(zvt) {
  const mode = zvt.data?.recipientLabelMode || "M";

  let base = "";
  if (mode === "custom") {
    base = (zvt.data?.recipientCustomText || "").trim();
  } else {
    const recipients = Array.isArray(zvt.data?.recipients)
      ? zvt.data.recipients
      : [];
    base = recipients.length ? `M${recipients.join(",")}` : "M";
  }

  const sign = zvt.data?.actionSign || "none";
  const signPosition = zvt.data?.actionSignPosition || "before";

  if (!base || sign === "none") return base;
  return signPosition === "after"
    ? `${base} ${sign}`
    : `${sign} ${base}`;
}

export function renderZvtSvg(zvt, scene) {
  const nodes = scene.nodes || [];
  const edges = scene.edges || [];

  const x = zvt.position?.x ?? 0;
  const y = zvt.position?.y ?? 0;
  const r = zvt.style?.radius ?? 22;
  const installPlace = getZvtInstallPlace(zvt.data?.installPlace);
  const functionLabel = buildFunctionLabel(zvt);
  const parameterLabel = buildParameterLabel(zvt);
  const stroke = zvt.style?.stroke || "black";
  const strokeWidth = zvt.style?.strokeWidth ?? 2;
  const fontFamily = zvt.style?.fontFamily || "Roboto Condensed";
  const fontSize = zvt.style?.fontSize ?? 13;

  const anchor = resolveZvtAnchor(zvt, nodes, edges);
  const leader = buildZvtLeaderRoute({
    anchor,
    symbol: zvt.position,
    radius: r,
    sourceSide:
      zvt.attachedTo?.type === "node"
        ? (
            zvt.route?.sourceSide ||
            zvt.route?.preferredSide ||
            anchor.side
          )
        : null,
    symbolSide: zvt.route?.symbolSide || "auto",
    sourceOffset: zvt.route?.sourceOffset ?? 18,
    symbolOffset: zvt.route?.symbolOffset ?? 12,
  });
  const signal = buildZvtSignalGeometry(zvt);
  const signalLabel = buildSignalLabel(zvt);

  let svg = "";

  if (leader.length >= 4) {
    svg += `<polyline points="${pointsToSvg(leader)}" fill="none" stroke="${stroke}" stroke-width="${strokeWidth}" stroke-linecap="round" stroke-linejoin="round" />`;
  }

  svg += `<circle cx="${x}" cy="${y}" r="${r}" fill="white" stroke="${stroke}" stroke-width="${strokeWidth}" />`;

  if (zvt.data?.showParameterGraphicMark ?? true) {
    const graphicScale = zvt.style?.parameterGraphicScale ?? 1;
    svg += renderParameterGraphicMarkSvg(zvt, {
      x,
      y: y - r - 12 * graphicScale,
      stroke,
      strokeWidth: 1.6,
    });
  }

  if (installPlace.hasDivider) {
    svg += `<line x1="${x - r}" y1="${y}" x2="${x + r}" y2="${y}" stroke="${stroke}" stroke-width="${strokeWidth}" />`;
    svg += `<text x="${x}" y="${y - 8}" text-anchor="middle" dominant-baseline="middle" font-family="${fontFamily}" font-size="${fontSize}" font-weight="700">${esc(functionLabel)}</text>`;
    svg += `<text x="${x}" y="${y + 12}" text-anchor="middle" dominant-baseline="middle" font-family="${fontFamily}" font-size="${fontSize}" font-weight="700">${esc(parameterLabel)}</text>`;
  } else {
    // "За місцем": функціональна ознака + вид параметру без горизонтальної лінії.
    if (functionLabel && parameterLabel) {
      svg += `<text x="${x}" y="${y - 7}" text-anchor="middle" dominant-baseline="middle" font-family="${fontFamily}" font-size="${fontSize}" font-weight="700">${esc(functionLabel)}</text>`;
      svg += `<text x="${x}" y="${y + 8}" text-anchor="middle" dominant-baseline="middle" font-family="${fontFamily}" font-size="${fontSize}" font-weight="700">${esc(parameterLabel)}</text>`;
    } else {
      const single = functionLabel || parameterLabel;
      svg += `<text x="${x}" y="${y}" text-anchor="middle" dominant-baseline="middle" font-family="${fontFamily}" font-size="${fontSize}" font-weight="700">${esc(single)}</text>`;
    }
  }

  if (signal) {
    svg += `<polyline points="${pointsToSvg(signal.line)}" fill="none" stroke="black" stroke-width="2" stroke-dasharray="18 8" />`;
    svg += `<polygon points="${pointsToSvg(signal.arrowPoints)}" fill="black" stroke="black" stroke-width="1" />`;
    svg += `<text x="${signal.labelX}" y="${signal.labelY + 13}" font-family="${fontFamily}" font-size="${fontSize}" font-weight="700">${esc(signalLabel)}</text>`;
  }

  return svg;
}

export function renderZvtsSvg(scene) {
  return (scene.zvts || []).map((zvt) => renderZvtSvg(zvt, scene)).join("\n");
}
