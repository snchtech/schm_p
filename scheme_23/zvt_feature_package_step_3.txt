export function createZvt({
  id = crypto.randomUUID(),
  attachedTo,
  anchor = {},
  position = {},
  data = {},
  route = {},
  style = {},
} = {}) {
  return {
    id,
    kind: "zvt",

    attachedTo: attachedTo || {
      type: "none",
      id: null,
      segmentId: null,
    },

    // anchor.x/y є кешем/резервом.
    // Для node фактична точка щоразу обчислюється з актуальної геометрії node.
    anchor: {
      x: anchor.x ?? 0,
      y: anchor.y ?? 0,
      side: anchor.side ?? null,
    },

    position: {
      x: position.x ?? 0,
      y: position.y ?? 0,
    },

    route: {
      type: route.type || "orthogonal",
      vertices: Array.isArray(route.vertices) ? route.vertices : [],

      // Сторона ВИХОДУ лінії з node:
      // auto | top | bottom | left | right.
      sourceSide:
        route.sourceSide ||
        route.preferredSide ||
        "auto",

      // Сторона ВХОДУ лінії у кружечок ЗВТ:
      // auto | top | bottom | left | right.
      symbolSide: route.symbolSide || "auto",

      // Короткий прямий відрізок від node перед першим поворотом.
      // Не дає ламаній "лягати" прямо на край фігури.
      sourceOffset: route.sourceOffset ?? 18,

      // Аналогічний короткий відрізок перед входом у кружечок.
      symbolOffset: route.symbolOffset ?? 12,

      // preferredSide лишаємо тільки для backward compatibility.
      preferredSide: route.preferredSide || route.sourceSide || "auto",
    },

    data: {
      installPlace: data.installPlace || "local",

      parameter: data.parameter || "F",
      showParameterGraphicMark: data.showParameterGraphicMark ?? true,
      parameterSuffix: data.parameterSuffix || "",

      functionSigns: Array.isArray(data.functionSigns) ? data.functionSigns : [],
      functionSuffix: data.functionSuffix || "",

      signal: {
        enabled: data.signal?.enabled ?? false,
        direction: data.signal?.direction || "right",
      },

      recipients: Array.isArray(data.recipients) ? data.recipients : [],
      recipientLabelMode: data.recipientLabelMode || "M",
      recipientCustomText: data.recipientCustomText || "",

      actionSign: data.actionSign || "none",
      actionSignPosition: data.actionSignPosition || "before",

      tag: data.tag || "",
      comment: data.comment || "",
    },

    style: {
      radius: style.radius ?? 22,
      stroke: style.stroke || "black",
      strokeWidth: style.strokeWidth ?? 2,
      fill: style.fill || "white",
      fontFamily: style.fontFamily || "Roboto Condensed",
      fontSize: style.fontSize ?? 13,
      smallFontSize: style.smallFontSize ?? 12,
      selectedStroke: style.selectedStroke || "#2F80ED",

      arrowSize: style.arrowSize ?? 8,
      signalLength: style.signalLength ?? 82,
      parameterGraphicScale: style.parameterGraphicScale ?? 1,
    },

    createdAt: Date.now(),
    updatedAt: Date.now(),
  };
}

export function updateZvtData(
  zvt,
  nextData,
  nextStyle = null,
  nextRoute = null
) {
  return {
    ...zvt,

    data: {
      ...zvt.data,
      ...nextData,
      signal: {
        ...zvt.data?.signal,
        ...nextData?.signal,
      },
    },

    style: nextStyle
      ? {
          ...zvt.style,
          ...nextStyle,
        }
      : zvt.style,

    route: nextRoute
      ? {
          ...zvt.route,
          ...nextRoute,
          preferredSide:
            nextRoute.sourceSide ??
            nextRoute.preferredSide ??
            zvt.route?.preferredSide ??
            "auto",
        }
      : zvt.route,

    updatedAt: Date.now(),
  };
}

  return {
    ...zvt,
    position: {
      x: position.x,
      y: position.y,
    },
    updatedAt: Date.now(),
  };
}

const VALID_SIDES = new Set(["top", "bottom", "left", "right"]);

function clamp(value, min, max) {
  return Math.max(min, Math.min(max, value));
}

function isValidSide(side) {
  return VALID_SIDES.has(side);
}

function sideVector(side) {
  switch (side) {
    case "top":
      return { x: 0, y: -1 };
    case "bottom":
      return { x: 0, y: 1 };
    case "left":
      return { x: -1, y: 0 };
    case "right":
      return { x: 1, y: 0 };
    default:
      return { x: 0, y: 0 };
  }
}

/**
 * Канонічні bounds node.
 *
 * Пріоритет:
 * 1. node.geometry.bounds
 * 2. node.bounds
 * 3. node.position + node.size
 * 4. node.x/y + node.width/height
 *
 * Якщо у твоєму NodesLayer position є ЦЕНТРОМ, а не top-left,
 * вистав node.positionOrigin = "center".
 *
 * Це прибирає типову причину зміщеного anchor:
 * різне трактування node.position у NodesLayer і ZVT geometry.
 */
export function getZvtNodeBounds(node) {
  const explicit =
    node?.geometry?.bounds ||
    node?.bounds ||
    null;

  if (explicit) {
    const x = explicit.x ?? explicit.left ?? 0;
    const y = explicit.y ?? explicit.top ?? 0;
    const width =
      explicit.width ??
      ((explicit.right ?? x) - x) ??
      120;
    const height =
      explicit.height ??
      ((explicit.bottom ?? y) - y) ??
      60;

    return {
      x,
      y,
      width,
      height,
      left: x,
      top: y,
      right: x + width,
      bottom: y + height,
      centerX: x + width / 2,
      centerY: y + height / 2,
    };
  }

  const width = node?.size?.width ?? node?.width ?? 120;
  const height = node?.size?.height ?? node?.height ?? 60;

  let x = node?.position?.x ?? node?.x ?? 0;
  let y = node?.position?.y ?? node?.y ?? 0;

  if (node?.positionOrigin === "center") {
    x -= width / 2;
    y -= height / 2;
  }

  return {
    x,
    y,
    width,
    height,
    left: x,
    top: y,
    right: x + width,
    bottom: y + height,
    centerX: x + width / 2,
    centerY: y + height / 2,
  };
}

/**
 * Якщо node вже має свої точні anchors — використовуємо їх.
 *
 * Підтримувані варіанти:
 *
 * node.connectionAnchors = {
 *   top: { x, y },
 *   bottom: { x, y },
 *   left: { x, y },
 *   right: { x, y },
 * }
 *
 * або node.anchors з такою самою структурою.
 *
 * Це найкращий варіант для НЕпрямокутних node:
 * circle, valve, custom SVG, складна Konva-фігура тощо.
 */
function getExplicitNodeAnchor(node, side) {
  const anchors = node?.connectionAnchors || node?.anchors;
  const point = anchors?.[side];

  if (
    point &&
    Number.isFinite(point.x) &&
    Number.isFinite(point.y)
  ) {
    return {
      x: point.x,
      y: point.y,
      side,
    };
  }

  return null;
}

function getNearestNodeSide(bounds, target) {
  const candidates = [
    {
      side: "left",
      d: Math.abs(target.x - bounds.left),
    },
    {
      side: "right",
      d: Math.abs(target.x - bounds.right),
    },
    {
      side: "top",
      d: Math.abs(target.y - bounds.top),
    },
    {
      side: "bottom",
      d: Math.abs(target.y - bounds.bottom),
    },
  ];

  candidates.sort((a, b) => a.d - b.d);
  return candidates[0].side;
}

/**
 * Точка на ФАКТИЧНОМУ краї прямокутних bounds.
 *
 * Для left/right проєктуємо y символу на вертикальну сторону.
 * Для top/bottom проєктуємо x символу на горизонтальну сторону.
 *
 * Тому лінія не "стрибає" в центр сторони і не виглядає зміщеною.
 */
function projectPointToNodeSide(bounds, side, target) {
  switch (side) {
    case "left":
      return {
        x: bounds.left,
        y: clamp(target.y, bounds.top, bounds.bottom),
        side,
      };

    case "right":
      return {
        x: bounds.right,
        y: clamp(target.y, bounds.top, bounds.bottom),
        side,
      };

    case "top":
      return {
        x: clamp(target.x, bounds.left, bounds.right),
        y: bounds.top,
        side,
      };

    case "bottom":
      return {
        x: clamp(target.x, bounds.left, bounds.right),
        y: bounds.bottom,
        side,
      };

    default:
      return {
        x: bounds.centerX,
        y: bounds.centerY,
        side: null,
      };
  }
}

function getEdgeCenter(edge) {
  const points = edge?.geometry?.points || [];

  if (points.length >= 4) {
    const middleIndex = Math.max(
      0,
      Math.floor(points.length / 4) * 2 - 2
    );

    const x1 = points[middleIndex] ?? points[0];
    const y1 = points[middleIndex + 1] ?? points[1];
    const x2 = points[middleIndex + 2] ?? points[points.length - 2];
    const y2 = points[middleIndex + 3] ?? points[points.length - 1];

    return {
      x: (x1 + x2) / 2,
      y: (y1 + y2) / 2,
    };
  }

  return {
    x: edge?.geometry?.start?.x ?? edge?.start?.x ?? 0,
    y: edge?.geometry?.start?.y ?? edge?.start?.y ?? 0,
  };
}

function getEdgeSegmentCenter(edge, segmentId) {
  const segment = edge?.geometry?.segmentsMeta?.find(
    (seg) => seg.id === segmentId
  );

  if (!segment) return getEdgeCenter(edge);

  return {
    x: (segment.x1 + segment.x2) / 2,
    y: (segment.y1 + segment.y2) / 2,
  };
}

/**
 * Визначає anchor ЗВТ.
 *
 * NODE:
 * - route.sourceSide = auto => найближча сторона;
 * - route.sourceSide = top/bottom/left/right => чітко задана сторона.
 *
 * EDGE:
 * - поточна логіка segmentId/center збережена.
 */
export function resolveZvtAnchor(zvt, nodes = [], edges = []) {
  if (!zvt?.attachedTo) {
    return {
      x: zvt?.anchor?.x ?? 0,
      y: zvt?.anchor?.y ?? 0,
      side: zvt?.anchor?.side ?? null,
    };
  }

  if (zvt.attachedTo.type === "node") {
    const node = nodes.find(
      (item) => item.id === zvt.attachedTo.id
    );

    if (!node) {
      return zvt.anchor || {
        x: 0,
        y: 0,
        side: null,
      };
    }

    const bounds = getZvtNodeBounds(node);
    const target = zvt.position || {
      x: bounds.centerX + 100,
      y: bounds.centerY,
    };

    const requestedSide =
      zvt.route?.sourceSide ||
      zvt.route?.preferredSide ||
      "auto";

    const resolvedSide = isValidSide(requestedSide)
      ? requestedSide
      : getNearestNodeSide(bounds, target);

    // Для складних фігур NodesLayer може дати точний anchor.
    const explicitAnchor = getExplicitNodeAnchor(
      node,
      resolvedSide
    );

    if (explicitAnchor) {
      return explicitAnchor;
    }

    return projectPointToNodeSide(
      bounds,
      resolvedSide,
      target
    );
  }

  if (zvt.attachedTo.type === "edge") {
    const edge = edges.find(
      (item) => item.id === zvt.attachedTo.id
    );

    if (!edge) {
      return zvt.anchor || {
        x: 0,
        y: 0,
        side: null,
      };
    }

    const center = zvt.attachedTo.segmentId
      ? getEdgeSegmentCenter(
          edge,
          zvt.attachedTo.segmentId
        )
      : getEdgeCenter(edge);

    return {
      x: center.x,
      y: center.y,
      side: null,
    };
  }

  return zvt.anchor || {
    x: 0,
    y: 0,
    side: null,
  };
}

/**
 * Початкова позиція нового ЗВТ тепер залежить від вибраної сторони node.
 */
export function createInitialZvtPosition(
  anchor,
  sourceSide = "auto"
) {
  const distance = 110;

  switch (sourceSide) {
    case "top":
      return {
        x: anchor.x,
        y: anchor.y - distance,
      };

    case "bottom":
      return {
        x: anchor.x,
        y: anchor.y + distance,
      };

    case "left":
      return {
        x: anchor.x - distance,
        y: anchor.y,
      };

    case "right":
      return {
        x: anchor.x + distance,
        y: anchor.y,
      };

    default:
      return {
        x: anchor.x + 110,
        y: anchor.y - 75,
      };
  }
}

function resolveAutoSymbolSide(anchor, symbol) {
  const dx = anchor.x - symbol.x;
  const dy = anchor.y - symbol.y;

  if (Math.abs(dx) >= Math.abs(dy)) {
    return dx < 0 ? "left" : "right";
  }

  return dy < 0 ? "top" : "bottom";
}

function getSymbolAttachPoint(
  symbol,
  radius,
  requestedSide,
  anchor
) {
  const side = isValidSide(requestedSide)
    ? requestedSide
    : resolveAutoSymbolSide(anchor, symbol);

  switch (side) {
    case "top":
      return {
        x: symbol.x,
        y: symbol.y - radius,
        side,
      };

    case "bottom":
      return {
        x: symbol.x,
        y: symbol.y + radius,
        side,
      };

    case "left":
      return {
        x: symbol.x - radius,
        y: symbol.y,
        side,
      };

    case "right":
      return {
        x: symbol.x + radius,
        y: symbol.y,
        side,
      };

    default:
      return {
        x: symbol.x,
        y: symbol.y,
        side: "left",
      };
  }
}

function dedupeOrthogonalPoints(points) {
  if (!Array.isArray(points) || points.length < 4) {
    return points || [];
  }

  const result = [];

  for (let i = 0; i < points.length; i += 2) {
    const x = points[i];
    const y = points[i + 1];

    const prevX = result[result.length - 2];
    const prevY = result[result.length - 1];

    if (prevX === x && prevY === y) continue;

    result.push(x, y);
  }

  return result;
}

/**
 * Ортогональна траса:
 *
 * node anchor
 *   -> короткий source stub
 *   -> ламана
 *   -> короткий symbol stub
 *   -> край кружечка ЗВТ
 *
 * Таким чином перший відрізок гарантовано виходить
 * перпендикулярно до вибраної сторони node.
 */
export function buildZvtLeaderRoute({
  anchor,
  symbol,
  radius = 22,
  sourceSide = null,
  symbolSide = "auto",
  sourceOffset = 18,
  symbolOffset = 12,
}) {
  if (!anchor || !symbol) return [];

  const resolvedSourceSide = isValidSide(sourceSide)
    ? sourceSide
    : isValidSide(anchor.side)
      ? anchor.side
      : null;

  const target = getSymbolAttachPoint(
    symbol,
    radius,
    symbolSide,
    anchor
  );

  // Для edge sourceSide може бути null.
  // Тоді визначаємо напрямок від edge до ЗВТ автоматично.
  const effectiveSourceSide =
    resolvedSourceSide ||
    (() => {
      const dx = symbol.x - anchor.x;
      const dy = symbol.y - anchor.y;

      if (Math.abs(dx) >= Math.abs(dy)) {
        return dx >= 0 ? "right" : "left";
      }

      return dy >= 0 ? "bottom" : "top";
    })();

  const sourceVector = sideVector(effectiveSourceSide);
  const targetVector = sideVector(target.side);

  const sourceStub = {
    x: anchor.x + sourceVector.x * sourceOffset,
    y: anchor.y + sourceVector.y * sourceOffset,
  };

  const targetStub = {
    x: target.x + targetVector.x * symbolOffset,
    y: target.y + targetVector.y * symbolOffset,
  };

  const points = [
    anchor.x,
    anchor.y,
    sourceStub.x,
    sourceStub.y,
  ];

  const sourceHorizontal =
    effectiveSourceSide === "left" ||
    effectiveSourceSide === "right";

  const targetHorizontal =
    target.side === "left" ||
    target.side === "right";

  if (sourceHorizontal && targetHorizontal) {
    const midX =
      (sourceStub.x + targetStub.x) / 2;

    points.push(
      midX,
      sourceStub.y,
      midX,
      targetStub.y
    );
  } else if (!sourceHorizontal && !targetHorizontal) {
    const midY =
      (sourceStub.y + targetStub.y) / 2;

    points.push(
      sourceStub.x,
      midY,
      targetStub.x,
      midY
    );
  } else if (sourceHorizontal && !targetHorizontal) {
    points.push(
      targetStub.x,
      sourceStub.y
    );
  } else {
    points.push(
      sourceStub.x,
      targetStub.y
    );
  }

  points.push(
    targetStub.x,
    targetStub.y,
    target.x,
    target.y
  );

  return dedupeOrthogonalPoints(points);
}

export function buildZvtSignalGeometry(zvt) {
  const radius = zvt?.style?.radius ?? 22;
  const x = zvt?.position?.x ?? 0;
  const y = zvt?.position?.y ?? 0;
  const signal = zvt?.data?.signal;

  if (!signal?.enabled) return null;

  const direction = signal.direction || "right";
  const sign = direction === "left" ? -1 : 1;
  const signalLength = Math.max(
    30,
    zvt?.style?.signalLength ?? 82
  );
  const arrowSize = Math.max(
    4,
    zvt?.style?.arrowSize ?? 8
  );

  const startX = x + sign * radius;
  const endX =
    x + sign * (radius + signalLength);
  const lineY = y;

  const arrowPoints =
    direction === "right"
      ? [
          endX,
          lineY,
          endX - arrowSize,
          lineY - arrowSize / 2,
          endX - arrowSize,
          lineY + arrowSize / 2,
        ]
      : [
          endX,
          lineY,
          endX + arrowSize,
          lineY - arrowSize / 2,
          endX + arrowSize,
          lineY + arrowSize / 2,
        ];

  const labelX =
    direction === "right"
      ? startX + 8
      : endX;

  return {
    line: [
      startX,
      lineY,
      endX,
      lineY,
    ],
    arrowPoints,
    labelX,
    labelY: y - 30,
    labelWidth: Math.abs(endX - startX),
  };
}

function toggleArrayValue(array, value) {
  if (array.includes(value)) {
    return array.filter((item) => item !== value);
  }

  return [...array, value];
}

function onlyDigits(value) {
  return value.replace(/\D/g, "");
}

export default function ZvtEditorModal() {
  const zvtEditor = useCanvasStore((s) => s.zvtEditor);
  const zvts = useCanvasStore((s) => s.zvts || []);
  const accessMode = useCanvasStore((s) => s.accessMode || "editor");

  const readOnly = accessMode === "reader";
  const open = !!zvtEditor;

  const existing = useMemo(() => {
    if (!zvtEditor?.zvtId) return null;
    return zvts.find((item) => item.id === zvtEditor.zvtId) || null;
  }, [zvtEditor, zvts]);

  const [installPlace, setInstallPlace] = useState("local");
  const [parameter, setParameter] = useState("F");

  const [parameterGraphicScale, setParameterGraphicScale] = useState(1);
  const [functionSigns, setFunctionSigns] = useState([]);
  const [functionSuffix, setFunctionSuffix] = useState("");

  const [recipientLabelMode, setRecipientLabelMode] = useState("M");
  const [recipients, setRecipients] = useState([]);
  const [recipientCustomText, setRecipientCustomText] = useState("");

  const [actionSign, setActionSign] = useState("none");
  const [actionSignPosition, setActionSignPosition] = useState("before");

  const [radius, setRadius] = useState(22);
  const [arrowSize, setArrowSize] = useState(8);
  const [signalLength, setSignalLength] = useState(82);

  const [sourceSide, setSourceSide] = useState("auto");
  const [symbolSide, setSymbolSide] = useState("auto");
  const [sourceOffset, setSourceOffset] = useState(18);
  const [symbolOffset, setSymbolOffset] = useState(12);

  const [tag, setTag] = useState("");
  const [comment, setComment] = useState("");

  useEffect(() => {
    if (!open) return;

    if (existing) {
      setInstallPlace(existing.data?.installPlace || "local");
      setParameter(existing.data?.parameter || "F");
      setParameterSuffix(existing.data?.parameterSuffix || "");
      setShowParameterGraphicMark(existing.data?.showParameterGraphicMark ?? true);
      setParameterGraphicScale(existing.style?.parameterGraphicScale ?? 1);
      setFunctionSigns(existing.data?.functionSigns || []);
      setFunctionSuffix(existing.data?.functionSuffix || "");

      setSignalDirection(
        existing.data?.signal?.enabled
          ? existing.data?.signal?.direction || "right"
          : "none"
      );

      setRecipientLabelMode(existing.data?.recipientLabelMode || "M");
      setRecipients(existing.data?.recipients || []);
      setRecipientCustomText(existing.data?.recipientCustomText || "");

      setActionSign(existing.data?.actionSign || "none");
      setActionSignPosition(existing.data?.actionSignPosition || "before");

      setRadius(existing.style?.radius ?? 22);
      setArrowSize(existing.style?.arrowSize ?? 8);
      setSignalLength(existing.style?.signalLength ?? 82);

      setSourceSide(
        existing.route?.sourceSide ||
        existing.route?.preferredSide ||
        "auto"
      );
      setSymbolSide(existing.route?.symbolSide || "auto");
      setSourceOffset(existing.route?.sourceOffset ?? 18);
      setSymbolOffset(existing.route?.symbolOffset ?? 12);

      setTag(existing.data?.tag || "");
      setComment(existing.data?.comment || "");
      return;
    }

    setInstallPlace("local");
    setParameter("F");
    setParameterSuffix("");

    setParameterGraphicScale(1);

    setActionSignPosition("before");

    zvts: (s.zvts || []).filter((zvt) => zvt.id !== id),
    selection: s.selection?.type === "zvt" && s.selection?.id === id ? null : s.selection,
  })),

clearZvtEditor: () => set({ zvtEditor: null }),

openZvtEditor({
  attachedTo = null,
  zvtId = null,
} = {}) {
  const store = useCanvasStore.getState();

  store.setZvtEditor?.({
    attachedTo,
    zvtId,
  });
},

closeZvtEditor() {
  const store = useCanvasStore.getState();
  store.clearZvtEditor?.();
},

createZvtFromEditor({
  attachedTo,
  data,
  style = {},
  route = {},
}) {
  const store = useCanvasStore.getState();

  if (
    (store.accessMode || "editor") === "reader"
  ) {
    return;
  }

  if (!attachedTo?.type || !attachedTo?.id) {
    return;
  }

  /**
   * Спочатку створюємо temp із route,
   * бо resolveZvtAnchor повинен уже знати sourceSide.
   */
  const tempZvt = createZvt({
    attachedTo,
    data,
    style,
    route,
  });

  const anchor = resolveZvtAnchor(
    tempZvt,
    store.nodes,
    store.edges
  );

  const position = createInitialZvtPosition(
    anchor,
    anchor.side
  );

  const zvt = createZvt({
    attachedTo,
    anchor,
    position,
    data,
    style,
    route,
  });

  store.addZvt?.(zvt);
  store.setSelection?.({
    type: "zvt",
    id: zvt.id,
  });
  store.clearZvtEditor?.();
},

updateZvtFromEditor(
  zvtId,
  data,
  style = {},
  route = {}
) {
  const store = useCanvasStore.getState();

  if (
    (store.accessMode || "editor") === "reader"
  ) {
    return;
  }

  const zvt = (store.zvts || []).find(
    (item) => item.id === zvtId
  );

  if (!zvt) return;

  const next = updateZvtData(
    zvt,
    data,
    style,
    route
  );

  store.replaceZvt?.(zvtId, next);
  store.clearZvtEditor?.();
},

moveZvt(zvtId, position) {
  const store = useCanvasStore.getState();

  if (
    (store.accessMode || "editor") === "reader"
  ) {
    return;
  }

  const zvt = (store.zvts || []).find(
    (item) => item.id === zvtId
  );

  if (!zvt) return;

  const next = moveZvt(
    zvt,
    position
  );

  store.replaceZvt?.(
    zvtId,
    next
  );
},

handleZvtContextMenu(e, zvt) {
  e.evt.preventDefault();

  const stage = e.target.getStage();
  const pointer = stage?.getPointerPosition();

  if (!pointer) return;

  const store = useCanvasStore.getState();

  store.setContextMenu?.({
    type: "zvt",
    zvtId: zvt.id,
    x: pointer.x,
    y: pointer.y,
  });
},

editZvtFromContextMenu() {
  const store = useCanvasStore.getState();
  const menu = store.contextMenu;

  if (
    !menu ||
    menu.type !== "zvt" ||
    !menu.zvtId
  ) {
    return;
  }

  const zvt = (store.zvts || []).find(
    (item) => item.id === menu.zvtId
  );

  if (!zvt) return;

  store.setZvtEditor?.({
    zvtId: zvt.id,
    attachedTo: zvt.attachedTo,
  });

  store.clearContextMenu?.();
},

deleteZvtFromContextMenu() {
  const store = useCanvasStore.getState();
  const menu = store.contextMenu;

  if (
    (store.accessMode || "editor") === "reader"
  ) {
    return;
  }

  if (
    !menu ||
    menu.type !== "zvt" ||
    !menu.zvtId
  ) {
    return;
  }

  store.removeZvt?.(
    menu.zvtId
  );
  store.clearContextMenu?.();
},

import { getZvtInstallPlace } from "@/lib/zvt/zvtDictionaries";
import { buildZvtLeaderRoute, buildZvtSignalGeometry, resolveZvtAnchor } from "@/lib/zvt/zvtGeometry";

function esc(value) {
  return String(value ?? "")
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;")
    .replaceAll('"', "&quot;");
}

function pointsToSvg(points) {
  const pairs = [];
  for (let i = 0; i < points.length; i += 2) {
    pairs.push(`${points[i]},${points[i + 1]}`);
  }
  return pairs.join(" ");
}

function buildParameterLabel(zvt) {
  return `${zvt.data?.parameter || ""}${zvt.data?.parameterSuffix || ""}`;
}

function buildFunctionLabel(zvt) {
  const signs = Array.isArray(zvt.data?.functionSigns)
    ? zvt.data.functionSigns.join("")
    : "";
  return `${signs}${zvt.data?.functionSuffix || ""}`;
}

function buildSignalLabel(zvt) {
  const mode = zvt.data?.recipientLabelMode || "M";

  let base = "";
  if (mode === "custom") {
    base = (zvt.data?.recipientCustomText || "").trim();
  } else {
    const recipients = Array.isArray(zvt.data?.recipients)
      ? zvt.data.recipients
      : [];
    base = recipients.length ? `M${recipients.join(",")}` : "M";
  }

  const sign = zvt.data?.actionSign || "none";
  const signPosition = zvt.data?.actionSignPosition || "before";

  if (!base || sign === "none") return base;
  return signPosition === "after"
    ? `${base} ${sign}`
    : `${sign} ${base}`;
}

export function renderZvtSvg(zvt, scene) {
  const nodes = scene.nodes || [];
  const edges = scene.edges || [];

  const x = zvt.position?.x ?? 0;
  const y = zvt.position?.y ?? 0;
  const r = zvt.style?.radius ?? 22;
  const installPlace = getZvtInstallPlace(zvt.data?.installPlace);
  const functionLabel = buildFunctionLabel(zvt);
  const parameterLabel = buildParameterLabel(zvt);
  const stroke = zvt.style?.stroke || "black";
  const strokeWidth = zvt.style?.strokeWidth ?? 2;
  const fontFamily = zvt.style?.fontFamily || "Roboto Condensed";
  const fontSize = zvt.style?.fontSize ?? 13;

  const anchor = resolveZvtAnchor(zvt, nodes, edges);
  const leader = buildZvtLeaderRoute({
    anchor,
    symbol: zvt.position,
    radius: r,
    sourceSide:
      zvt.attachedTo?.type === "node"
        ? (
            zvt.route?.sourceSide ||
            zvt.route?.preferredSide ||
            anchor.side
          )
        : null,
    symbolSide: zvt.route?.symbolSide || "auto",
    sourceOffset: zvt.route?.sourceOffset ?? 18,
    symbolOffset: zvt.route?.symbolOffset ?? 12,
  });
  const signal = buildZvtSignalGeometry(zvt);
  const signalLabel = buildSignalLabel(zvt);

  let svg = "";

  if (leader.length >= 4) {
    svg += `<polyline points="${pointsToSvg(leader)}" fill="none" stroke="${stroke}" stroke-width="${strokeWidth}" stroke-linecap="round" stroke-linejoin="round" />`;
  }

  svg += `<circle cx="${x}" cy="${y}" r="${r}" fill="white" stroke="${stroke}" stroke-width="${strokeWidth}" />`;

  if (zvt.data?.showParameterGraphicMark ?? true) {
    const graphicScale = zvt.style?.parameterGraphicScale ?? 1;
    svg += renderParameterGraphicMarkSvg(zvt, {
      x,
      y: y - r - 12 * graphicScale,
      stroke,
      strokeWidth: 1.6,
    });
  }

  if (installPlace.hasDivider) {
    svg += `<line x1="${x - r}" y1="${y}" x2="${x + r}" y2="${y}" stroke="${stroke}" stroke-width="${strokeWidth}" />`;
    svg += `<text x="${x}" y="${y - 8}" text-anchor="middle" dominant-baseline="middle" font-family="${fontFamily}" font-size="${fontSize}" font-weight="700">${esc(functionLabel)}</text>`;
    svg += `<text x="${x}" y="${y + 12}" text-anchor="middle" dominant-baseline="middle" font-family="${fontFamily}" font-size="${fontSize}" font-weight="700">${esc(parameterLabel)}</text>`;
  } else {
    // "За місцем": функціональна ознака + вид параметру без горизонтальної лінії.
    if (functionLabel && parameterLabel) {
      svg += `<text x="${x}" y="${y - 7}" text-anchor="middle" dominant-baseline="middle" font-family="${fontFamily}" font-size="${fontSize}" font-weight="700">${esc(functionLabel)}</text>`;
      svg += `<text x="${x}" y="${y + 8}" text-anchor="middle" dominant-baseline="middle" font-family="${fontFamily}" font-size="${fontSize}" font-weight="700">${esc(parameterLabel)}</text>`;
    } else {
      const single = functionLabel || parameterLabel;
      svg += `<text x="${x}" y="${y}" text-anchor="middle" dominant-baseline="middle" font-family="${fontFamily}" font-size="${fontSize}" font-weight="700">${esc(single)}</text>`;
    }
  }

  if (signal) {
    svg += `<polyline points="${pointsToSvg(signal.line)}" fill="none" stroke="black" stroke-width="2" stroke-dasharray="18 8" />`;
    svg += `<polygon points="${pointsToSvg(signal.arrowPoints)}" fill="black" stroke="black" stroke-width="1" />`;
    svg += `<text x="${signal.labelX}" y="${signal.labelY + 13}" font-family="${fontFamily}" font-size="${fontSize}" font-weight="700">${esc(signalLabel)}</text>`;
  }

  return svg;
}

export function renderZvtsSvg(scene) {
  return (scene.zvts || []).map((zvt) => renderZvtSvg(zvt, scene)).join("\n");
}

// ============================================================================
// STEP 3 — ОБОВ'ЯЗКОВА ІНТЕГРАЦІЙНА ПЕРЕВІРКА NODE GEOMETRY
// ============================================================================
//
// Пункт 6 виправляє геометрію на стороні ЗВТ, але є одна важлива умова:
//
// ZvtLayer та NodesLayer повинні однаково трактувати координати node.
//
// Варіант A — position = TOP-LEFT:
// --------------------------------
// node = {
//   id: "...",
//   position: { x: 100, y: 200 },
//   size: { width: 120, height: 60 },
// }
//
// Нічого додатково не потрібно.
//
// Варіант B — position = CENTER:
// ------------------------------
// node = {
//   id: "...",
//   position: { x: 160, y: 230 },
//   size: { width: 120, height: 60 },
//   positionOrigin: "center",
// }
//
// ВАЖЛИВО:
// якщо NodesLayer реально малює Group/Circle/Rect від центру,
// але store не містить positionOrigin="center",
// anchor буде зміщений на width/2 та height/2.
//
// ---------------------------------------------------------------------------
// РЕКОМЕНДОВАНО ДЛЯ СКЛАДНИХ ФІГУР
// ---------------------------------------------------------------------------
//
// Якщо node не прямокутний (circle, valve, custom SVG, equipment shape),
// НЕ треба намагатися вгадувати точку за bounding box.
//
// Додай у модель node точні connectionAnchors:
//
// node = {
//   ...,
//   connectionAnchors: {
//     top:    { x: 160, y: 200 },
//     bottom: { x: 160, y: 260 },
//     left:   { x: 100, y: 230 },
//     right:  { x: 220, y: 230 },
//   },
// }
//
// resolveZvtAnchor() автоматично використає їх першими.
//
// Це дає один і той самий контракт для:
// - звичайних connectors;
// - ЗВТ;
// - майбутніх групових ZVT branches;
// - будь-яких інших attached elements.
//
// ---------------------------------------------------------------------------
// ЩО ВІДБУВАЄТЬСЯ ПРИ DRAG NODE
// ---------------------------------------------------------------------------
//
// ZvtLayer читає актуальний nodes[] із Zustand.
// resolveZvtAnchor() виконується на кожному render.
// Тому при оновленні node.position:
// 1. anchor переходить разом із node;
// 2. ламана перебудовується;
// 3. сам ZVT лишається там, де його розмістив користувач.
//
// ---------------------------------------------------------------------------
// ЩО ВІДБУВАЄТЬСЯ ПРИ DRAG ZVT
// ---------------------------------------------------------------------------
//
// sourceSide = "auto":
//   anchor може автоматично перейти на найближчу сторону node.
//
// sourceSide = top/bottom/left/right:
//   сторона ЗАФІКСОВАНА.
//   При drag ЗВТ точка рухається тільки вздовж обраної сторони.
//
// Це саме та поведінка, яка потрібна для ручного контролю схеми.
//
// ---------------------------------------------------------------------------
// BACKWARD COMPATIBILITY
// ---------------------------------------------------------------------------
//
// Старі zvt:
// route.preferredSide => автоматично підхоплюється як route.sourceSide.
// Якщо поля немає:
// sourceSide = "auto"
// symbolSide = "auto"
// sourceOffset = 18
// symbolOffset = 12
//
// Старі JSON не потрібно мігрувати окремим скриптом.
// ============================================================================
