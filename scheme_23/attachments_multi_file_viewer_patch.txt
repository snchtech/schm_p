ATTACHMENTS FEATURE — MULTI-FILE VIEWER PATCH
=============================================

Призначення
-----------
Цей пакет є доопрацюванням наданого рішення для Next.js + React + MUI + Konva.

Реалізовано:
1. Один блакитний badge для одного node або edge незалежно від кількості attachments.
2. Відкриття одного модального вікна зі списком усіх файлів, прикріплених до об'єкта.
3. Перемикання між файлами в межах модального вікна.
4. Видалення тільки вибраного attachment через кнопку в модальному вікні.
5. Видалення фізичного файлу через DELETE API перед видаленням metadata зі store.
6. Масштабування зображень від 25% до 400%.
7. Автоматичне перемикання на сусідній файл після видалення поточного.
8. Автоматичне закриття viewer, коли видалено останній attachment об'єкта.

Важлива зміна моделі viewer
----------------------------
Було:

attachmentViewer: {
  attachmentId: "..."
}

Стало:

attachmentViewer: {
  attachedTo: {
    type: "node" | "edge",
    id: "..."
  },
  activeAttachmentId: "..."
}

Тобто viewer відкривається для об'єкта, а не для одного файла.

function getNodePosition(node) {
  const x = node?.position?.x ?? node?.x ?? 0;
  const y = node?.position?.y ?? node?.y ?? 0;
  const width = node?.size?.width ?? node?.width ?? 120;
  const height = node?.size?.height ?? node?.height ?? 60;

  return { x, y, width, height };
}

function getEdgeCenter(edge) {
  const pts = edge?.geometry?.points || [];

  if (pts.length >= 4) {
    const midIndex = Math.floor((pts.length / 2 - 1) / 2) * 2;
    const x1 = pts[midIndex] ?? pts[0];
    const y1 = pts[midIndex + 1] ?? pts[1];
    const x2 = pts[midIndex + 2] ?? pts[pts.length - 2];
    const y2 = pts[midIndex + 3] ?? pts[pts.length - 1];

    return {
      x: (x1 + x2) / 2,
      y: (y1 + y2) / 2,
    };
  }

  const start = edge?.geometry?.start || edge?.start;
  const end = edge?.geometry?.end || edge?.end;

  if (start && end) {
    return {
      x: (start.x + end.x) / 2,
      y: (start.y + end.y) / 2,
    };
  }

  return { x: 0, y: 0 };
}

export function getAttachmentTargetKey(attachedTo) {
  if (!attachedTo?.type || !attachedTo?.id) return "";
  return `${attachedTo.type}:${attachedTo.id}`;
}

export function isSameAttachmentTarget(left, right) {
  return (
    left?.type === right?.type &&
    String(left?.id) === String(right?.id)
  );
}

export function groupAttachmentsByTarget(attachments = []) {
  const groupsMap = new Map();

  for (const attachment of attachments) {
    const attachedTo = attachment?.attachedTo;
    const key = getAttachmentTargetKey(attachedTo);

    if (!key) continue;

    if (!groupsMap.has(key)) {
      groupsMap.set(key, {
        key,
        attachedTo: {
          type: attachedTo.type,
          id: attachedTo.id,
        },
        attachments: [],
      });
    }

    groupsMap.get(key).attachments.push(attachment);
  }

  return Array.from(groupsMap.values());
}

export function resolveAttachmentBadgePosition({
  attachedTo,
  nodes = [],
  edges = [],
}) {
  const targetType = attachedTo?.type;
  const targetId = attachedTo?.id;

  if (targetType === "node") {
    const node = nodes.find((item) => String(item.id) === String(targetId));
    if (!node) return null;

    const rect = getNodePosition(node);

    return {
      x: rect.x + rect.width + 8,
      y: rect.y + 8,
    };
  }

  if (targetType === "edge") {
    const edge = edges.find((item) => String(item.id) === String(targetId));
    if (!edge) return null;

    const center = getEdgeCenter(edge);

    return {
      x: center.x + 10,
      y: center.y - 10,
    };
  }

  return null;
}

  if (store.accessMode === "reader") {

    },
  });

closeAttachmentEditor() {
  useCanvasStore.getState().clearAttachmentEditor?.();

openAttachmentViewer(attachedTo, activeAttachmentId = null) {
  const store = useCanvasStore.getState();

  if (!attachedTo?.type || !attachedTo?.id) return;

  const targetAttachments = (store.attachments || []).filter((attachment) =>
    isSameAttachmentTarget(attachment?.attachedTo, attachedTo)
  );

  if (!targetAttachments.length) return;

  const requestedAttachmentExists = targetAttachments.some(
    (attachment) => attachment.id === activeAttachmentId
  );

  store.setAttachmentViewer?.({
    attachedTo: {
      type: attachedTo.type,
      id: attachedTo.id,
    },
    activeAttachmentId: requestedAttachmentExists
      ? activeAttachmentId
      : targetAttachments[0].id,
  });
},

setActiveViewerAttachment(attachmentId) {
  const store = useCanvasStore.getState();
  const viewer = store.attachmentViewer;

  if (!viewer?.attachedTo || !attachmentId) return;

  const belongsToViewerTarget = (store.attachments || []).some(
    (attachment) =>
      attachment.id === attachmentId &&
      isSameAttachmentTarget(
        attachment?.attachedTo,
        viewer.attachedTo
      )
  );

  if (!belongsToViewerTarget) return;

  store.setActiveViewerAttachment?.(attachmentId);
},

closeAttachmentViewer() {
  useCanvasStore.getState().clearAttachmentViewer?.();
},

async uploadAttachment({ attachedTo, file, comment }) {
  const store = useCanvasStore.getState();

  if (store.accessMode === "reader") {
    return null;
  }

  if (!attachedTo?.type || !attachedTo?.id || !file) return null;

  const formData = new FormData();
  formData.append("file", file);
  formData.append("attachedToType", attachedTo.type);
  formData.append("attachedToId", attachedTo.id);
  formData.append("comment", comment || "");

  const response = await fetch("/api/attachments/upload", {
    method: "POST",
    body: formData,
  });

  if (!response.ok) {
    const text = await response.text();
    throw new Error(text || "Attachment upload failed");
  }

  const result = await response.json();

  if (result?.attachment) {
    store.addAttachment?.(result.attachment);
    store.clearAttachmentEditor?.();
    return result.attachment;
  }

  return null;
},

async deleteAttachment(attachmentId) {
  const storeBeforeDelete = useCanvasStore.getState();

  if (storeBeforeDelete.accessMode === "reader") {
    return false;
  }

  const attachmentToDelete = (storeBeforeDelete.attachments || []).find(
    (attachment) => attachment.id === attachmentId
  );

  if (!attachmentToDelete) {
    return false;
  }

  const viewerBeforeDelete = storeBeforeDelete.attachmentViewer;

  const response = await fetch(`/api/attachments/${attachmentId}`, {
    method: "DELETE",
  });

  if (!response.ok) {
    const text = await response.text();
    throw new Error(text || "Attachment delete failed");
  }

  // Видаляємо metadata зі store тільки після успішної відповіді сервера.
  useCanvasStore.getState().removeAttachment?.(attachmentId);

  const storeAfterDelete = useCanvasStore.getState();

  if (
    !viewerBeforeDelete?.attachedTo ||
    !isSameAttachmentTarget(
      attachmentToDelete.attachedTo,
      viewerBeforeDelete.attachedTo
    )
  ) {
    return true;
  }

  const remainingTargetAttachments = (storeAfterDelete.attachments || []).filter(
    (attachment) =>
      isSameAttachmentTarget(
        attachment?.attachedTo,
        viewerBeforeDelete.attachedTo
      )
  );

  if (!remainingTargetAttachments.length) {
    storeAfterDelete.clearAttachmentViewer?.();
    return true;
  }

  // Якщо видалено неактивний файл, активний залишається без змін.
  if (viewerBeforeDelete.activeAttachmentId !== attachmentId) {
    return true;
  }

  const deletedIndex = (storeBeforeDelete.attachments || [])
    .filter((attachment) =>
      isSameAttachmentTarget(
        attachment?.attachedTo,
        viewerBeforeDelete.attachedTo
      )
    )
    .findIndex((attachment) => attachment.id === attachmentId);

  const nextIndex = Math.min(
    Math.max(deletedIndex, 0),
    remainingTargetAttachments.length - 1
  );

  storeAfterDelete.setActiveViewerAttachment?.(
    remainingTargetAttachments[nextIndex].id
  );

  return true;
},

"use client";

import React from "react";
import { Circle, Group, Text } from "react-konva";

import { useCanvasStore } from "@/store/useCanvasStore";
import { canvasController } from "@/controllers/canvasController";
import {
  groupAttachmentsByTarget,
  resolveAttachmentBadgePosition,
} from "@/lib/attachments/attachmentGeometry";

export default function AttachmentBadgesLayer() {
  const attachments = useCanvasStore((state) => state.attachments || []);
  const nodes = useCanvasStore((state) => state.nodes || []);
  const edges = useCanvasStore((state) => state.edges || []);

  if (!attachments.length) return null;

  const attachmentGroups = groupAttachmentsByTarget(attachments);

  return (
    <>
      {attachmentGroups.map((group) => {
        const position = resolveAttachmentBadgePosition({
          attachedTo: group.attachedTo,
          nodes,
          edges,
        });

        if (!position) return null;

        const attachmentCount = group.attachments.length;
        const countLabel =
          attachmentCount > 99 ? "99+" : String(attachmentCount);

        return (
          <Group
            key={group.key}
            x={position.x}
            y={position.y}
            onMouseDown={(event) => {
              event.cancelBubble = true;
            }}
            onTouchStart={(event) => {
              event.cancelBubble = true;
            }}
            onClick={(event) => {
              event.cancelBubble = true;

              canvasController.openAttachmentViewer?.(
                group.attachedTo,
                group.attachments[0]?.id
              );
            }}
            onTap={(event) => {
              event.cancelBubble = true;

              canvasController.openAttachmentViewer?.(
                group.attachedTo,
                group.attachments[0]?.id
              );
            }}
          >
            <Circle
              radius={11}
              fill="#2D9CDB"
              stroke="white"
              strokeWidth={2}
              shadowColor="black"
              shadowOpacity={0.18}
              shadowBlur={3}
            />

            <Text
              x={attachmentCount >= 10 ? -7.5 : -3.5}
              y={-5}
              width={attachmentCount >= 10 ? 15 : 7}
              text={countLabel}
              align="center"
              fontSize={attachmentCount > 99 ? 7 : 10}
              fontStyle="bold"
              fill="white"
              listening={false}
            />
          </Group>
        );
      })}
    </>
  );
}

function formatFileSize(size = 0) {
  const bytes = Number(size || 0);

  if (bytes < 1024) return `${bytes} B`;
  if (bytes < 1024 * 1024) return `${Math.round(bytes / 1024)} KB`;

  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
}

function getTargetTitle(attachedTo) {
  if (attachedTo?.type === "node") {
    return `Файли елемента ${attachedTo.id}`;

  if (attachedTo?.type === "edge") {
    return `Файли з'єднання ${attachedTo.id}`;

  return "Прикріплені файли";
}

                            }
                            aria-label="Зменшити зображення"
                          >
                            <ZoomOutIcon />
                          </IconButton>
                        </span>
                      </Tooltip>

                      <Button
                        onClick={handleResetZoom}
                        startIcon={<RestartAltIcon />}
                        sx={{ minWidth: 100 }}
                      >
                        {Math.round(imageZoom * 100)}%
                      </Button>

                      <Tooltip title="Збільшити">
                        <span>
                          <IconButton
                            onClick={handleZoomIn}
                            disabled={
                              imageZoom >= MAX_IMAGE_ZOOM
                            }
                            aria-label="Збільшити зображення"
                          >
                            <ZoomInIcon />
                          </IconButton>
                        </span>
                      </Tooltip>
                    </>
                  )}

                  <Button
                    component="a"
                    href={downloadUrl}
                    target="_blank"
                    rel="noreferrer"
                    startIcon={<DownloadIcon />}
                  >
                    Відкрити / завантажити
                  </Button>

                  {!readOnly && (
                    <Button
                      color="error"
                      variant="outlined"
                      startIcon={<DeleteOutlineIcon />}
                      disabled={deleting}
                      onClick={() =>
                        requestDelete(activeAttachment.id)
                      }
                    >
                      Видалити файл
                    </Button>
                  )}
                </Stack>
              </Stack>
            </Paper>

            {activeAttachment.comment && (
              <Paper
                variant="outlined"
                sx={{ p: 1.5 }}
              >
                <Typography variant="subtitle2">
                  Коментар
                </Typography>

                <Typography
                  variant="body2"
                  sx={{ whiteSpace: "pre-wrap" }}
                >
                  {activeAttachment.comment}
                </Typography>
              </Paper>
            )}

            <Divider />

            <Box
              sx={{
                flex: 1,
                minHeight: 0,
                overflow: "hidden",
              }}
            >
              {previewKind === "pdf" && (
                <Box
                  component="iframe"
                  key={activeAttachment.id}
                  src={previewUrl}
                  title={activeAttachment.file.originalName}
                  sx={{
                    width: "100%",
                    height: "100%",
                    minHeight: 450,
                    border: "1px solid",
                    borderColor: "divider",
                    borderRadius: 1,
                  }}
                />
              )}

              {previewKind === "image" && (
                <Box
                  onWheel={handleImageWheel}
                  sx={{
                    width: "100%",
                    height: "100%",
                    minHeight: 450,
                    overflow: "auto",
                    border: "1px solid",
                    borderColor: "divider",
                    borderRadius: 1,
                    backgroundColor: "#fafafa",
                  }}
                >
                  <Box
                    sx={{
                      minWidth: "100%",
                      minHeight: "100%",
                      display: "flex",
                      alignItems: "flex-start",
                      justifyContent: "center",
                      p: 2,
                    }}
                  >
                    <Box
                      component="img"
                      key={activeAttachment.id}
                      src={previewUrl}
                      alt={activeAttachment.file.originalName}
                      draggable={false}
                      sx={{
                        display: "block",
                        width: `${imageZoom * 100}%`,
                        maxWidth: "none",
                        height: "auto",
                        objectFit: "contain",
                        userSelect: "none",
                      }}
                    />
                  </Box>
                </Box>
              )}

              {previewKind === "download-only" && (
                <Paper
                  variant="outlined"
                  sx={{
                    p: 4,
                    minHeight: 300,
                    display: "flex",
                    alignItems: "center",
                    justifyContent: "center",
                    textAlign: "center",
                  }}
                >
                  <Box>
                    <Typography
                      variant="h6"
                      sx={{ mb: 1 }}
                    >
                      Попередній перегляд недоступний
                    </Typography>

                    <Typography
                      variant="body2"
                      color="text.secondary"
                      sx={{ mb: 2 }}
                    >
                      Цей тип файлу потрібно завантажити
                      та відкрити у зовнішній програмі.
                    </Typography>

                    <Button
                      component="a"
                      href={downloadUrl}
                      target="_blank"
                      rel="noreferrer"
                      variant="contained"
                      startIcon={<DownloadIcon />}
                    >
                      Завантажити файл
                    </Button>
                  </Box>
                </Paper>
              )}
            </Box>
          </Box>
        </DialogContent>

        <DialogActions>
          <Button
            onClick={handleClose}
            variant="contained"
            disabled={deleting}
          >
            Закрити
          </Button>
        </DialogActions>
      </Dialog>

      <Dialog
        open={Boolean(deleteCandidateId)}
        onClose={cancelDelete}
        maxWidth="xs"
        fullWidth
      >
        <DialogTitle>
          Видалити прикріплений файл?
        </DialogTitle>

        <DialogContent dividers>
          <Typography variant="body2">
            Файл буде видалено зі схеми та з файлового
            сховища. Цю дію неможливо скасувати.
          </Typography>
        </DialogContent>

        <DialogActions>
          <Button
            onClick={cancelDelete}
            disabled={deleting}
          >
            Скасувати
          </Button>

          <Button
            onClick={confirmDelete}
            color="error"
            variant="contained"
            disabled={deleting}
          >
            {deleting ? "Видалення..." : "Видалити"}
          </Button>
        </DialogActions>
      </Dialog>
    </>
  );
}

import { NextResponse } from "next/server";
import fs from "fs/promises";
import path from "path";

function sanitizeId(value = "") {
  return String(value).replace(/[^a-zA-Z0-9_-]/g, "");
}

export async function DELETE(_request, context) {
  try {
    const rawId = context?.params?.id || "";
    const id = sanitizeId(rawId);

    if (!id || id !== rawId) {
      return new NextResponse("Invalid attachment id", {
        status: 400,
      });
    }

    const attachmentDir = path.join(
      process.cwd(),
      "uploads",
      "attachments",
      id
    );

    try {
      await fs.access(attachmentDir);
    } catch {
      return new NextResponse("Attachment not found", {
        status: 404,
      });
    }

    await fs.rm(attachmentDir, {
      recursive: true,
      force: false,
    });

    return NextResponse.json({
      ok: true,
      attachmentId: id,
    });
  } catch (error) {
    console.error("Attachment delete error:", error);

    return new NextResponse(
      "Attachment delete failed",
      { status: 500 }
    );
  }
}

import fs from "fs/promises";

function sanitizeId(value = "") {

}

export default async function handler(req, res) {
  if (req.method !== "DELETE") {
    res.setHeader("Allow", ["DELETE"]);
    res.status(405).send("Method not allowed");
    return;
  }

  try {
    const rawId = String(req.query.id || "");
    const id = sanitizeId(rawId);

    if (!id || id !== rawId) {
      res.status(400).send("Invalid attachment id");
      return;
    }

    const attachmentDir = path.join(
      process.cwd(),
      "uploads",
      "attachments",
      id
    );

    try {
      await fs.access(attachmentDir);
    } catch {
      res.status(404).send("Attachment not found");
      return;
    }

    await fs.rm(attachmentDir, {
      recursive: true,
      force: false,
    });

    res.status(200).json({
      ok: true,
      attachmentId: id,
    });
  } catch (error) {
    console.error("Attachment delete error:", error);
    res.status(500).send("Attachment delete failed");
  }
}

13) IMPORTANT INTEGRATION NOTES
================================================================================

1. Не викликайте старий метод:

canvasController.openAttachmentViewer(attachment.id)

Новий виклик:

canvasController.openAttachmentViewer(
  attachment.attachedTo,
  attachment.id
)

або для групи:

canvasController.openAttachmentViewer(group.attachedTo)

2. Стару функцію groupAttachmentIndexes потрібно видалити. Вона створювала
окрему позицію для кожного файла, що суперечить вимозі одного badge.

3. Старий параметр indexForTarget у resolveAttachmentBadgePosition більше
не потрібний.

4. Не видаляйте metadata зі store до успішного DELETE API. Інакше при
серверній помилці UI покаже, що файл видалено, хоча він залишиться у сховищі.

5. У режимі reader:
   - список і preview доступні;
   - кнопки видалення приховані;
   - upload залишається заблокованим.

6. Поточна файлова реалізація не має транзакції між файловою системою та JSON
схеми. Якщо JSON схеми зберігається окремою операцією, після видалення файла
потрібно запустити ваш звичайний механізм позначення документа як changed/dirty.

Наприклад, після store.removeAttachment:

storeAfterDelete.markDocumentDirty?.();

або іншу дію вашого проєкту.

7. Якщо useCanvasStore має middleware persist, переконайтеся, що
attachmentViewer та attachmentEditor не записуються до постійного стану.
Це тимчасовий UI state.

8. SVG з недовірених джерел варто віддавати з правильною Content Security
Policy або конвертувати на сервері, оскільки SVG може містити активний вміст.

9. DELETE API потрібно захистити тією самою авторизацією та перевіркою доступу
до схеми, що й upload API. Наведений код демонструє файлову операцію, але не
реалізує бізнес-авторизацію.

10. Для Oracle BLOB frontend-контракт не змінюється. Метод DELETE повинен
видаляти рядок attachment/BLOB у БД, після чого повертати:

{
  "ok": true,
  "attachmentId": "..."
}
