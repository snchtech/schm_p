Приклад 1. Фільтрація по одній колонці PLANNING_DATE_START

SELECT t.ID, t.NAME, t.PLANNING_DATE_START, t.PLANNING_DATE_END FROM
YOUR_TABLE t WHERE ( :P10_CUST_DATE_START IS NULL OR
t.PLANNING_DATE_START >= TO_DATE(:P10_CUST_DATE_START, ‘DD.MM.YYYY’) )
AND ( :P10_CUST_DATE_END IS NULL OR t.PLANNING_DATE_START <
TO_DATE(:P10_CUST_DATE_END, ‘DD.MM.YYYY’) + 1 ) ORDER BY
t.PLANNING_DATE_START;
