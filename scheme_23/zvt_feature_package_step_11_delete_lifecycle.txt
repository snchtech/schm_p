// DELETE LIFECYCLE + AUTO REBUILD/SHORTEN BRANCH + LAST ZVT => DELETE BRANCH
// ============================================================================
//
// НОВІ ОБОВ'ЯЗКОВІ ПРАВИЛА ЖИТТЄВОГО ЦИКЛУ:
//
// 1. Видалення branch:
//    - видаляється branch;
//    - каскадно видаляються ВСІ ZVT з zvt.branchId === branch.id;
//    - очищаються runtime-дані цих ZVT.
//
// 2. Видалення одного ZVT:
//    - ZVT видаляється;
//    - branch НЕ видаляється, якщо на ньому лишилися інші ZVT;
//    - branch автоматично перебудовується;
//    - effective terminal скорочується до tap найдальшого ZVT.
//
// 3. Видалення ОСТАННЬОГО ZVT:
//    - ZVT видаляється;
//    - branch автоматично видаляється;
//    - orphan branch без ZVT після такого delete не залишається.
//
// 4. Source of truth:
//    - branch.attachedTo -> node/edge;
//    - zvt.branchId -> branch;
//    - branch НЕ зберігає memberIds[];
//    - склад members визначається через zvt.branchId.
//
// 5. Геометрія branch після delete не мутується вручну.
//    Вона derived через:
//      getBranchMembers()
//      getEffectiveTerminalEnd()
//      buildBranchPointsWithMembers()
//
// 6. Delete branch та delete ZVT мають бути окремими controller/store actions.
//
// ЦЕЙ STEP 11 Є SOURCE OF TRUTH ДЛЯ DELETE-ПОВЕДІНКИ.

removeZvtAndRebuildBranch: (zvtId) =>
  set((state) => {
    const target =
      (state.zvts || []).find(
        (zvt) => zvt.id === zvtId
      );

    if (!target) {
      return state;
    }

    const branchId =
      target.branchId;

    const remainingZvts =
      (state.zvts || []).filter(
        (zvt) => zvt.id !== zvtId
      );

    const remainingOnBranch =
      remainingZvts.filter(
        (zvt) =>
          zvt.branchId === branchId
      );

    const nextRuntime = {
      ...(state.zvtRuntimeById || {}),
    };

    delete nextRuntime[zvtId];

    /**
     * Якщо це був останній ZVT на branch:
     * branch теж видаляється.
     */
    if (
      branchId &&
      remainingOnBranch.length === 0
    ) {
      return {
        zvts:
          remainingZvts,

        zvtBranches:
          (state.zvtBranches || [])
            .filter(
              (branch) =>
                branch.id !== branchId
            ),

        zvtRuntimeById:
          nextRuntime,

        selection:
          (
            state.selection?.type === "zvt" &&
            state.selection?.id === zvtId
          ) ||
          (
            state.selection?.type === "zvtBranch" &&
            state.selection?.id === branchId
          )
            ? null
            : state.selection,

        zvtEditor:
          state.zvtEditor?.zvtId === zvtId
            ? null
            : state.zvtEditor,

        zvtBranchEdit:
          state.zvtBranchEdit?.branchId === branchId
            ? null
            : state.zvtBranchEdit,
      };
    }

    /**
     * Якщо members лишилися:
     * branch object сам по собі НЕ треба вручну "скорочувати".
     *
     * Його Canvas/SVG terminal буде автоматично
     * скорочено через derived geometry.
     */
    return {
      zvts:
        remainingZvts,

      zvtRuntimeById:
        nextRuntime,

      selection:
        state.selection?.type === "zvt" &&
        state.selection?.id === zvtId
          ? null
          : state.selection,

      zvtEditor:
        state.zvtEditor?.zvtId === zvtId
          ? null
          : state.zvtEditor,
    };
  }),


removeZvtBranchCascade: (branchId) =>
  set((state) => {
    const removedZvtIds =
      new Set(
        (state.zvts || [])
          .filter(
            (zvt) =>
              zvt.branchId === branchId
          )
          .map(
            (zvt) => zvt.id
          )
      );

    const nextRuntime = {
      ...(state.zvtRuntimeById || {}),
    };

    for (
      const zvtId of removedZvtIds
    ) {
      delete nextRuntime[zvtId];
    }

    return {
      zvtBranches:
        (state.zvtBranches || [])
          .filter(
            (branch) =>
              branch.id !== branchId
          ),

      zvts:
        (state.zvts || [])
          .filter(
            (zvt) =>
              !removedZvtIds.has(
                zvt.id
              )
          ),

      zvtRuntimeById:
        nextRuntime,

      selection:
        (
          state.selection?.type === "zvtBranch" &&
          state.selection?.id === branchId
        ) ||
        (
          state.selection?.type === "zvt" &&
          removedZvtIds.has(
            state.selection?.id
          )
        )
          ? null
          : state.selection,

      zvtEditor:
        removedZvtIds.has(
          state.zvtEditor?.zvtId
        )
          ? null
          : state.zvtEditor,

      zvtBranchEdit:
        state.zvtBranchEdit?.branchId === branchId
          ? null
          : state.zvtBranchEdit,

      zvtBranchPicker:
        (
          state.zvtBranchPicker?.selectedBranchId === branchId ||
          removedZvtIds.has(
            state.zvtBranchPicker?.zvtId
          )
        )
          ? null
          : state.zvtBranchPicker,
    };
  }),

deleteZvt(
  zvtId
) {
  const store =
    useCanvasStore.getState();

  if (
    (store.accessMode || "editor") === "reader"
  ) {
    return {
      ok: false,
      code: "READ_ONLY",
      message:
        "У режимі перегляду видалення ЗВТ заборонено.",
    };
  }

  const zvt =
    (store.zvts || [])
      .find(
        (item) => item.id === zvtId
      );

  if (!zvt) {
    return {
      ok: false,
      code: "ZVT_NOT_FOUND",
      message:
        "ЗВТ не знайдено.",
    };
  }

  const branchId =
    zvt.branchId;

  const countBefore =
    (store.zvts || [])
      .filter(
        (item) =>
          item.branchId === branchId
      )
      .length;

  store.removeZvtAndRebuildBranch?.(
    zvtId
  );

  return {
    ok: true,

    code:
      countBefore <= 1
        ? "ZVT_AND_BRANCH_DELETED"
        : "ZVT_DELETED_BRANCH_REBUILT",

    branchDeleted:
      countBefore <= 1,

    message:
      countBefore <= 1
        ? "Останній ЗВТ видалено разом із branch."
        : "ЗВТ видалено, branch автоматично скорочено.",
  };
},


deleteZvtBranch(
  branchId
) {
  const store =
    useCanvasStore.getState();

  if (
    (store.accessMode || "editor") === "reader"
  ) {
    return {
      ok: false,
      code: "READ_ONLY",
      message:
        "У режимі перегляду видалення branch заборонено.",
    };
  }

  const branch =
    (store.zvtBranches || [])
      .find(
        (item) =>
          item.id === branchId
      );

  if (!branch) {
    return {
      ok: false,
      code: "BRANCH_NOT_FOUND",
      message:
        "Branch не знайдено.",
    };
  }

  const memberCount =
    (store.zvts || [])
      .filter(
        (zvt) =>
          zvt.branchId === branchId
      )
      .length;

  store.removeZvtBranchCascade?.(
    branchId
  );

  return {
    ok: true,
    code:
      "BRANCH_CASCADE_DELETED",
    removedZvtCount:
      memberCount,

    message:
      memberCount
        ? `Branch видалено разом із ${memberCount} ЗВТ.`
        : "Branch видалено.",
  };
},

// STEP 11 — DELETE-SAFE DERIVED TERMINAL
// ============================================================================
//
// Цей блок уточнює, що branch shortening — derived,
// а не окремий mutation route.terminalPoint.
//

export function getBranchMembers(
  branchId,
  zvts
) {
  return (zvts || [])
    .filter(
      (zvt) =>
        zvt.branchId === branchId
    )
    .slice()
    .sort(
      (a, b) =>
        (
          a.connection?.offsetAlongBranch ?? 0
        ) -
        (
          b.connection?.offsetAlongBranch ?? 0
        )
    );
}


export function getEffectiveTerminalEnd(
  branch,
  basePoints,
  zvts
) {
  const terminal =
    getTerminalSegment(
      basePoints
    );

  if (!terminal) {
    return null;
  }

  const members =
    getBranchMembers(
      branch.id,
      zvts
    );

  /**
   * Для active branch із ZVT:
   * endpoint = tap найдальшого ZVT.
   *
   * Якщо members = 0, цей branch у нормальному
   * workflow існує тільки ДО додавання першого ZVT.
   *
   * Після delete останнього ZVT branch буде видалено
   * store action-ом, тому orphan branch не лишиться.
   */
  if (!members.length) {
    return terminal.baseEnd;
  }

  const maxOffset =
    Math.max(
      ...members.map(
        (zvt) =>
          zvt.connection?.offsetAlongBranch ?? 0
      )
    );

  return pointAtTerminalOffset(
    terminal,
    maxOffset
  );
}


export function buildBranchPointsWithMembers(
  branch,
  basePoints,
  zvts
) {
  if (
    !basePoints ||
    basePoints.length < 2
  ) {
    return basePoints || [];
  }

  const members =
    getBranchMembers(
      branch.id,
      zvts
    );

  if (!members.length) {
    return basePoints;
  }

  const end =
    getEffectiveTerminalEnd(
      branch,
      basePoints,
      zvts
    );

  return [
    ...basePoints.slice(0, -1),
    end,
  ];
}

      (s) =>

      <DialogTitle>

          onClick={() =>
            clear?.()
          }

          Скасувати

        <Button
          color="error"
          variant="contained"
          onClick={handleDelete}
        >

  );

export function createZvtBranch({
  id = crypto.randomUUID(),
  attachedTo,
  parameter = "F",
  route = {},
  style = {},
} = {}) {
  if (
    !attachedTo?.id ||
    !["node", "edge"].includes(attachedTo.type)
  ) {
    throw new Error(
      "ZVT branch must have attachedTo node/edge"
    );
  }

  return {
    id,
    kind: "zvtBranch",

    // THIS IS PERSISTED AND REQUIRED.
    attachedTo: {
      type: attachedTo.type,
      id: attachedTo.id,
      segmentId: attachedTo.segmentId || null,

      // node side, or preferred side for root.
      side: attachedTo.side || null,

      // Exact point from context menu when available.
      preferredPoint:
        attachedTo.preferredPoint || null,
    },

    data: {
      parameter,
    },

    route: {
      // horizontal | vertical | elbow | free
      type:
        route.type || "horizontal",

      direction:

function normalizeVector(x, y) {
  const length =
    Math.hypot(x, y) || 1;

  return {
    x: x / length,
    y: y / length,
  };
}

export function getEdgeAttachmentSegment(
  edge,
  segmentId
) {
  const meta =
    edge?.geometry
      ?.segmentsMeta
      ?.find(
        (segment) =>
          segment.id === segmentId
      );

  if (meta) {
    return {
      x1: meta.x1,
      y1: meta.y1,
      x2: meta.x2,
      y2: meta.y2,
    };
  }

  const points =
    edge?.geometry?.points || [];

  if (points.length >= 4) {
    return {
      x1: points[0],
      y1: points[1],
      x2: points[2],
      y2: points[3],
    };
  }

  return {
    x1:
      edge?.start?.x ?? 0,
    y1:
      edge?.start?.y ?? 0,
    x2:
      edge?.end?.x ?? 100,
    y2:
      edge?.end?.y ?? 0,
  };
}

export function getEdgeDirectionVector(
  edge,
  segmentId
) {
  const segment =
    getEdgeAttachmentSegment(
      edge,
      segmentId
    );

  return normalizeVector(
    segment.x2 - segment.x1,
    segment.y2 - segment.y1
  );
}

export function getEdgePerpendicularVectors(
  edge,
  segmentId
) {
  const tangent =
    getEdgeDirectionVector(
      edge,
      segmentId
    );

  return {
    tangent,

    positiveNormal: {
      x: -tangent.y,
      y: tangent.x,
    },

    negativeNormal: {
      x: tangent.y,
      y: -tangent.x,
    },
  };
}

export function resolveEdgeExitVector(
  branch,
  edge
) {
  const {
    tangent,
    positiveNormal,
    negativeNormal,
  } =
    getEdgePerpendicularVectors(
      edge,
      branch.attachedTo?.segmentId
    );

  if (
    branch.route?.edgeRootMode === "manual"
  ) {
    return null;
  }

  const exit =
    branch.route
      ?.edgeExitDirection ||
    "auto";

  /**
   * Horizontal edge:
   * tangent ~= (+/-1, 0)
   *
   * branch should be vertical.
   */
  if (
    Math.abs(tangent.x) >=
    Math.abs(tangent.y)
  ) {
    if (exit === "top") {
      return { x: 0, y: -1 };
    }

    if (exit === "bottom") {
      return { x: 0, y: 1 };
    }

    // auto default = bottom.
    return { x: 0, y: 1 };
  }

  /**
   * Vertical edge:
   * branch should be horizontal.
   */
  if (
    Math.abs(tangent.y) >
    Math.abs(tangent.x)
  ) {
    if (exit === "left") {
      return { x: -1, y: 0 };
    }

    if (exit === "right") {
      return { x: 1, y: 0 };
    }

    // auto default = right.
    return { x: 1, y: 0 };
  }

  /**
   * Diagonal fallback.
   */
  return exit === "negativeNormal"
    ? negativeNormal
    : positiveNormal;
}

// STEP 10 — STABLE ROOT, NO GAP FROM NODE/EDGE

import {
  getEdgeAttachmentSegment,
  resolveEdgeExitVector,
} from "@/lib/zvt/zvtEdgeGeometry";

function pointOnSegmentByProjection(
  pointer,
  segment
) {
  const dx =
    segment.x2 - segment.x1;

  const dy =
    segment.y2 - segment.y1;

  const lenSq =
    dx * dx + dy * dy;

  if (!lenSq) {
    return {
      x: segment.x1,
      y: segment.y1,
    };
  }

  let t =
    (
      (pointer.x - segment.x1) * dx +
      (pointer.y - segment.y1) * dy
    ) / lenSq;

  t =
    Math.max(
      0,
      Math.min(1, t)
    );

  return {
    x:
      segment.x1 + dx * t,

    y:
      segment.y1 + dy * t,
  };
}

export function resolveStableEdgeBranchRoot(
  branch,
  edge
) {
  const segment =
    getEdgeAttachmentSegment(
      edge,
      branch.attachedTo
        ?.segmentId
    );

  const preferred =
    branch.attachedTo
      ?.preferredPoint;

  const root =
    preferred &&
    Number.isFinite(preferred.x) &&
    Number.isFinite(preferred.y)
      ? pointOnSegmentByProjection(
          preferred,
          segment
        )
      : {
          x:
            (
              segment.x1 +
              segment.x2
            ) / 2,

          y:
            (
              segment.y1 +
              segment.y2
            ) / 2,
        };

  return {
    ...root,

    exitVector:
      resolveEdgeExitVector(
        branch,
        edge
      ),
  };
}

/**
 * NODE:
 *
 * IMPORTANT:
 * - use branch.attachedTo.side or exact preferredPoint;
 * - NEVER use ZVT position to choose another node side.
 * - root is directly on node boundary.
 *
 * For complex node prefer node.connectionAnchors.
 */
export function resolveStableNodeBranchRoot(
  branch,
  node,
  getNodeBounds
) {
  const preferred =
    branch.attachedTo
      ?.preferredPoint;

  if (
    preferred &&
    Number.isFinite(preferred.x) &&
    Number.isFinite(preferred.y)
  ) {
    return {
      x: preferred.x,
      y: preferred.y,
      side:
        branch.attachedTo.side ||
        null,
    };
  }

  const side =
    branch.attachedTo
      ?.side || "right";

  const anchors =
    node?.connectionAnchors ||
    node?.anchors;

  if (
    anchors?.[side]
  ) {
    return {
      x:
        anchors[side].x,

      y:
        anchors[side].y,

      side,
    };
  }

  const b =
    getNodeBounds(node);

  switch (side) {
    case "left":
      return {
        x: b.left,
        y: b.centerY,
        side,
      };

    case "top":
      return {
        x: b.centerX,
        y: b.top,
        side,
      };

    case "bottom":
      return {
        x: b.centerX,
        y: b.bottom,
        side,
      };

    case "right":
    default:
      return {
        x: b.right,
        y: b.centerY,
        side: "right",
      };
  }
}

function normalizeVector(x, y) {
  const len =
    Math.hypot(x, y) || 1;

  return {
    x: x / len,
    y: y / len,
  };
}

export function sideVector(side) {
  switch (side) {
    case "top":
      return { x: 0, y: -1 };

    case "bottom":
      return { x: 0, y: 1 };

    case "left":
      return { x: -1, y: 0 };

    case "right":
      return { x: 1, y: 0 };

    default:
      return { x: 0, y: 1 };
  }
}

export function getTerminalSegment(
  basePoints
) {
  if (
    !basePoints ||
    basePoints.length < 2
  ) {
    return null;
  }

  const a =
    basePoints[
      basePoints.length - 2
    ];

  const b =
    basePoints[
      basePoints.length - 1
    ];

  const vector =
    normalizeVector(
      b.x - a.x,
      b.y - a.y
    );

  return {
    start: a,
    baseEnd: b,
    vector,
  };
}

export function getBranchMembers(
  branchId,
  zvts
) {
  return (zvts || [])
    .filter(
      (zvt) =>
        zvt.branchId ===
        branchId
    )
    .slice()
    .sort(
      (a, b) =>
        (
          a.connection
            ?.offsetAlongBranch ??
          0
        ) -
        (
          b.connection
            ?.offsetAlongBranch ??
          0
        )
    );
}

/**
 * Scalar projection onto terminal direction.
 */
export function projectPointToTerminalOffset(
  point,
  terminal
) {
  const dx =
    point.x -
    terminal.start.x;

  const dy =
    point.y -
    terminal.start.y;

  return (
    dx * terminal.vector.x +
    dy * terminal.vector.y
  );
}

export function pointAtTerminalOffset(
  terminal,
  offset
) {
  return {
    x:
      terminal.start.x +
      terminal.vector.x *
        offset,

    y:
      terminal.start.y +
      terminal.vector.y *
        offset,
  };
}

/**
 * Build member visual geometry.
 *
 * tap = point on branch line.
 *
 * circleCenter is placed opposite the selected
 * circle connection side.
 *
 * Example:
 * side="bottom":
 *
 *       O  ZVT
 *       |
 *       | stub
 * ------+------ branch
 *
 * so circle is ABOVE branch.
 */
export function buildZvtMemberGeometry(
  zvt,
  terminal
) {
  const offset =
    zvt.connection
      ?.offsetAlongBranch ?? 0;

  const tap =
    pointAtTerminalOffset(
      terminal,
      offset
    );

  const side =
    zvt.connection?.side ||
    "bottom";

  const sv =
    sideVector(side);

  const radius =
    zvt.style?.radius ?? 22;

  const stubLength =
    zvt.connection
      ?.stubLength ?? 22;

  const contactPoint = {
    x:
      tap.x -
      sv.x *
        stubLength,

    y:
      tap.y -
      sv.y *
        stubLength,
  };

  const center = {
    x:
      contactPoint.x -
      sv.x *
        radius,

    y:
      contactPoint.y -
      sv.y *
        radius,
  };

  return {
    tap,
    stub: [
      tap.x,
      tap.y,
      contactPoint.x,
      contactPoint.y,
    ],

    contactPoint,
    center,
  };
}

/**
 * Effective branch endpoint:
 * if members exist -> tap of the furthest member.
 *
 * This satisfies:
 * "кінець branch має бути приєднаний до ЗВТ".
 */
export function getEffectiveTerminalEnd(
  branch,
  basePoints,
  zvts
) {
  const terminal =
    getTerminalSegment(
      basePoints
    );

  if (!terminal) {
    return null;
  }

  const members =
    getBranchMembers(
      branch.id,
      zvts
    );

  if (!members.length) {
    return terminal.baseEnd;
  }

  const maxOffset =
    Math.max(
      ...members.map(
        (zvt) =>
          zvt.connection
            ?.offsetAlongBranch ??
          0
      )
    );

  return pointAtTerminalOffset(
    terminal,
    maxOffset +
      (
        branch.route
          ?.terminalPadding ??
        0
      )
  );
}

/**
 * Branch points with member-driven terminal extension.
 */
export function buildBranchPointsWithMembers(
  branch,
  basePoints,
  zvts
) {
  if (
    !basePoints ||
    basePoints.length < 2
  ) {
    return basePoints || [];
  }

  const effectiveEnd =
    getEffectiveTerminalEnd(
      branch,
      basePoints,
      zvts
    );

  return [
    ...basePoints.slice(
      0,
      -1
    ),

    effectiveEnd,
  ];
}

/**
 * New member always appends after the current last member.
 */
export function getNextMemberOffset(
  branch,
  basePoints,
  zvts
) {
  const terminal =
    getTerminalSegment(
      basePoints
    );

  if (!terminal) {
    return 0;
  }

  const baseLength =
    Math.hypot(
      terminal.baseEnd.x -
        terminal.start.x,

      terminal.baseEnd.y -
        terminal.start.y
    );

  const members =
    getBranchMembers(
      branch.id,
      zvts
    );

  const spacing =
    branch.route
      ?.memberSpacing ?? 90;

  if (!members.length) {
    // First ZVT exactly at the original branch terminal.
    return baseLength;
  }

  const lastOffset =
    members[
      members.length - 1
    ].connection
      ?.offsetAlongBranch ??
    baseLength;

  return (
    Math.max(
      lastOffset,
      baseLength
    ) +
    spacing
  );
}

/**
 * Drag constraint:
 * pointer -> scalar offset on the terminal axis.
 */
export function constrainZvtDragToBranch(
  pointer,
  branch,
  basePoints,
  zvts,
  zvtId,
  {
    minOffset = 20,
  } = {}
) {
  const terminal =
    getTerminalSegment(
      basePoints
    );

  if (!terminal) {
    return null;
  }

  let offset =
    projectPointToTerminalOffset(
      pointer,
      terminal
    );

  offset =
    Math.max(
      minOffset,
      offset
    );

  return {
    offsetAlongBranch:
      offset,

    tap:
      pointAtTerminalOffset(
        terminal,
        offset
      ),
  };
}

openZvtEditorFromBranch(
  branchId

  const store =
    useCanvasStore.getState();

  const branch =
    (store.zvtBranches || [])
      .find(
        (item) =>
          item.id === branchId
      );

  if (!branch) {

  }

  // New ZVT editor knows only branchId.
  store.setZvtEditor?.({
    mode: "create",
    branchId,
  });

createZvtFromBranchEditor({
  branchId,
  data,
  connection,
  style,
  integration,
}) {
  const store =
    useCanvasStore.getState();

  if (
    (store.accessMode || "editor") ===
    "reader"
  ) {
    return;
  }

  const branch =
    (store.zvtBranches || [])
      .find(
        (item) =>
          item.id === branchId
      );

  if (!branch) {
    return;
  }

  const root =
    resolveZvtBranchRoot(
      branch,
      store.nodes || [],
      store.edges || []
    );

  const basePoints =
    buildBaseBranchRoute(
      branch,
      root
    );

  const offset =
    getNextMemberOffset(
      branch,
      basePoints,
      store.zvts || []
    );

  const zvt =
    createZvt({
      branchId,

      data,

      connection: {
        ...connection,
        offsetAlongBranch:
          offset,
      },

      style,
      integration,
    });

  store.addZvt?.(zvt);

  store.setSelection?.({
    type: "zvt",
    id: zvt.id,
  });

  store.clearZvtEditor?.();

  return zvt;
},


moveZvtAlongBranch(
  zvtId,
  pointer
) {
  const store =
    useCanvasStore.getState();

  if (
    (store.accessMode || "editor") ===
    "reader"
  ) {
    return;
  }

  const zvt =
    (store.zvts || [])
      .find(
        (item) =>
          item.id === zvtId
      );

  if (!zvt?.branchId) {
    return;
  }

  const branch =
    (store.zvtBranches || [])
      .find(
        (item) =>
          item.id ===
          zvt.branchId
      );

  if (!branch) {
    return;
  }

  const root =
    resolveZvtBranchRoot(
      branch,
      store.nodes || [],
      store.edges || []
    );

  const basePoints =
    buildBaseBranchRoute(
      branch,
      root
    );

  const constrained =
    constrainZvtDragToBranch(
      pointer,
      branch,
      basePoints,
      store.zvts || [],
      zvt.id
    );

  if (!constrained) {
    return;
  }

  store.replaceZvt?.(
    zvt.id,
    {
      ...zvt,

      connection: {
        ...zvt.connection,

        offsetAlongBranch:
          constrained
            .offsetAlongBranch,
      },

      updatedAt:
        Date.now(),
    }
  );
},

// />
//
// This gives:

// - stub remains attached;

// - node/edge root never moves because of ZVT drag.

// ============================================================================


// ============================================================================
// 2. MEMBER ORDER BEHAVIOR
// ============================================================================
//
// No duplicated memberIds[] is required on branch.
//
// The relationship remains:
//
// branch.attachedTo = object
// zvt.branchId      = branch
//
// Order is derived from:
//
// zvt.connection.offsetAlongBranch
//
// Example:
//
// ZVT A offset = 140
// ZVT B offset = 230
// ZVT C offset = 320
//
// root ----------------+---------+---------+
//                      |         |         |
//                      A         B         C
//
// If user drags B after C:
//
// B offset becomes 380.
//
// Derived order becomes:
//
// A -> C -> B
//
// and branch endpoint automatically becomes B tap.
//
// This avoids duplicated relation state:
// branch.memberIds[] + zvt.branchId.
// ============================================================================


// ============================================================================
// 3. EDGE PERPENDICULAR EXAMPLES
// ============================================================================
//
// HORIZONTAL EDGE:
//
// ======================= edge
//            |
//            | branch
//            |
//            +---- ZVT / continuation
//
// edge tangent = horizontal.
// branch first vector = vertical.

// - top
// - bottom

//
// VERTICAL EDGE:

//          ||
//          || edge
//          ||----------- branch
//          ||
//
// edge tangent = vertical.
// branch first vector = horizontal.
//
// User chooses:
// - left
// - right
//
//
// DIAGONAL EDGE:
//
//      / edge
//     /
//    /
//    └──── branch normal vector
//
// Use geometric normal.
//
// ============================================================================


// ============================================================================
// 4. F ROOT MARK ORIENTATION
// ============================================================================
//
// F mark is oriented RELATIVE TO BRANCH:
//
// If branch is vertical:
//
//      |
//   ---+---    transverse base
//      ||      two short ticks parallel to branch
//      |
//      | branch
//
//
// If branch is horizontal:
//
//       |
//      -|-     transverse base is vertical
// -----+------ branch
//       |
//
// So existing vector rule is retained:
// base = perpendicular to branch;
// two short ticks = parallel to branch.
//
// This matches the clarification:
// marks remain in their expected orientation,
// while branch itself exits perpendicular to edge.

function getNodeBounds(node) {
  const explicit = node?.geometry?.bounds || node?.bounds || null;

  if (explicit) {
    const x = explicit.x ?? explicit.left ?? 0;
    const y = explicit.y ?? explicit.top ?? 0;
    const width =
      explicit.width ?? ((explicit.right ?? x) - x);
    const height =
      explicit.height ?? ((explicit.bottom ?? y) - y);

    return {
      left: x,
      top: y,
      right: x + width,
      bottom: y + height,
      centerX: x + width / 2,
      centerY: y + height / 2,
    };
  }

  const width = node?.size?.width ?? node?.width ?? 120;
  const height = node?.size?.height ?? node?.height ?? 60;

  let x = node?.position?.x ?? node?.x ?? 0;
  let y = node?.position?.y ?? node?.y ?? 0;

  if (node?.positionOrigin === "center") {
    x -= width / 2;
    y -= height / 2;
  }

  return {
    left: x,
    top: y,
    right: x + width,
    bottom: y + height,
    centerX: x + width / 2,
    centerY: y + height / 2,
  };
}

function resolveNodeRootPoint(branch, node) {
  const explicit = branch.attachedTo?.preferredPoint;

  if (
    Number.isFinite(explicit?.x) &&
    Number.isFinite(explicit?.y)
  ) {
    return {
      x: explicit.x,
      y: explicit.y,
      side: branch.attachedTo.side || null,
    };
  }

  const anchors = node?.connectionAnchors || node?.anchors;
  const side = branch.attachedTo?.side || null;

  if (
    side &&
    anchors?.[side] &&
    Number.isFinite(anchors[side].x) &&
    Number.isFinite(anchors[side].y)
  ) {
    return {
      x: anchors[side].x,
      y: anchors[side].y,
      side,
    };
  }

  const bounds = getNodeBounds(node);

  switch (side) {
    case "left":
      return { x: bounds.left, y: bounds.centerY, side };
    case "right":
      return { x: bounds.right, y: bounds.centerY, side };
    case "top":
      return { x: bounds.centerX, y: bounds.top, side };
    case "bottom":
      return { x: bounds.centerX, y: bounds.bottom, side };
    default:
      return {
        x: bounds.centerX,
        y: bounds.centerY,
        side: null,
      };
  }
}

function resolveEdgeRootPoint(branch, edge) {
  const explicit = branch.attachedTo?.preferredPoint;

  if (
    Number.isFinite(explicit?.x) &&
    Number.isFinite(explicit?.y)
  ) {
    return { x: explicit.x, y: explicit.y, side: null };
  }

  const segment = edge?.geometry?.segmentsMeta?.find(
    (item) => item.id === branch.attachedTo?.segmentId
  );

  if (segment) {
    return {
      x: (segment.x1 + segment.x2) / 2,
      y: (segment.y1 + segment.y2) / 2,
      side: null,
    };
  }

  return {
    x: edge?.geometry?.start?.x ?? 0,
    y: edge?.geometry?.start?.y ?? 0,
    side: null,
  };
}

export function resolveZvtBranchRoot(
  branch,
  nodes,
  edges
) {
  if (branch.attachedTo?.type === "node") {
    const node = nodes.find(
      (item) => item.id === branch.attachedTo.id
    );

    return node
      ? resolveNodeRootPoint(branch, node)
      : { x: 0, y: 0, side: null };
  }

  if (branch.attachedTo?.type === "edge") {
    const edge = edges.find(
      (item) => item.id === branch.attachedTo.id
    );

    return edge
      ? resolveEdgeRootPoint(branch, edge)
      : { x: 0, y: 0, side: null };
  }

  return { x: 0, y: 0, side: null };
}

export function buildZvtBranchRootMark(
  branch,
  root,
  firstDirectionVector
) {
  const parameter = branch.data?.parameter || "F";
  const size = branch.style?.rootMarkSize ?? 10;

  const dx = firstDirectionVector.x;
  const dy = firstDirectionVector.y;
  const px = -dy;
  const py = dx;

  const c = {
    x: root.x + dx * 5,
    y: root.y + dy * 5,
  };

  const line = (halfLength) => [
    c.x - px * halfLength,
    c.y - py * halfLength,
    c.x + px * halfLength,
    c.y + py * halfLength,
  ];

  if (parameter === "P") {
    return {
      type: "pressure-dot",
      dot: { x: c.x, y: c.y, radius: 2.8 },
    };
  }

  if (parameter === "F") {
    const base = line(size);
    const tickOffset = 4;
    const tickLength = 5;

    const c1 = {
      x: c.x - px * tickOffset,
      y: c.y - py * tickOffset,
    };

    const c2 = {
      x: c.x + px * tickOffset,
      y: c.y + py * tickOffset,
    };

    return {
      type: "flow",
      lines: [
        base,
        [
          c1.x - dx * tickLength,
          c1.y - dy * tickLength,
          c1.x + dx * tickLength,
          c1.y + dy * tickLength,
        ],
        [
          c2.x - dx * tickLength,
          c2.y - dy * tickLength,
          c2.x + dx * tickLength,
          c2.y + dy * tickLength,
        ],
      ],
    };
  }

  if (parameter === "T") {
    return {
      type: "temperature",
      lines: [
        line(size),
        [
          c.x - px * size + dx * 4,
          c.y - py * size + dy * 4,
          c.x + px * size + dx * 4,
          c.y + py * size + dy * 4,
        ],
      ],
    };
  }

  if (parameter === "ae" || parameter === "pH") {
    return {
      type: "conductivity",
      lines: [line(size + 3)],
    };
  }

  if (parameter === "L") {
    const stemEnd = {
      x: c.x + dx * 12,
      y: c.y + dy * 12,
    };

    return {
      type: "level",
      lines: [[root.x, root.y, stemEnd.x, stemEnd.y]],
      flag: [
        stemEnd.x,
        stemEnd.y,
        stemEnd.x - px * 5 - dx * 5,
        stemEnd.y - py * 5 - dy * 5,
        stemEnd.x + px * 5 - dx * 5,
        stemEnd.y + py * 5 - dy * 5,
      ],
    };
  }

  return { type: "none" };
}

function directionVector(direction) {
  switch (direction) {
    case "left":
      return { x: -1, y: 0 };
    case "right":
      return { x: 1, y: 0 };
    case "top":
      return { x: 0, y: -1 };
    case "bottom":
      return { x: 0, y: 1 };
    default:
      return { x: 1, y: 0 };
  }
}

export function buildBaseBranchRoute(branch, root) {
  const route = branch.route || {};
  const type = route.type || "horizontal";

  if (route.terminalPoint) {
    return [
      root,
      ...(route.manualVertices || []),
      route.terminalPoint,
    ];
  }

  if (type === "horizontal" || type === "vertical") {
    const vector = directionVector(
      route.direction ||
        (type === "horizontal" ? "right" : "bottom")
    );

    return [
      root,
      {
        x: root.x + vector.x * (route.primaryLength ?? 140),
        y: root.y + vector.y * (route.primaryLength ?? 140),
      },
    ];
  }

  if (type === "elbow") {
    const v1 = directionVector(
      route.primaryDirection || "right"
    );

    const v2 = directionVector(
      route.secondaryDirection || "bottom"
    );

    const bend = {
      x: root.x + v1.x * (route.primaryLength ?? 140),
      y: root.y + v1.y * (route.primaryLength ?? 140),
    };

    return [
      root,
      bend,
      {
        x: bend.x + v2.x * (route.secondaryLength ?? 90),
        y: bend.y + v2.y * (route.secondaryLength ?? 90),
      },
    ];
  }

  const angleRad =
    (route.freeAngle ?? 0) * Math.PI / 180;

  return [
    root,
    {
      x:
        root.x +
        Math.cos(angleRad) * (route.freeLength ?? 180),
      y:
        root.y +
        Math.sin(angleRad) * (route.freeLength ?? 180),
    },
  ];
}

export function buildFirstDirectionVector(
  branch,
  points
) {
  if (!points || points.length < 2) {
    return { x: 1, y: 0 };
  }

  const a = points[0];
  const b = points[1];

  const dx = b.x - a.x;
  const dy = b.y - a.y;
  const length = Math.hypot(dx, dy) || 1;

  return {
    x: dx / length,
    y: dy / length,
  };
}

  id = crypto.randomUUID(),
  attachedTo,
  anchor = {},
  position = {},
  data = {},
  route = {},
  style = {},
  integration = {},
} = {}) {
  return {
    id,
    kind: "zvt",

    attachedTo: attachedTo || {
      type: "none",
      id: null,
      segmentId: null,
    },

    anchor: {
      x: anchor.x ?? 0,
      y: anchor.y ?? 0,
      side: anchor.side ?? null,
    },

    position: {
      x: position.x ?? 0,
      y: position.y ?? 0,
    },

    route: {
      type: route.type || "orthogonal",
      vertices: Array.isArray(route.vertices) ? route.vertices : [],

      sourceSide:
        route.sourceSide ||
        route.preferredSide ||

const VALID_SIDES = new Set(["top", "bottom", "left", "right"]);

function clamp(value, min, max) {
  return Math.max(min, Math.min(max, value));
}

function isValidSide(side) {
  return VALID_SIDES.has(side);
}

function sideVector(side) {
  switch (side) {
    case "top":
      return { x: 0, y: -1 };
    case "bottom":
      return { x: 0, y: 1 };
    case "left":
      return { x: -1, y: 0 };
    case "right":
      return { x: 1, y: 0 };
    default:
      return { x: 0, y: 0 };
  }
}

/**
 * Канонічні bounds node.
 *
 * Пріоритет:
 * 1. node.geometry.bounds
 * 2. node.bounds
 * 3. node.position + node.size
 * 4. node.x/y + node.width/height
 *
 * Якщо у твоєму NodesLayer position є ЦЕНТРОМ, а не top-left,
 * вистав node.positionOrigin = "center".
 *
 * Це прибирає типову причину зміщеного anchor:
 * різне трактування node.position у NodesLayer і ZVT geometry.
 */
export function getZvtNodeBounds(node) {
  const explicit =
    node?.geometry?.bounds ||
    node?.bounds ||
    null;

  if (explicit) {
    const x = explicit.x ?? explicit.left ?? 0;
    const y = explicit.y ?? explicit.top ?? 0;
    const width =
      explicit.width ??
      ((explicit.right ?? x) - x) ??
      120;
    const height =
      explicit.height ??
      ((explicit.bottom ?? y) - y) ??
      60;

    return {
      x,
      y,
      width,
      height,
      left: x,
      top: y,
      right: x + width,
      bottom: y + height,
      centerX: x + width / 2,
      centerY: y + height / 2,
    };
  }

  const width = node?.size?.width ?? node?.width ?? 120;
  const height = node?.size?.height ?? node?.height ?? 60;

  let x = node?.position?.x ?? node?.x ?? 0;
  let y = node?.position?.y ?? node?.y ?? 0;

  if (node?.positionOrigin === "center") {
    x -= width / 2;
    y -= height / 2;
  }

  return {
    x,
    y,
    width,
    height,
    left: x,
    top: y,
    right: x + width,
    bottom: y + height,
    centerX: x + width / 2,
    centerY: y + height / 2,
  };
}

/**
 * Якщо node вже має свої точні anchors — використовуємо їх.
 *
 * Підтримувані варіанти:
 *
 * node.connectionAnchors = {
 *   top: { x, y },
 *   bottom: { x, y },
 *   left: { x, y },
 *   right: { x, y },
 * }
 *
 * або node.anchors з такою самою структурою.
 *
 * Це найкращий варіант для НЕпрямокутних node:
 * circle, valve, custom SVG, складна Konva-фігура тощо.
 */
function getExplicitNodeAnchor(node, side) {
  const anchors = node?.connectionAnchors || node?.anchors;
  const point = anchors?.[side];

  if (
    point &&
    Number.isFinite(point.x) &&
    Number.isFinite(point.y)
  ) {
    return {
      x: point.x,
      y: point.y,
      side,
    };
  }

  return null;
}

function getNearestNodeSide(bounds, target) {
  const candidates = [
    {
      side: "left",
      d: Math.abs(target.x - bounds.left),
    },
    {
      side: "right",
      d: Math.abs(target.x - bounds.right),
    },
    {
      side: "top",
      d: Math.abs(target.y - bounds.top),
    },
    {
      side: "bottom",
      d: Math.abs(target.y - bounds.bottom),
    },
  ];

  candidates.sort((a, b) => a.d - b.d);
  return candidates[0].side;
}

/**
 * Точка на ФАКТИЧНОМУ краї прямокутних bounds.
 *
 * Для left/right проєктуємо y символу на вертикальну сторону.
 * Для top/bottom проєктуємо x символу на горизонтальну сторону.
 *
 * Тому лінія не "стрибає" в центр сторони і не виглядає зміщеною.
 */
function projectPointToNodeSide(bounds, side, target) {
  switch (side) {
    case "left":
      return {
        x: bounds.left,
        y: clamp(target.y, bounds.top, bounds.bottom),
        side,
      };

    case "right":
      return {
        x: bounds.right,
        y: clamp(target.y, bounds.top, bounds.bottom),
        side,
      };

    case "top":
      return {
        x: clamp(target.x, bounds.left, bounds.right),
        y: bounds.top,
        side,
      };

    case "bottom":
      return {
        x: clamp(target.x, bounds.left, bounds.right),
        y: bounds.bottom,
        side,
      };

    default:
      return {
        x: bounds.centerX,
        y: bounds.centerY,
        side: null,
      };
  }
}

function getEdgeCenter(edge) {
  const points = edge?.geometry?.points || [];

  if (points.length >= 4) {
    const middleIndex = Math.max(
      0,
      Math.floor(points.length / 4) * 2 - 2
    );

    const x1 = points[middleIndex] ?? points[0];
    const y1 = points[middleIndex + 1] ?? points[1];
    const x2 = points[middleIndex + 2] ?? points[points.length - 2];
    const y2 = points[middleIndex + 3] ?? points[points.length - 1];

    return {
      x: (x1 + x2) / 2,
      y: (y1 + y2) / 2,
    };
  }

  return {
    x: edge?.geometry?.start?.x ?? edge?.start?.x ?? 0,
    y: edge?.geometry?.start?.y ?? edge?.start?.y ?? 0,
  };
}

function getEdgeSegmentCenter(edge, segmentId) {
  const segment = edge?.geometry?.segmentsMeta?.find(
    (seg) => seg.id === segmentId
  );

  if (!segment) return getEdgeCenter(edge);

  return {
    x: (segment.x1 + segment.x2) / 2,
    y: (segment.y1 + segment.y2) / 2,
  };
}

/**
 * Визначає anchor ЗВТ.
 *
 * NODE:
 * - route.sourceSide = auto => найближча сторона;
 * - route.sourceSide = top/bottom/left/right => чітко задана сторона.
 *
 * EDGE:
 * - поточна логіка segmentId/center збережена.
 */
export function resolveZvtAnchor(zvt, nodes = [], edges = []) {
  if (!zvt?.attachedTo) {
    return {
      x: zvt?.anchor?.x ?? 0,
      y: zvt?.anchor?.y ?? 0,
      side: zvt?.anchor?.side ?? null,
    };
  }

  if (zvt.attachedTo.type === "node") {
    const node = nodes.find(
      (item) => item.id === zvt.attachedTo.id
    );

    if (!node) {
      return zvt.anchor || {
        x: 0,
        y: 0,
        side: null,
      };
    }

    const bounds = getZvtNodeBounds(node);
    const target = zvt.position || {
      x: bounds.centerX + 100,
      y: bounds.centerY,
    };

    const requestedSide =
      zvt.route?.sourceSide ||
      zvt.route?.preferredSide ||
      "auto";

    const resolvedSide = isValidSide(requestedSide)
      ? requestedSide
      : getNearestNodeSide(bounds, target);

    // Для складних фігур NodesLayer може дати точний anchor.
    const explicitAnchor = getExplicitNodeAnchor(
      node,
      resolvedSide
    );

    if (explicitAnchor) {
      return explicitAnchor;
    }

    return projectPointToNodeSide(
      bounds,
      resolvedSide,
      target
    );
  }

  if (zvt.attachedTo.type === "edge") {
    const edge = edges.find(
      (item) => item.id === zvt.attachedTo.id
    );

    if (!edge) {
      return zvt.anchor || {
        x: 0,
        y: 0,
        side: null,
      };
    }

    const center = zvt.attachedTo.segmentId
      ? getEdgeSegmentCenter(
          edge,
          zvt.attachedTo.segmentId
        )
      : getEdgeCenter(edge);

    return {
      x: center.x,
      y: center.y,
      side: null,
    };
  }

  return zvt.anchor || {
    x: 0,
    y: 0,
    side: null,
  };
}

function resolveAutoSymbolSide(anchor, symbol) {
  const dx = anchor.x - symbol.x;
  const dy = anchor.y - symbol.y;

  if (Math.abs(dx) >= Math.abs(dy)) {
    return dx < 0 ? "left" : "right";
  }

  return dy < 0 ? "top" : "bottom";
}

function getSymbolAttachPoint(
  symbol,
  radius,
  requestedSide,
  anchor
) {
  const side = isValidSide(requestedSide)
    ? requestedSide
    : resolveAutoSymbolSide(anchor, symbol);

  switch (side) {
    case "top":
      return {
        x: symbol.x,
        y: symbol.y - radius,
        side,
      };

    case "bottom":
      return {
        x: symbol.x,
        y: symbol.y + radius,
        side,
      };

    case "left":
      return {
        x: symbol.x - radius,
        y: symbol.y,
        side,
      };

    case "right":
      return {
        x: symbol.x + radius,
        y: symbol.y,
        side,
      };

    default:
      return {
        x: symbol.x,
        y: symbol.y,
        side: "left",
      };
  }
}

function dedupeOrthogonalPoints(points) {
  if (!Array.isArray(points) || points.length < 4) {
    return points || [];
  }

  const result = [];

  for (let i = 0; i < points.length; i += 2) {
    const x = points[i];
    const y = points[i + 1];

    const prevX = result[result.length - 2];
    const prevY = result[result.length - 1];

    if (prevX === x && prevY === y) continue;

    result.push(x, y);
  }

  return result;
}

/**
 * Ортогональна траса:
 *
 * node anchor
 *   -> короткий source stub
 *   -> ламана
 *   -> короткий symbol stub
 *   -> край кружечка ЗВТ
 *
 * Таким чином перший відрізок гарантовано виходить
 * перпендикулярно до вибраної сторони node.
 */
export function buildZvtLeaderRoute({
  anchor,
  symbol,
  radius = 22,
  sourceSide = null,
  symbolSide = "auto",
  sourceOffset = 18,
  symbolOffset = 12,
}) {
  if (!anchor || !symbol) return [];

  const resolvedSourceSide = isValidSide(sourceSide)
    ? sourceSide
    : isValidSide(anchor.side)
      ? anchor.side
      : null;

  const target = getSymbolAttachPoint(
    symbol,
    radius,
    symbolSide,
    anchor
  );

  // Для edge sourceSide може бути null.
  // Тоді визначаємо напрямок від edge до ЗВТ автоматично.
  const effectiveSourceSide =
    resolvedSourceSide ||
    (() => {
      const dx = symbol.x - anchor.x;
      const dy = symbol.y - anchor.y;

      if (Math.abs(dx) >= Math.abs(dy)) {
        return dx >= 0 ? "right" : "left";
      }

      return dy >= 0 ? "bottom" : "top";
    })();

  const sourceVector = sideVector(effectiveSourceSide);
  const targetVector = sideVector(target.side);

  const sourceStub = {
    x: anchor.x + sourceVector.x * sourceOffset,
    y: anchor.y + sourceVector.y * sourceOffset,
  };

  const targetStub = {
    x: target.x + targetVector.x * symbolOffset,
    y: target.y + targetVector.y * symbolOffset,
  };

  const points = [
    anchor.x,
    anchor.y,
    sourceStub.x,
    sourceStub.y,
  ];

  const sourceHorizontal =
    effectiveSourceSide === "left" ||
    effectiveSourceSide === "right";

  const targetHorizontal =
    target.side === "left" ||
    target.side === "right";

  if (sourceHorizontal && targetHorizontal) {
    const midX =
      (sourceStub.x + targetStub.x) / 2;

    points.push(
      midX,
      sourceStub.y,
      midX,
      targetStub.y
    );
  } else if (!sourceHorizontal && !targetHorizontal) {
    const midY =
      (sourceStub.y + targetStub.y) / 2;

    points.push(
      sourceStub.x,
      midY,
      targetStub.x,
      midY
    );
  } else if (sourceHorizontal && !targetHorizontal) {
    points.push(
      targetStub.x,
      sourceStub.y
    );
  } else {
    points.push(
      sourceStub.x,
      targetStub.y
    );
  }

  points.push(
    targetStub.x,
    targetStub.y,
    target.x,
    target.y
  );

  return dedupeOrthogonalPoints(points);
}

export function buildZvtSignalGeometry(zvt) {
  const radius = zvt?.style?.radius ?? 22;
  const x = zvt?.position?.x ?? 0;
  const y = zvt?.position?.y ?? 0;
  const signal = zvt?.data?.signal;

  if (!signal?.enabled) return null;

  const direction = signal.direction || "right";
  const sign = direction === "left" ? -1 : 1;
  const signalLength = Math.max(
    30,
    zvt?.style?.signalLength ?? 82
  );
  const arrowSize = Math.max(
    4,
    zvt?.style?.arrowSize ?? 8
  );

  const startX = x + sign * radius;
  const endX =
    x + sign * (radius + signalLength);
  const lineY = y;

  const arrowPoints =
    direction === "right"
      ? [
          endX,
          lineY,
          endX - arrowSize,
          lineY - arrowSize / 2,
          endX - arrowSize,
          lineY + arrowSize / 2,
        ]
      : [
          endX,
          lineY,
          endX + arrowSize,
          lineY - arrowSize / 2,
          endX + arrowSize,
          lineY + arrowSize / 2,
        ];

  const labelX =
    direction === "right"
      ? startX + 8
      : endX;

  return {
    line: [
      startX,
      lineY,
      endX,
      lineY,
    ],
    arrowPoints,
    labelX,
    labelY: y - 30,
    labelWidth: Math.abs(endX - startX),
  };
}

              {/* T — температура: дві паралельні горизонтальні риски. */}
              <Line
                points={[
                  cx - baseHalf,
                  stemBottom - 3 * scale,
                  cx + baseHalf,
                  stemBottom - 3 * scale,
                ]}
                stroke={stroke}
                strokeWidth={sw}
              />
              <Line
                points={[
                  cx - baseHalf,
                  stemBottom + 2 * scale,
                  cx + baseHalf,
                  stemBottom + 2 * scale,
                ]}
                stroke={stroke}
                strokeWidth={sw}
              />
            </>
          )}

          {mark === "conductivity" && (
            <>
              {/* ae / pH — електропровідність / pH: довша нижня риска. */}
              <Line
                points={[
                  cx - 13 * scale,
                  stemBottom,
                  cx + 13 * scale,
                  stemBottom,
                ]}
                stroke={stroke}
                strokeWidth={sw}
              />
            </>
          )}
        </>
      )}
    </Group>
  );
}

function formatRealtimeValue(zvt, runtime) {
  if (!runtime) return "";

  const telemetry = zvt.integration?.telemetry;
  if (!telemetry?.enabled || !telemetry?.showOnCanvas) {
    return "";
  }

  if (runtime.connectionState === "error") {
    return "ERR";
  }

  if (runtime.value === null || runtime.value === undefined) {
    return runtime.connectionState === "connecting"
      ? "..."
      : "";
  }

  const precision = Number.isFinite(telemetry.precision)
    ? telemetry.precision
    : 2;

  const numeric = Number(runtime.value);
  const valueText = Number.isFinite(numeric)
    ? numeric.toFixed(precision)
    : String(runtime.value);

  const unit = runtime.unit || telemetry.unit || "";

  return unit
    ? `${valueText} ${unit}`
    : valueText;
}

  const receivedAt = runtime?.receivedAt;
  if (!receivedAt) return false;

  const staleAfterMs =
    zvt.integration?.telemetry?.staleAfterMs ?? 15000;

  return Date.now() - receivedAt > staleAfterMs;
}

function sameAttachedType(a, b) {
  return (
    a?.type &&
    b?.type &&
    a.type === b.type
  );
}

function hasAttachedId(value) {
  return !!(
    value?.id &&
    ["node", "edge"].includes(
      value.type
    )
  );
}

/**
 * Production rule:
 *
 * ZVT can join branch only if they share the same semantic root.
 *
 * node:
 *   zvt.attachedTo.type === "node"
 *   branch.attachedTo.type === "node"
 *   IDs equal
 *
 * edge:
 *   IDs equal
 *
 * segmentId:
 *   default policy = same edge is enough.
 *   strictEdgeSegment = true can additionally require same segmentId
 *   when both sides have segmentId.
 */
export function validateZvtBranchCompatibility(
  zvt,
  branch,
  {
    strictEdgeSegment = false,

    // Legacy mode:
    // ZVT without attachedTo may be allowed only by explicit opt-in.
    allowUnattachedZvt = false,
  } = {}
) {
  if (!zvt) {
    return {
      ok: false,
      code: "ZVT_NOT_FOUND",
      message: "ЗВТ не знайдено.",
    };
  }

  if (!branch) {
    return {
      ok: false,
      code: "BRANCH_NOT_FOUND",
      message:
        "Групову лінію ЗВТ не знайдено.",
    };
  }

  if (
    !hasAttachedId(
      branch.attachedTo
    )
  ) {
    return {
      ok: false,
      code:
        "BRANCH_ROOT_INVALID",
      message:
        "Групова лінія не має коректного root node/edge.",
    };
  }

  if (
    !hasAttachedId(
      zvt.attachedTo
    )
  ) {
    if (allowUnattachedZvt) {
      return {
        ok: true,
        code:
          "LEGACY_ZVT_ALLOWED",
        warning: true,
        message:
          "ЗВТ не має attachedTo. Підключення дозволено лише у legacy-режимі.",
      };
    }

    return {
      ok: false,
      code:
        "ZVT_ROOT_MISSING",
      message:
        "ЗВТ не прив'язаний до node або edge. Спочатку задайте основний об'єкт приєднання.",
    };
  }

  if (
    !sameAttachedType(
      zvt.attachedTo,
      branch.attachedTo
    )
  ) {
    return {
      ok: false,
      code:
        "ROOT_TYPE_MISMATCH",
      message:
        `ЗВТ прив'язаний до ${zvt.attachedTo.type}, ` +
        `а branch — до ${branch.attachedTo.type}.`,
    };
  }

  if (
    zvt.attachedTo.id !==
    branch.attachedTo.id
  ) {
    return {
      ok: false,
      code:
        "ROOT_ID_MISMATCH",
      message:
        "ЗВТ і групова лінія належать до різних об'єктів.",
    };
  }

  if (
    strictEdgeSegment &&
    zvt.attachedTo.type ===
      "edge"
  ) {
    const zvtSegment =
      zvt.attachedTo
        .segmentId || null;

    const branchSegment =
      branch.attachedTo
        .segmentId || null;

    if (
      zvtSegment &&
      branchSegment &&
      zvtSegment !==
        branchSegment
    ) {
      return {
        ok: false,
        code:
          "EDGE_SEGMENT_MISMATCH",
        message:
          "ЗВТ і branch прив'язані до різних сегментів одного connector.",
      };
    }
  }

  return {
    ok: true,
    code: "OK",
    message:
      "ЗВТ сумісний з груповою лінією.",
  };
}

/**
 * Повертає options для UI.
 */
export function buildZvtBranchCompatibilityOptions(
  zvt,
  branches,
  options = {}
) {
  return (branches || []).map(
    (branch) => {
      const validation =
        validateZvtBranchCompatibility(
          zvt,
          branch,
          options
        );

      return {
        branch,
        validation,
        disabled:
          !validation.ok,
      };
    }
  );
}

/**
 * Audit existing document:
 * корисно при відкритті старих схем.
 */
export function auditZvtBranchMembership(
  zvts,
  branches,
  options = {}
) {
  const branchById =
    new Map(
      (branches || []).map(
        (branch) => [
          branch.id,
          branch,
        ]
      )
    );

  const issues = [];

  for (const zvt of zvts || []) {
    const branchId =
      zvt.route?.branchId;

    if (!branchId) {
      continue;
    }

    const branch =
      branchById.get(
        branchId
      );

    const validation =
      validateZvtBranchCompatibility(
        zvt,
        branch,
        options
      );

    if (!validation.ok) {
      issues.push({
        zvtId: zvt.id,
        branchId,
        ...validation,
      });
    }
  }

  return issues;
}

export function createZvtBranch({
  id = crypto.randomUUID(),
  attachedTo,
  route = {},
  style = {},
} = {}) {
  if (
    !attachedTo ||
    !["node", "edge"].includes(attachedTo.type) ||
    !attachedTo.id
  ) {
    throw new Error(
      "ZVT branch root must be attached to node or edge"
    );
  }

  return {
    id,
    kind: "zvtBranch",

    // ROOT CONNECTION — обов'язковий.
    attachedTo: {
      type: attachedTo.type,
      id: attachedTo.id,
      segmentId: attachedTo.segmentId || null,
    },

    route: {
      sourceSide:
        route.sourceSide || "auto",

      sourceOffset:
        route.sourceOffset ?? 18,

      // Координати у world/canvas space.
      // Першою точкою фактично є dynamic root anchor,
      // тому її сюди НЕ зберігаємо.
      vertices: Array.isArray(route.vertices)
        ? route.vertices
        : [],

      // Кінцева точка магістралі.
      endPoint: route.endPoint || null,
    },

    style: {
      stroke: style.stroke || "black",
      strokeWidth: style.strokeWidth ?? 2,
      dash: Array.isArray(style.dash)
        ? style.dash
        : [],
    },

    createdAt: Date.now(),
    updatedAt: Date.now(),
  };
}

export const ZVT_BRANCH_GRID_SIZE = 10;

export function snapZvtBranchValue(
  value,
  gridSize = ZVT_BRANCH_GRID_SIZE
) {
  if (!gridSize || gridSize <= 1) {
    return value;
  }

  return Math.round(value / gridSize) * gridSize;
}

export function snapZvtBranchPoint(
  point,
  gridSize = ZVT_BRANCH_GRID_SIZE
) {
  return {
    x: snapZvtBranchValue(
      point.x,
      gridSize
    ),
    y: snapZvtBranchValue(
      point.y,
      gridSize
    ),
  };
}

function samePoint(a, b) {
  return (
    a &&
    b &&
    a.x === b.x &&
    a.y === b.y
  );
}

function isHorizontal(a, b) {
  return a.y === b.y;
}

function isVertical(a, b) {
  return a.x === b.x;
}

function isOrthogonal(a, b) {
  return (
    isHorizontal(a, b) ||
    isVertical(a, b)
  );
}

function isCollinearOrthogonal(
  a,
  b,
  c
) {
  return (
    (
      a.x === b.x &&
      b.x === c.x
    ) ||
    (
      a.y === b.y &&
      b.y === c.y
    )
  );
}

function dedupePoints(points) {
  const result = [];

  for (const point of points) {
    if (
      !result.length ||
      !samePoint(
        result[result.length - 1],
        point
      )
    ) {
      result.push(point);
    }
  }

  return result;
}

export function simplifyOrthogonalPoints(
  points
) {
  const deduped =
    dedupePoints(points);

  if (deduped.length <= 2) {
    return deduped;
  }

  const result = [
    deduped[0],
  ];

  for (
    let i = 1;
    i < deduped.length - 1;
    i++
  ) {
    const prev =
      result[result.length - 1];

    const current =
      deduped[i];

    const next =
      deduped[i + 1];

    if (
      isCollinearOrthogonal(
        prev,
        current,
        next
      )
    ) {
      continue;
    }

    result.push(current);
  }

  result.push(
    deduped[deduped.length - 1]
  );

  return result;
}

function distanceSq(a, b) {
  const dx =
    a.x - b.x;

  const dy =
    a.y - b.y;

  return dx * dx + dy * dy;
}

function projectToSegment(
  point,
  a,
  b
) {
  const dx =
    b.x - a.x;

  const dy =
    b.y - a.y;

  const lengthSq =
    dx * dx + dy * dy;

  if (!lengthSq) {
    return {
      point: { ...a },
      t: 0,
      distanceSq:
        distanceSq(point, a),
    };
  }

  let t =
    (
      (point.x - a.x) * dx +
      (point.y - a.y) * dy
    ) /
    lengthSq;

  t =
    Math.max(
      0,
      Math.min(1, t)
    );

  const projected = {
    x: a.x + dx * t,
    y: a.y + dy * t,
  };

  return {
    point: projected,
    t,
    distanceSq:
      distanceSq(
        point,
        projected
      ),
  };
}

export function resolveZvtBranchRootAnchor(
  branch,

  edges
) {
  const referencePoint =

  const pseudoZvt = {

    },

    position:

      side: null,

  );
}

function createDefaultBranchEnd(
  root,
  sourceSide
) {
  const distance = 150;

  switch (sourceSide) {
    case "left":
      return {
        x: root.x - distance,
        y: root.y,
      };

    case "top":
      return {
        x: root.x,
        y: root.y - distance,
      };

    case "bottom":
      return {
        x: root.x,
        y: root.y + distance,
      };

    case "right":
    default:
      return {
        x: root.x + distance,
        y: root.y,
      };
  }
}

/**
 * Гарантує, що кожен наступний segment є горизонтальним або вертикальним.
 *
 * Якщо в persisted JSON випадково потрапили діагональні точки,
 * функція вставить проміжний bend.
 */
export function orthogonalizeBranchPoints(
  points,
  {
    prefer = "horizontal-first",
  } = {}
) {
  if (!points?.length) {
    return [];
  }

  const result = [
    { ...points[0] },
  ];

  for (
    let i = 1;
    i < points.length;
    i++
  ) {
    const prev =
      result[result.length - 1];

    const next = {
      ...points[i],
    };

    if (
      isOrthogonal(
        prev,
        next
      )
    ) {
      result.push(next);
      continue;
    }

    const bend =
      prefer === "vertical-first"
        ? {
            x: prev.x,
            y: next.y,
          }
        : {
            x: next.x,
            y: prev.y,
          };

    result.push(
      bend,
      next
    );
  }

  return simplifyOrthogonalPoints(
    result
  );
}

/**
 * Повна route geometry branch.
 * root є dynamic і НЕ зберігається в persisted vertices.
 */
export function buildZvtBranchPoints(
  branch,
  nodes,
  edges
) {
  const root =
    resolveZvtBranchRootAnchor(
      branch,
      nodes,
      edges
    );

  const vertices =
    branch.route?.vertices || [];

  const endPoint =
    branch.route?.endPoint ||
    createDefaultBranchEnd(
      root,
      root.side ||
        branch.route?.sourceSide ||
        "right"
    );

  return orthogonalizeBranchPoints(
    [
      root,
      ...vertices,
      endPoint,
    ],
    {
      prefer:
        branch.route
          ?.orthogonalPreference ||
        "horizontal-first",
    }
  );
}

/**
 * Ортогональне переміщення interior vertex.
 *
 * Замість вільного x/y:
 * - орієнтація prev->vertex зберігається;
 * - орієнтація vertex->next зберігається;
 * - сусідні точки коригуються за потреби.
 *
 * Повертає НОВИЙ масив повних points включно root/end.
 */
export function moveOrthogonalBranchVertex(
  points,
  index,
  nextPoint,
  {
    gridSize =
      ZVT_BRANCH_GRID_SIZE,
  } = {}
) {
  if (
    index <= 0 ||
    index >= points.length - 1
  ) {
    return points;
  }

  const snapped =
    snapZvtBranchPoint(
      nextPoint,
      gridSize
    );

  const result =
    points.map((p) => ({
      ...p,
    }));

  const prev =
    result[index - 1];

  const current =
    result[index];

  const next =
    result[index + 1];

  const prevHorizontal =
    isHorizontal(
      prev,
      current
    );

  const nextHorizontal =
    isHorizontal(
      current,
      next
    );

  /**
   * Corner:
   * prev H + next V
   * або prev V + next H
   *
   * Тоді можна вільно рухати corner,
   * а координати сусідів підлаштовуємо.
   */
  if (
    prevHorizontal &&
    !nextHorizontal
  ) {
    result[index] = snapped;

    result[index - 1] = {
      ...prev,
      y: snapped.y,
    };

    result[index + 1] = {
      ...next,
      x: snapped.x,
    };
  } else if (
    !prevHorizontal &&
    nextHorizontal
  ) {
    result[index] = snapped;

    result[index - 1] = {
      ...prev,
      x: snapped.x,
    };

    result[index + 1] = {
      ...next,
      y: snapped.y,
    };
  } else if (
    prevHorizontal &&
    nextHorizontal
  ) {
    /**
     * Обидва сегменти горизонтальні.
     * Рухається тільки Y,
     * X залишаємо.
     */
    result[index] = {
      x: current.x,
      y: snapped.y,
    };

    result[index - 1] = {
      ...prev,
      y: snapped.y,
    };

    result[index + 1] = {
      ...next,
      y: snapped.y,
    };
  } else {
    /**
     * Обидва вертикальні.
     * Рухається тільки X.
     */
    result[index] = {
      x: snapped.x,
      y: current.y,
    };

    result[index - 1] = {
      ...prev,
      x: snapped.x,
    };

    result[index + 1] = {
      ...next,
      x: snapped.x,
    };
  }

  return simplifyOrthogonalPoints(
    orthogonalizeBranchPoints(
      result
    )
  );
}

/**
 * Рух endpoint.
 *
 * Останній segment лишається ортогональним:
 * - якщо був horizontal -> endpoint рухаємо по x,
 *   а y прив'язуємо до prev;
 * - якщо vertical -> endpoint рухаємо по y,
 *   x прив'язуємо до prev.
 *
 * Якщо користувач тягне далеко по іншій осі —
 * створюємо новий corner перед endpoint.
 */
export function moveOrthogonalBranchEndpoint(
  points,
  nextPoint,
  {
    gridSize =
      ZVT_BRANCH_GRID_SIZE,
  } = {}
) {
  if (
    !points ||
    points.length < 2
  ) {
    return points;
  }

  const snapped =
    snapZvtBranchPoint(
      nextPoint,
      gridSize
    );

  const result =
    points.map((p) => ({
      ...p,
    }));

  const lastIndex =
    result.length - 1;

  const prev =
    result[lastIndex - 1];

  const current =
    result[lastIndex];

  if (
    isHorizontal(
      prev,
      current
    )
  ) {
    result[lastIndex] = {
      x: snapped.x,
      y: prev.y,
    };

    if (
      Math.abs(
        snapped.y - prev.y
      ) >= gridSize
    ) {
      result.splice(
        lastIndex,
        0,
        {
          x: snapped.x,
          y: prev.y,
        }
      );

      result[result.length - 1] = {
        x: snapped.x,
        y: snapped.y,
      };
    }
  } else {
    result[lastIndex] = {
      x: prev.x,
      y: snapped.y,
    };

    if (
      Math.abs(
        snapped.x - prev.x
      ) >= gridSize
    ) {
      result.splice(
        lastIndex,
        0,
        {
          x: prev.x,
          y: snapped.y,
        }
      );

      result[result.length - 1] = {
        x: snapped.x,
        y: snapped.y,
      };
    }
  }

  return simplifyOrthogonalPoints(
    orthogonalizeBranchPoints(
      result
    )
  );
}

/**
 * Double click по segment.
 *
 * Для horizontal segment вставляємо ДВІ точки,
 * щоб створити ортогональний dog-leg,
 * а не одну точку, яка дала б діагоналі.
 *
 * Для UX користувач потім може drag нову пару.
 */
export function insertOrthogonalBranchBend(
  points,
  segmentIndex,
  clickPoint,
  {
    gridSize =
      ZVT_BRANCH_GRID_SIZE,
    bendOffset =
      ZVT_BRANCH_GRID_SIZE * 2,
  } = {}
) {
  if (
    segmentIndex < 0 ||
    segmentIndex >=
      points.length - 1
  ) {
    return points;
  }

  const a =
    points[segmentIndex];

  const b =
    points[segmentIndex + 1];

  const snapped =
    snapZvtBranchPoint(
      clickPoint,
      gridSize
    );

  const result =
    points.map((p) => ({
      ...p,
    }));

  if (
    isHorizontal(a, b)
  ) {
    const x =
      Math.max(
        Math.min(
          snapped.x,
          Math.max(a.x, b.x)
        ),
        Math.min(a.x, b.x)
      );

    const y =
      snapZvtBranchValue(
        snapped.y === a.y
          ? a.y + bendOffset
          : snapped.y,
        gridSize
      );

    result.splice(
      segmentIndex + 1,
      0,
      {
        x,
        y: a.y,
      },
      {
        x,
        y,
      },
      {
        x:
          Math.min(
            Math.max(
              x + bendOffset,
              Math.min(a.x, b.x)
            ),
            Math.max(a.x, b.x)
          ),
        y,
      },
      {
        x:
          Math.min(
            Math.max(
              x + bendOffset,
              Math.min(a.x, b.x)
            ),
            Math.max(a.x, b.x)
          ),
        y: a.y,
      }
    );
  } else {
    const y =
      Math.max(
        Math.min(
          snapped.y,
          Math.max(a.y, b.y)
        ),
        Math.min(a.y, b.y)
      );

    const x =
      snapZvtBranchValue(
        snapped.x === a.x
          ? a.x + bendOffset
          : snapped.x,
        gridSize
      );

    result.splice(
      segmentIndex + 1,
      0,
      {
        x: a.x,
        y,
      },
      {
        x,
        y,
      },
      {
        x,
        y:
          Math.min(
            Math.max(
              y + bendOffset,
              Math.min(a.y, b.y)
            ),
            Math.max(a.y, b.y)
          ),
      },
      {
        x: a.x,
        y:
          Math.min(
            Math.max(
              y + bendOffset,
              Math.min(a.y, b.y)
            ),
            Math.max(a.y, b.y)
          ),
      }
    );
  }

  return simplifyOrthogonalPoints(
    orthogonalizeBranchPoints(
      result
    )
  );
}

/**
 * Видалення interior vertex.
 *
 * Після видалення route знову orthogonalize.
 */
export function removeOrthogonalBranchVertex(
  points,
  index
) {
  if (
    index <= 0 ||
    index >= points.length - 1
  ) {
    return points;
  }

  const result =
    points.filter(
      (_, i) =>
        i !== index
    );

  return simplifyOrthogonalPoints(
    orthogonalizeBranchPoints(
      result
    )
  );
}

/**
 * Перетворити full points назад у persisted route.
 * points[0] = dynamic root, не серіалізуємо.
 * остання = endPoint.
 */
export function fullBranchPointsToRoute(
  points,
  oldRoute = {}
) {
  if (
    !points ||
    points.length < 2
  ) {
    return {
      ...oldRoute,
      vertices: [],
      endPoint: null,
    };
  }

  const middle =
    points.slice(
      1,
      -1
    );

  return {
    ...oldRoute,

    vertices:
      middle.map(
        (point, index) => ({
          id:
            point.id ||
            oldRoute.vertices?.[index]
              ?.id ||
            crypto.randomUUID(),
          x: point.x,
          y: point.y,
        })
      ),

    endPoint: {
      x:
        points[
          points.length - 1
        ].x,

      y:
        points[
          points.length - 1
        ].y,
    },
  };
}

export function findNearestBranchTap(
  branchPoints,
  zvtPosition
) {
  let best = null;

  for (
    let i = 0;
    i < branchPoints.length - 1;
    i++
  ) {
    const a =
      branchPoints[i];

    const b =
      branchPoints[i + 1];

    const candidate =
      projectToSegment(
        zvtPosition,
        a,
        b
      );

    if (
      !best ||
      candidate.distanceSq <
        best.distanceSq
    ) {
      best = {
        segmentIndex: i,
        t: candidate.t,
        point:
          candidate.point,
        distanceSq:
          candidate.distanceSq,
      };
    }
  }

  return best;
}

export function resolveFixedBranchTap(
  branchPoints,
  tap
) {
  const index =
    Math.max(
      0,
      Math.min(
        branchPoints.length - 2,
        tap?.segmentIndex ?? 0
      )
    );

  const a =
    branchPoints[index];

  const b =
    branchPoints[index + 1];

  const t =
    Math.max(
      0,
      Math.min(
        1,
        tap?.t ?? 0.5
      )
    );

  return {
    segmentIndex: index,
    t,

    point: {
      x:
        a.x +
        (b.x - a.x) * t,

      y:
        a.y +
        (b.y - a.y) * t,
    },
  };
}

export function resolveZvtBranchTap(
  branchPoints,
  zvt
) {
  const tap =
    zvt.route?.branchTap;

  if (
    tap?.mode === "fixed"
  ) {
    return resolveFixedBranchTap(
      branchPoints,
      tap
    );
  }

  return findNearestBranchTap(
    branchPoints,
    zvt.position
  );
}

export function buildZvtBranchMemberLeader({
  branchPoints,
  zvt,
}) {
  const tap =
    resolveZvtBranchTap(
      branchPoints,
      zvt
    );

  if (!tap?.point) {
    return [];
  }

  return buildZvtLeaderRoute({
    anchor: {
      x: tap.point.x,
      y: tap.point.y,
      side: null,
    },

    symbol:
      zvt.position,

    radius:
      zvt.style?.radius ?? 22,

    sourceSide:
      null,

    symbolSide:
      zvt.route?.symbolSide ||
      "auto",

    sourceOffset:
      zvt.route
        ?.branchSourceOffset ?? 0,

    symbolOffset:
      zvt.route
        ?.symbolOffset ?? 12,
  });
}

export function branchPointsToKonva(
  branch,
  nodes,
  edges
) {
  return buildZvtBranchPoints(
    branch,
    nodes,
    edges
  ).flatMap(
    (point) => [
      point.x,
      point.y,
    ]
  );
}

/**
 * Hit-test segment by pointer position.
 */
export function findNearestBranchSegment(
  branchPoints,
  pointer
) {
  let best = null;

  for (
    let i = 0;
    i < branchPoints.length - 1;
    i++
  ) {
    const candidate =
      projectToSegment(
        pointer,
        branchPoints[i],
        branchPoints[i + 1]
      );

    if (
      !best ||
      candidate.distanceSq <
        best.distanceSq
    ) {
      best = {
        segmentIndex: i,
        point:
          candidate.point,
        distanceSq:
          candidate.distanceSq,
      };
    }
  }

  return best;
}

"use client";

import React from "react";
import {
  Circle,
  Group,
  Line,
} from "react-konva";

import { useCanvasStore } from "@/store/useCanvasStore";
import { canvasController } from "@/controllers/canvasController";
import ZvtSymbol from "@/components/zvt/ZvtSymbol";

import {
  buildZvtLeaderRoute,
  resolveZvtAnchor,
} from "@/lib/zvt/zvtGeometry";

export default function ZvtLayer() {
  const zvts = useCanvasStore((s) => s.zvts || []);
  const nodes = useCanvasStore((s) => s.nodes || []);
  const edges = useCanvasStore((s) => s.edges || []);
  const selection = useCanvasStore((s) => s.selection);
  const zvtRuntimeById = useCanvasStore(
    (s) => s.zvtRuntimeById || {}
  );

  const showRealtimeOnCanvas =
    useCanvasStore(
      (s) =>
        s.zvtSettings
          ?.showRealtimeOnCanvas ?? true
    );
  const setSelection = useCanvasStore((s) => s.setSelection);
  const accessMode = useCanvasStore(
    (s) => s.accessMode || "editor"
  );

  const readOnly = accessMode === "reader";

  return (
    <>
      {zvts.map((zvt) => {
        const selected =
          selection?.type === "zvt" &&
          selection?.id === zvt.id;

        /**
         * КЛЮЧОВА ЗМІНА:
         * anchor НЕ беремо зі старого zvt.anchor як постійну координату.
         * Він щоразу перераховується з актуального nodes[] / edges[].
         *
         * Тому після переміщення node лінія ЗВТ перебудовується разом із ним.
         */
        const anchor = resolveZvtAnchor(
          zvt,
          nodes,
          edges
        );

        const leaderPoints = buildZvtLeaderRoute({
          anchor,
          symbol: zvt.position,
          radius: zvt.style?.radius ?? 22,

          sourceSide:
            zvt.attachedTo?.type === "node"
              ? (
                  zvt.route?.sourceSide ||
                  zvt.route?.preferredSide ||
                  anchor.side
                )
              : null,

          symbolSide:
            zvt.route?.symbolSide || "auto",

          sourceOffset:
            zvt.route?.sourceOffset ?? 18,

          symbolOffset:
            zvt.route?.symbolOffset ?? 12,
        });

        return (
          <Group key={zvt.id}>
            {!zvt.route?.branchId &&
              leaderPoints.length >= 4 && (
              <Line
                points={leaderPoints}
                stroke={
                  selected
                    ? zvt.style?.selectedStroke ||
                      "#2F80ED"
                    : zvt.style?.stroke || "black"
                }
                strokeWidth={selected ? 2.5 : 2}
                lineCap="round"
                lineJoin="round"
                listening={false}
              />
            )}

            {/*
              Малий anchor marker показуємо тільки для вибраного ЗВТ.
              Це дуже корисно для візуальної перевірки:
              точка повинна лежати РІВНО на межі node.
            */}
            {selected &&
              zvt.attachedTo?.type === "node" && (
                <Circle
                  x={anchor.x}
                  y={anchor.y}
                  radius={4}
                  fill="white"
                  stroke={
                    zvt.style?.selectedStroke ||
                    "#2F80ED"
                  }
                  strokeWidth={2}
                  listening={false}
                />
              )}

            <ZvtSymbol
              zvt={zvt}
              runtime={zvtRuntimeById[zvt.id] || null}
              showRealtime={showRealtimeOnCanvas}
              selected={selected}
              draggable={!readOnly}
              onClick={(e) => {
                e.cancelBubble = true;
                setSelection?.({
                  type: "zvt",
                  id: zvt.id,
                });
              }}
              onDblClick={(e) => {
                e.cancelBubble = true;
                canvasController.openZvtEditor?.({
                  zvtId: zvt.id,
                  attachedTo: zvt.attachedTo,
                });
              }}
              onContextMenu={(e) => {
                e.evt.preventDefault();
                e.cancelBubble = true;
                canvasController.handleZvtContextMenu?.(
                  e,
                  zvt
                );
              }}
              onDragMove={(e) => {
                canvasController.moveZvt?.(
                  zvt.id,
                  {
                    x: e.target.x(),
                    y: e.target.y(),
                  }
                );
              }}
              onDragEnd={(e) => {
                canvasController.moveZvt?.(
                  zvt.id,
                  {
                    x: e.target.x(),
                    y: e.target.y(),
                  }
                );
              }}
            />
          </Group>
        );
      })}
    </>
  );
}

function toggleArrayValue(array, value) {
  if (array.includes(value)) {
    return array.filter((item) => item !== value);
  }

  return [...array, value];
}

function onlyDigits(value) {
  return value.replace(/\D/g, "");
}

              <Typography

            <Box sx={{ mt: 2 }}>
              <FormControl disabled={readOnly}>
                <FormLabel>

                </FormLabel>

                  value={symbolSide}
                  onChange={(e) =>
                    setSymbolSide(e.target.value)
                  }
                >
                  {ZVT_SYMBOL_CONNECTION_SIDES.map((item) => (
                    <FormControlLabel
                      key={item.value}
                      value={item.value}
                      control={<Radio />}
                      label={item.label}
                    />
                  ))}
                </RadioGroup>
              </FormControl>
            </Box>

            <Stack
              direction={{ xs: "column", sm: "row" }}
              spacing={2}
              sx={{ mt: 2 }}
            >
              <TextField
                label="Відступ від node"
                type="number"
                value={sourceOffset}
                onChange={(e) =>
                  setSourceOffset(e.target.value)
                }
                disabled={readOnly}
                inputProps={{
                  min: 0,
                  max: 100,
                  step: 1,
                }}
              />

              <TextField
                label="Відступ перед ЗВТ"
                type="number"
                value={symbolOffset}
                onChange={(e) =>
                  setSymbolOffset(e.target.value)
                }
                disabled={readOnly}
                inputProps={{
                  min: 0,
                  max: 100,

  const store = useCanvasStore.getState();

createZvtFromEditor({
  attachedTo,
  data,
  style = {},
  route = {},
  integration = {},
}) {
  const store = useCanvasStore.getState();

  if (
    (store.accessMode || "editor") === "reader"
  ) {
    return;
  }

  if (!attachedTo?.type || !attachedTo?.id) {
    return;
  }

  /**
   * Спочатку створюємо temp із route,
   * бо resolveZvtAnchor повинен уже знати sourceSide.
   */
  const tempZvt = createZvt({
    attachedTo,
    data,
    style,
    route,
    integration,
  });

  const anchor = resolveZvtAnchor(
    tempZvt,
    store.nodes,
    store.edges
  );

  const position = createInitialZvtPosition(
    anchor,
    anchor.side
  );

  const zvt = createZvt({
    attachedTo,
    anchor,
    position,
    data,
    style,
    route,
    integration,
  });

  store.addZvt?.(zvt);
  store.setSelection?.({
    type: "zvt",
    id: zvt.id,
  });
  store.clearZvtEditor?.();
},

updateZvtFromEditor(
  zvtId,
  data,
  style = {},
  route = {},
  integration = {}
) {
  const store = useCanvasStore.getState();

  if (
    (store.accessMode || "editor") === "reader"
  ) {
    return;
  }

  const zvt = (store.zvts || []).find(
    (item) => item.id === zvtId
  );

  if (!zvt) return;

  const next = updateZvtData(
    zvt,
    data,
    style,
    route,
    integration
  );

  store.replaceZvt?.(zvtId, next);
  store.clearZvtEditor?.();
},

createZvtBranch({

attachZvtToBranch(
  zvtId,
  branchId,
  {
    tapMode = "auto",
    segmentIndex = 0,
    t = 0.5,

    validationOptions = {
      strictEdgeSegment: false,
      allowUnattachedZvt: false,
    },
  } = {}
) {
  const store =
    useCanvasStore.getState();

  if (
    (store.accessMode || "editor") ===
    "reader"
  ) {
    return {
      ok: false,
      code: "READ_ONLY",
      message:
        "У режимі перегляду змінювати підключення ЗВТ не можна.",
    };
  }

  const branch =
    (store.zvtBranches || [])
      .find(
        (item) =>
          item.id === branchId
      );

  const zvt =
    (store.zvts || [])
      .find(
        (item) =>
          item.id === zvtId
      );

  const validation =
    validateZvtBranchCompatibility(
      zvt,
      branch,
      validationOptions
    );

  if (!validation.ok) {
    return validation;
  }

  store.replaceZvt?.(
    zvtId,
    {
      ...zvt,

      route: {
        ...zvt.route,

        branchId,

        branchTap: {
          mode: tapMode,

          ...(tapMode === "fixed"
            ? {
                segmentIndex,
                t,
              }
            : {}),
        },
      },

      updatedAt: Date.now(),
    }
  );

  return {
    ok: true,
    code: "ATTACHED",
    message:
      "ЗВТ підключено до групової лінії.",
  };
},

      store.edges || []
    );

  const nextPoints =
    moveOrthogonalBranchEndpoint(
      fullPoints,
      position,
      {
        gridSize:
          store.gridSize || 10,
      }
    );

  store.replaceZvtBranch?.(
    branchId,
    updateZvtBranch(
      branch,
      {
        route:
          fullBranchPointsToRoute(
            nextPoints,
            branch.route
          ),
      }
    )
  );
},

insertZvtBranchBend(
  branchId,
  segmentIndex,
  pointer
) {
  const store =
    useCanvasStore.getState();

  if (
    (store.accessMode || "editor") ===
    "reader"
  ) {
    return;
  }

  const branch =
    (store.zvtBranches || [])
      .find(
        (item) =>
          item.id === branchId
      );

  if (!branch) return;

  const fullPoints =
    buildZvtBranchPoints(
      branch,
      store.nodes || [],
      store.edges || []
    );

  const nextPoints =
    insertOrthogonalBranchBend(
      fullPoints,
      segmentIndex,
      pointer,
      {
        gridSize:
          store.gridSize || 10,
      }
    );

  store.replaceZvtBranch?.(
    branchId,
    updateZvtBranch(
      branch,
      {
        route:
          fullBranchPointsToRoute(
            nextPoints,
            branch.route
          ),
      }
    )
  );
},

deleteSelectedZvtBranchVertex() {
  const store =
    useCanvasStore.getState();

  if (
    (store.accessMode || "editor") ===
    "reader"
  ) {
    return;
  }

  const edit =
    store.zvtBranchEdit;

  if (
    !edit?.branchId ||
    !Number.isInteger(
      edit.vertexIndex
    )
  ) {
    return;
  }

  const branch =
    (store.zvtBranches || [])
      .find(
        (item) =>
          item.id ===
          edit.branchId
      );

  if (!branch) return;

  const fullPoints =
    buildZvtBranchPoints(
      branch,
      store.nodes || [],
      store.edges || []
    );

  const nextPoints =
    removeOrthogonalBranchVertex(
      fullPoints,
      edit.vertexIndex
    );

  store.replaceZvtBranch?.(
    branch.id,
    updateZvtBranch(
      branch,
      {
        route:
          fullBranchPointsToRoute(
            nextPoints,
            branch.route
          ),
      }
    )
  );

  store.clearZvtBranchEdit?.();
},

confirmZvtBranchPicker() {
  const store =
    useCanvasStore.getState();

  const picker =
    store.zvtBranchPicker;

  if (!picker?.zvtId) {
    return;
  }

  if (
    picker.selectedBranchId
  ) {
    const result =
      canvasController
        .attachZvtToBranch?.(
          picker.zvtId,
          picker.selectedBranchId,
          {
            tapMode: "auto",

            validationOptions: {
              strictEdgeSegment:
                false,

              allowUnattachedZvt:
                false,
            },
          }
        );

    if (
      result &&
      !result.ok
    ) {
      store.setZvtBranchPicker?.({
        ...picker,

        error: {
          code:
            result.code,

          message:
            result.message,
        },
      });

      return;
    }
  } else {
    canvasController
      .detachZvtFromBranch?.(
        picker.zvtId
      );
  }

  store.clearZvtBranchPicker?.();
},

createBranchForZvt(
  zvtId,
  {
    sourceSide = "auto",
  } = {}
) {
  const store =
    useCanvasStore.getState();

  if (
    (store.accessMode || "editor") ===
    "reader"
  ) {
    return;
  }

  const zvt =
    (store.zvts || [])
      .find(
        (item) =>
          item.id === zvtId
      );

  if (!zvt) return;

  const attachedTo =
    zvt.attachedTo;

  if (
    !attachedTo?.id ||
    !["node", "edge"].includes(
      attachedTo.type
    )
  ) {
    return;
  }

  const branch =
    createZvtBranch({
      attachedTo,

      route: {
        sourceSide:
          attachedTo.type ===
          "node"
            ? sourceSide
            : "auto",
      },
    });

  store.addZvtBranch?.(
    branch
  );

  canvasController
    .attachZvtToBranch?.(
      zvt.id,
      branch.id,
      {
        tapMode: "auto",
      }
    );

  store.setSelection?.({
    type: "zvtBranch",
    id: branch.id,
  });

  if (
    store.zvtBranchPicker
  ) {
    store.setZvtBranchPicker?.({
      ...store.zvtBranchPicker,
      selectedBranchId:
        branch.id,
      hoveredBranchId:
        branch.id,
    });
  }

  return branch;
},


validateZvtBranchSelection(

  return validateZvtBranchCompatibility(
    zvt,
    branch,
    {
      strictEdgeSegment:
        false,

      allowUnattachedZvt:
        false,

      ...options,
    }
  );
},

/**
 * Монтувати один раз поза Stage.
 */
export default function ZvtBranchKeyboardShortcuts() {
  const accessMode =
    useCanvasStore(
      (s) =>
        s.accessMode || "editor"
    );

  useEffect(() => {
    const handleKeyDown = (
      event
    ) => {
      if (
        accessMode === "reader"
      ) {
        return;
      }

      const target =
        event.target;

      const tagName =
        target?.tagName
          ?.toLowerCase();

      if (
        tagName === "input" ||
        tagName === "textarea" ||
        target?.isContentEditable
      ) {
        return;
      }

      if (
        event.key === "Delete" ||
        event.key === "Backspace"
      ) {
        const edit =
          useCanvasStore
            .getState()
            .zvtBranchEdit;

        if (
          edit?.branchId &&
          Number.isInteger(
            edit.vertexIndex
          )
        ) {
          event.preventDefault();

          canvasController
            .deleteSelectedZvtBranchVertex?.();
        }
      }
    };

    window.addEventListener(
      "keydown",
      handleKeyDown
    );

    return () => {
      window.removeEventListener(
        "keydown",
        handleKeyDown
      );
    };
  }, [accessMode]);

  return null;
}

function getAttachedToLabel(
  branch
) {
  if (
    branch.attachedTo?.type ===
    "node"
  ) {
    return (
      `Node: ` +
      `${branch.attachedTo.id}`
    );
  }

  if (
    branch.attachedTo?.type ===
    "edge"
  ) {
    const segment =
      branch.attachedTo
        ?.segmentId
        ? ` / segment: ${branch.attachedTo.segmentId}`
        : "";

    return (
      `Edge: ` +
      `${branch.attachedTo.id}` +
      segment
    );
  }

  return "Без root";
}

/**
 * Низькорівневий WebSocket client.
 *
 * Він НЕ знає про React/Konva.
 * Його задача:
 * - connect;
 * - subscribe sensorIds;
 * - normalize incoming messages;
 * - передати normalized telemetry у callback.
 *
 * Очікуваний payload за замовчуванням:
 *
 * {
 *   "sensorId": "SENSOR-PT-101",
 *   "value": 12.34,
 *   "timestamp": "2026-09-07T16:40:00Z",
 *   "quality": "good"
 * }
 *
 * Якщо backend матиме інший формат —
 * змінюємо тільки normalizeMessage/subscribeMessageFactory.
 */

export class ZvtRealtimeClient {
  constructor({
    url,
    onTelemetry,
    onStateChange,
    subscribeMessageFactory,
  }) {
    this.url = url;
    this.onTelemetry = onTelemetry;
    this.onStateChange = onStateChange;

    this.subscribeMessageFactory =
      subscribeMessageFactory ||
      ((sensorIds) => ({
        type: "subscribe",
        sensorIds,
      }));

    this.socket = null;
    this.sensorIds = [];
    this.reconnectTimer = null;
    this.closedManually = false;
  }

  connect(sensorIds = []) {
    this.sensorIds = [
      ...new Set(sensorIds.filter(Boolean)),
    ];

    if (!this.url || this.sensorIds.length === 0) {
      return;
    }

    this.closedManually = false;
    this.onStateChange?.("connecting");

    this.socket = new WebSocket(this.url);

    this.socket.onopen = () => {
      this.onStateChange?.("connected");
      this.sendSubscription();
    };

    this.socket.onmessage = (event) => {
      try {
        const raw = JSON.parse(event.data);
        const normalized = this.normalizeMessage(raw);

        if (normalized) {
          this.onTelemetry?.(normalized);
        }
      } catch (error) {
        console.warn(
          "[ZVT realtime] invalid message",
          error
        );
      }
    };

    this.socket.onerror = () => {
      this.onStateChange?.("error");
    };

    this.socket.onclose = () => {
      this.onStateChange?.("disconnected");

      if (!this.closedManually) {
        this.scheduleReconnect();
      }
    };
  }

  updateSubscriptions(sensorIds = []) {
    this.sensorIds = [
      ...new Set(sensorIds.filter(Boolean)),
    ];

    if (this.sensorIds.length === 0) {
      this.disconnect();
      return;
    }

    if (
      !this.socket ||
      this.socket.readyState === WebSocket.CLOSED
    ) {
      this.connect(this.sensorIds);
      return;
    }

    if (
      this.socket.readyState === WebSocket.OPEN
    ) {
      this.sendSubscription();
    }
  }

  sendSubscription() {
    if (
      !this.socket ||
      this.socket.readyState !== WebSocket.OPEN
    ) {
      return;
    }

    this.socket.send(
      JSON.stringify(
        this.subscribeMessageFactory(this.sensorIds)
      )
    );
  }

  normalizeMessage(raw) {
    if (!raw || !raw.sensorId) return null;

    return {
      sensorId: String(raw.sensorId),
      value: raw.value,
      timestamp:
        raw.timestamp || new Date().toISOString(),
      quality: raw.quality || "unknown",
      raw,
    };
  }

  scheduleReconnect() {
    clearTimeout(this.reconnectTimer);

    this.reconnectTimer = setTimeout(() => {
      this.connect(this.sensorIds);
    }, 3000);
  }

  disconnect() {
    this.closedManually = true;
    clearTimeout(this.reconnectTimer);

    if (this.socket) {
      this.socket.close();
      this.socket = null;
    }
  }
}

"use client";

import React, {
  useEffect,
  useMemo,
  useRef,
} from "react";

import { useCanvasStore } from "@/store/useCanvasStore";
import { ZvtRealtimeClient } from "@/services/zvt/zvtRealtimeClient";

/**
 * React bridge між persisted zvt.integration і runtime Zustand.
 *
 * Монтувати ОДИН раз поза <Stage>.
 *
 * НЕ створювати WebSocket у кожному ZvtSymbol.
 * Інакше 50 ЗВТ = 50 socket connections.
 */
export default function ZvtRealtimeBridge() {
  const zvts = useCanvasStore((s) => s.zvts || []);
  const setZvtRuntime = useCanvasStore(
    (s) => s.setZvtRuntime
  );

  const realtimeEnabled = useCanvasStore(
    (s) =>
      s.zvtSettings?.realtimeEnabled ?? true
  );

  const clientRef = useRef(null);

  const bindings = useMemo(() => {
    return (zvts || [])
      .filter(
        (zvt) =>
          zvt.integration?.telemetry?.enabled &&
          zvt.integration?.telemetry?.sensorId
      )
      .map((zvt) => ({
        zvtId: zvt.id,
        sensorId:
          zvt.integration.telemetry.sensorId,
        config: zvt.integration.telemetry,
      }));
  }, [zvts]);

  const bindingBySensorId = useMemo(() => {
    const map = new Map();

    for (const binding of bindings) {
      const list =
        map.get(binding.sensorId) || [];

      list.push(binding);
      map.set(binding.sensorId, list);
    }

    return map;
  }, [bindings]);

  const sensorIds = useMemo(
    () => bindings.map((item) => item.sensorId),
    [bindings]
  );

  useEffect(() => {
    const url =
      process.env.NEXT_PUBLIC_ZVT_WS_URL;

    if (!realtimeEnabled) {
      return;
    }

    if (!url) {
      // Поки backend не підключений — просто нічого не робимо.
      return;
    }

    const client = new ZvtRealtimeClient({
      url,

      onTelemetry: (message) => {
        const currentBindings =
          bindingBySensorId.get(message.sensorId) || [];

        for (const binding of currentBindings) {
          const config = binding.config;

          /**
           * Якщо backend надалі передаватиме raw payload
           * з іншим valueField, можна брати значення так:
           */
          const rawValue =
            message.raw?.[config.valueField] ??
            message.value;

          const timestamp =
            message.raw?.[config.timestampField] ??
            message.timestamp;

          const quality =
            message.raw?.[config.qualityField] ??
            message.quality;

          setZvtRuntime?.(
            binding.zvtId,
            {
              sensorId: message.sensorId,
              rawValue,
              value: rawValue,
              unit: config.unit || "",
              timestamp,
              quality,
              connectionState: "connected",
              receivedAt: Date.now(),
              error: null,
            }
          );
        }
      },

      onStateChange: (state) => {
        for (const binding of bindings) {
          setZvtRuntime?.(
            binding.zvtId,
            {
              sensorId: binding.sensorId,
              connectionState: state,
            }
          );
        }
      },
    });

    clientRef.current = client;
    client.connect(sensorIds);

    return () => {
      client.disconnect();
      clientRef.current = null;
    };
  }, [
    bindingBySensorId,
    bindings,
    sensorIds,
    setZvtRuntime,
    realtimeEnabled,
  ]);

  return null;
}

import { getZvtInstallPlace } from "@/lib/zvt/zvtDictionaries";
import { buildZvtLeaderRoute, buildZvtSignalGeometry, resolveZvtAnchor } from "@/lib/zvt/zvtGeometry";

function esc(value) {
  return String(value ?? "")
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;")
    .replaceAll('"', "&quot;");
}

function pointsToSvg(points) {
  const pairs = [];
  for (let i = 0; i < points.length; i += 2) {
    pairs.push(`${points[i]},${points[i + 1]}`);
  }
  return pairs.join(" ");
}

function buildParameterLabel(zvt) {
  return `${zvt.data?.parameter || ""}${zvt.data?.parameterSuffix || ""}`;
}

function buildFunctionLabel(zvt) {
  const signs = Array.isArray(zvt.data?.functionSigns)
    ? zvt.data.functionSigns.join("")
    : "";
  return `${signs}${zvt.data?.functionSuffix || ""}`;
}

function buildSignalLabel(zvt) {
  const mode = zvt.data?.recipientLabelMode || "M";

  let base = "";
  if (mode === "custom") {
    base = (zvt.data?.recipientCustomText || "").trim();
  } else {
    const recipients = Array.isArray(zvt.data?.recipients)
      ? zvt.data.recipients
      : [];
    base = recipients.length ? `M${recipients.join(",")}` : "M";
  }

  const sign = zvt.data?.actionSign || "none";
  const signPosition = zvt.data?.actionSignPosition || "before";

  if (!base || sign === "none") return base;
  return signPosition === "after"
    ? `${base} ${sign}`
    : `${sign} ${base}`;
}

export function renderZvtSvg(zvt, scene) {
  const nodes = scene.nodes || [];
  const edges = scene.edges || [];

  const x = zvt.position?.x ?? 0;
  const y = zvt.position?.y ?? 0;
  const r = zvt.style?.radius ?? 22;
  const installPlace = getZvtInstallPlace(zvt.data?.installPlace);
  const functionLabel = buildFunctionLabel(zvt);
  const parameterLabel = buildParameterLabel(zvt);
  const stroke = zvt.style?.stroke || "black";
  const strokeWidth = zvt.style?.strokeWidth ?? 2;
  const fontFamily = zvt.style?.fontFamily || "Roboto Condensed";
  const fontSize = zvt.style?.fontSize ?? 13;

  const anchor = resolveZvtAnchor(zvt, nodes, edges);
  const leader = buildZvtLeaderRoute({
    anchor,
    symbol: zvt.position,
    radius: r,
    sourceSide:
      zvt.attachedTo?.type === "node"
        ? (
            zvt.route?.sourceSide ||
            zvt.route?.preferredSide ||
            anchor.side
          )
        : null,
    symbolSide: zvt.route?.symbolSide || "auto",
    sourceOffset: zvt.route?.sourceOffset ?? 18,
    symbolOffset: zvt.route?.symbolOffset ?? 12,
  });
  const signal = buildZvtSignalGeometry(zvt);
  const signalLabel = buildSignalLabel(zvt);

  let svg = "";

  if (leader.length >= 4) {
    svg += `<polyline points="${pointsToSvg(leader)}" fill="none" stroke="${stroke}" stroke-width="${strokeWidth}" stroke-linecap="round" stroke-linejoin="round" />`;
  }

  svg += `<circle cx="${x}" cy="${y}" r="${r}" fill="white" stroke="${stroke}" stroke-width="${strokeWidth}" />`;

  if (zvt.data?.showParameterGraphicMark ?? true) {
    const graphicScale = zvt.style?.parameterGraphicScale ?? 1;
    svg += renderParameterGraphicMarkSvg(zvt, {
      x,
      y: y - r - 12 * graphicScale,
      stroke,
      strokeWidth: 1.6,
    });
  }

  if (installPlace.hasDivider) {
    svg += `<line x1="${x - r}" y1="${y}" x2="${x + r}" y2="${y}" stroke="${stroke}" stroke-width="${strokeWidth}" />`;
    svg += `<text x="${x}" y="${y - 8}" text-anchor="middle" dominant-baseline="middle" font-family="${fontFamily}" font-size="${fontSize}" font-weight="700">${esc(functionLabel)}</text>`;
    svg += `<text x="${x}" y="${y + 12}" text-anchor="middle" dominant-baseline="middle" font-family="${fontFamily}" font-size="${fontSize}" font-weight="700">${esc(parameterLabel)}</text>`;
  } else {
    // "За місцем": функціональна ознака + вид параметру без горизонтальної лінії.
    if (functionLabel && parameterLabel) {
      svg += `<text x="${x}" y="${y - 7}" text-anchor="middle" dominant-baseline="middle" font-family="${fontFamily}" font-size="${fontSize}" font-weight="700">${esc(functionLabel)}</text>`;
      svg += `<text x="${x}" y="${y + 8}" text-anchor="middle" dominant-baseline="middle" font-family="${fontFamily}" font-size="${fontSize}" font-weight="700">${esc(parameterLabel)}</text>`;
    } else {
      const single = functionLabel || parameterLabel;
      svg += `<text x="${x}" y="${y}" text-anchor="middle" dominant-baseline="middle" font-family="${fontFamily}" font-size="${fontSize}" font-weight="700">${esc(single)}</text>`;
    }
  }

  if (signal) {
    svg += `<polyline points="${pointsToSvg(signal.line)}" fill="none" stroke="black" stroke-width="2" stroke-dasharray="18 8" />`;
    svg += `<polygon points="${pointsToSvg(signal.arrowPoints)}" fill="black" stroke="black" stroke-width="1" />`;
    svg += `<text x="${signal.labelX}" y="${signal.labelY + 13}" font-family="${fontFamily}" font-size="${fontSize}" font-weight="700">${esc(signalLabel)}</text>`;
  }

  return svg;
}

export function renderZvtsSvg(scene) {
  return (scene.zvts || []).map((zvt) => renderZvtSvg(zvt, scene)).join("\n");
}

import {
  buildZvtBranchMemberLeader,
  buildZvtBranchPoints,
} from "@/lib/zvt/zvtBranchGeometry";

function flatPointsToSvg(
  flat
) {
  const pairs = [];

  for (
    let i = 0;
    i < flat.length;
    i += 2
  ) {
    pairs.push(
      `${flat[i]},${flat[i + 1]}`
    );
  }

  return pairs.join(" ");
}

function objectPointsToSvg(
  points
) {
  return points
    .map(
      (point) =>
        `${point.x},${point.y}`
    )
    .join(" ");
}

export function renderZvtBranchesSvg(
  scene
) {
  const branches =
    scene.zvtBranches || [];

  const zvts =
    scene.zvts || [];

  return branches
    .map((branch) => {
      /**
       * КЛЮЧОВО:
       * buildZvtBranchPoints()
       * вже:
       * - resolve dynamic root;
       * - orthogonalize;
       * - simplify.
       *
       * Тому SVG = Canvas geometry.
       */
      const branchPoints =
        buildZvtBranchPoints(
          branch,
          scene.nodes || [],
          scene.edges || []
        );

      const dash =
        branch.style?.dash?.length
          ? ` stroke-dasharray="${branch.style.dash.join(" ")}"`
          : "";

      let svg =
        `<polyline ` +
        `points="${objectPointsToSvg(branchPoints)}" ` +
        `fill="none" ` +
        `stroke="${branch.style?.stroke || "black"}" ` +
        `stroke-width="${branch.style?.strokeWidth ?? 2}" ` +
        `stroke-linecap="round" ` +
        `stroke-linejoin="round"` +
        `${dash} />`;

      const members =
        zvts.filter(
          (zvt) =>
            zvt.route?.branchId ===
            branch.id
        );

      for (
        const zvt of members
      ) {
        const leader =
          buildZvtBranchMemberLeader({
            branchPoints,
            zvt,
          });

        if (
          leader.length >= 4
        ) {
          svg +=
            `<polyline ` +
            `points="${flatPointsToSvg(leader)}" ` +
            `fill="none" ` +
            `stroke="${zvt.style?.stroke || "black"}" ` +
            `stroke-width="${zvt.style?.strokeWidth ?? 2}" ` +
            `stroke-linecap="round" ` +
            `stroke-linejoin="round" />`;
        }
      }

      return svg;
    })
    .join("\n");
}
