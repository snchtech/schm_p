// - editor:

export const USER_ROLES = Object.freeze({
  EDITOR: "editor",
  READER: "reader",
});

export const PERMISSIONS = Object.freeze({
  ELEMENT_INFO_VIEW: "elementInfo.view",
  ELEMENT_INFO_EDIT: "elementInfo.edit",
});

const ROLE_PERMISSIONS = Object.freeze({
  [USER_ROLES.EDITOR]: [
    PERMISSIONS.ELEMENT_INFO_VIEW,
    PERMISSIONS.ELEMENT_INFO_EDIT,
  ],
  [USER_ROLES.READER]: [PERMISSIONS.ELEMENT_INFO_VIEW],
});

export function hasPermission(role, permission) {
  return ROLE_PERMISSIONS[role]?.includes(permission) ?? false;
}

export function createDefaultElementInfo(existing = {}) {
  return {
    title: existing.title || "",

    // ID запису обладнання у майбутній БД.
    equipmentId: existing.equipmentId || null,

    // Не є джерелом істини. Це лише локальний snapshot для demo/offline UI.
    // Після підключення БД актуальне значення треба отримувати за equipmentId.
    equipmentSnapshot: existing.equipmentSnapshot || null,

    // Reserved for future system links.
    systemId: existing.systemId || null,
    externalId: existing.externalId || null,

    // Reserved for future dynamic parameters.
    parameters: existing.parameters || {},

    updatedAt: existing.updatedAt || null,
  };
}

export function patchElementInfo(existing = {}, patch = {}) {
  return {
    ...createDefaultElementInfo(existing),
    ...patch,
    updatedAt: new Date().toISOString(),
  };
}

const DEMO_EQUIPMENT = Object.freeze([
  {
    id: "eq-pump-001",
    code: "P-001",
    name: "Насос циркуляційний №1",
    type: "pump",
  },
  {
    id: "eq-valve-014",
    code: "V-014",
    name: "Засувка запірна №14",
    type: "valve",
  },
  {
    id: "eq-tank-003",
    code: "T-003",
    name: "Резервуар технологічний №3",
    type: "tank",
  },
  {
    id: "eq-sensor-021",
    code: "S-021",
    name: "Датчик тиску №21",
    type: "sensor",
  },
]);

export const equipmentRepository = {
  async listEquipment({ search = "" } = {}) {
    const normalizedSearch = search.trim().toLowerCase();

    if (!normalizedSearch) {
      return [...DEMO_EQUIPMENT];
    }

    return DEMO_EQUIPMENT.filter((item) => {
      const value = `${item.code} ${item.name} ${item.type}`.toLowerCase();
      return value.includes(normalizedSearch);
    });
  },

  async getEquipmentById(id) {
    return DEMO_EQUIPMENT.find((item) => item.id === id) || null;
  },
};

"use client";

import React, { useEffect, useMemo, useState } from "react";
import {
  Alert,
  Autocomplete,
  Button,
  Chip,
  CircularProgress,
  Dialog,
  DialogActions,
  DialogContent,
  DialogTitle,
  Divider,
  Stack,
  TextField,
  Typography,
} from "@mui/material";
import EditOutlinedIcon from "@mui/icons-material/EditOutlined";

import { useCanvasStore } from "@/store/useCanvasStore";
import { canvasController } from "@/controllers/canvasController";
import { createDefaultElementInfo } from "@/lib/elementInfo/elementInfoModel";
import { equipmentRepository } from "@/lib/equipment/equipmentRepository";
import {
  hasPermission,
  PERMISSIONS,
} from "@/lib/access/accessControl";

export default function ElementInfoCardModal() {
  const dialog = useCanvasStore((state) => state.elementInfoDialog);
  const nodes = useCanvasStore((state) => state.nodes);
  const edges = useCanvasStore((state) => state.edges);
  const userRole = useCanvasStore((state) => state.userRole);

  const [title, setTitle] = useState("");
  const [equipmentOptions, setEquipmentOptions] = useState([]);
  const [selectedEquipment, setSelectedEquipment] = useState(null);
  const [isEquipmentLoading, setIsEquipmentLoading] = useState(false);
  const [equipmentError, setEquipmentError] = useState("");

  const open = Boolean(dialog?.target);
  const isEditMode = dialog?.mode === "edit";
  const canEdit = hasPermission(userRole, PERMISSIONS.ELEMENT_INFO_EDIT);

  const entity = useMemo(() => {
    if (!dialog?.target) return null;

    if (dialog.target.type === "node") {
      return nodes.find((node) => node.id === dialog.target.id) || null;
    }

    if (dialog.target.type === "edge") {
      return edges.find((edge) => edge.id === dialog.target.id) || null;
    }

    return null;
  }, [dialog, nodes, edges]);

  const info = useMemo(
    () => createDefaultElementInfo(entity?.info || {}),
    [entity]
  );

  useEffect(() => {
    if (!open) return;

    let active = true;

    async function loadEquipment() {
      setIsEquipmentLoading(true);
      setEquipmentError("");

      try {
        const options = await equipmentRepository.listEquipment();
        if (!active) return;

        setEquipmentOptions(options);

        const selected =
          options.find((item) => item.id === info.equipmentId) ||
          info.equipmentSnapshot ||
          null;

        setSelectedEquipment(selected);
      } catch (error) {
        if (!active) return;
        console.error("Failed to load equipment dictionary", error);
        setEquipmentError("Не вдалося завантажити довідник обладнання.");
      } finally {
        if (active) setIsEquipmentLoading(false);
      }
    }

    setTitle(info.title || "");
    loadEquipment();

    return () => {
      active = false;
    };
  }, [open, dialog?.target?.type, dialog?.target?.id, info.title, info.equipmentId]);

  function handleClose() {
    canvasController.closeElementInfoCard?.();
  }

  function handleSave() {
    canvasController.saveElementInfoCard?.({
      title,
      equipment: selectedEquipment,
    });
  }

  function handleStartEdit() {
    canvasController.switchElementInfoToEdit?.();
  }

  if (!open) return null;

  return (
    <Dialog open={open} onClose={handleClose} fullWidth maxWidth="sm">
      <DialogTitle>
        {isEditMode ? "Редагування картки об’єкта" : "Картка об’єкта"}
      </DialogTitle>

      <DialogContent dividers>
        <Stack spacing={2.5}>
          <Stack direction="row" spacing={1} alignItems="center" flexWrap="wrap">
            <Chip
              size="small"
              label={dialog?.target?.type === "node" ? "Node" : "Edge"}
            />

            <Typography variant="caption" color="text.secondary">
              ID: {dialog?.target?.id}
            </Typography>

            <Chip
              size="small"
              variant="outlined"
              label={isEditMode ? "Редагування" : "Перегляд"}
            />
          </Stack>

          <Divider />

          <TextField
            label="Назва об’єкта"
            value={title}
            onChange={(event) => setTitle(event.target.value)}
            fullWidth
            disabled={!isEditMode}
            placeholder="Введіть назву об’єкта або з’єднання"
            inputProps={{ maxLength: 255 }}
          />

          <Autocomplete
            options={equipmentOptions}
            value={selectedEquipment}
            onChange={(_event, value) => setSelectedEquipment(value)}
            disabled={!isEditMode || isEquipmentLoading}
            loading={isEquipmentLoading}
            isOptionEqualToValue={(option, value) => option.id === value.id}
            getOptionLabel={(option) =>
              option ? `${option.code} — ${option.name}` : ""
            }
            renderInput={(params) => (
              <TextField
                {...params}
                label="Обладнання"
                placeholder="Оберіть обладнання з довідника"
                InputProps={{
                  ...params.InputProps,
                  endAdornment: (
                    <>
                      {isEquipmentLoading ? (
                        <CircularProgress size={20} />
                      ) : null}
                      {params.InputProps.endAdornment}
                    </>
                  ),
                }}
              />
            )}
            noOptionsText="Обладнання не знайдено"
          />

          {equipmentError && (
            <Alert severity="error">{equipmentError}</Alert>
          )}

          {!isEditMode && !selectedEquipment && (
            <Typography color="text.secondary">
              Обладнання не прив’язано.
            </Typography>
          )}

          <Alert severity="info">
            Зараз використовується локальний демонстраційний довідник. Після
            підключення БД у картці зберігатиметься equipmentId, а список буде
            завантажуватися через repository/API.
          </Alert>
        </Stack>
      </DialogContent>

      <DialogActions>
        <Button onClick={handleClose}>
          {isEditMode ? "Скасувати" : "Закрити"}
        </Button>

        {!isEditMode && canEdit && (
          <Button
            variant="outlined"
            startIcon={<EditOutlinedIcon />}
            onClick={handleStartEdit}
          >
            Редагувати
          </Button>
        )}

        {isEditMode && canEdit && (
          <Button
            variant="contained"
            onClick={handleSave}
            disabled={!title.trim()}
          >
            Зберегти
          </Button>
        )}
      </DialogActions>
    </Dialog>
  );
}
