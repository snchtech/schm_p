ORACLE 21 — СТРУКТУРА БД СИСТЕМИ РОЛЕЙ

Сумісність: ACCESS_ROLE / ACCESS_ASSIGNMENT / ACCESS_AUDIT з попереднім пакетом admin_roles_implementation_nextjs_konva.txt.

Таблиці:
1. ACCESS_ROLE — ролі.
2. ACCESS_PERMISSION — атомарні дозволи.
3. ACCESS_ROLE_PERMISSION — дозволи ролей.
4. ACCESS_ASSIGNMENT — роль користувача та область дії (unit/workshop/department/album/scheme/type).
5. ACCESS_AUDIT — аудит змін.
6. ACCESS_ORG_UNIT — необов'язкова ієрархія організації.
7. ACCESS_USER_ORG — необов'язкове членство в підрозділах.

APP_USER(USER_ID, USERNAME, UNIT_ID, ACTIVE) передбачається вже наявною; не створюємо дубль.

ВАЖЛИВО: для чинного admin API змінити UPDATE при відкликанні ролі так, щоб заповнювалися REVOKED_AT та REVOKED_BY, інакше спрацює CHECK.

SQL ДЛЯ ВИКОНАННЯ (також окремо у файлі .sql):

-- Oracle Database 21c. Access control schema for Next.js + Konva + Node.js.
-- VERSION 1.0. Run in target application schema, once, on a TEST database first.
-- Compatible with admin_roles_implementation_nextjs_konva.txt:
-- ACCESS_ROLE(ID,CODE,TITLE), ACCESS_ASSIGNMENT(ID,USER_ID,ROLE_ID,SCHEME_TYPE,
-- UNIT_ID,WORKSHOP_ID,DEPARTMENT_ID,ALBUM_ID,SCHEME_ID,ACTIVE,CREATED_AT,CREATED_BY),
-- ACCESS_AUDIT(ID,ACTOR_USER_ID,TARGET_USER_ID,ACTION,ASSIGNMENT_ID,ROLE_CODE,DETAILS_JSON,CREATED_AT).
-- Prerequisite: existing APP_USER(USER_ID NUMBER,USERNAME VARCHAR2,UNIT_ID NUMBER,ACTIVE NUMBER).
-- Do not execute CREATE TABLE statements if same-name tables already exist; migrate instead.

-- 1. Role dictionary
CREATE TABLE ACCESS_ROLE (
  ID NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY NOT NULL,
  CODE VARCHAR2(64 CHAR) NOT NULL,
  TITLE VARCHAR2(200 CHAR) NOT NULL,
  DESCRIPTION VARCHAR2(1000 CHAR),
  IS_SYSTEM NUMBER(1) DEFAULT 1 NOT NULL,
  ACTIVE NUMBER(1) DEFAULT 1 NOT NULL,
  CREATED_AT TIMESTAMP(6) DEFAULT SYSTIMESTAMP NOT NULL,
  CONSTRAINT PK_ACCESS_ROLE PRIMARY KEY (ID),
  CONSTRAINT UQ_ACCESS_ROLE_CODE UNIQUE (CODE),
  CONSTRAINT CK_ACCESS_ROLE_SYSTEM CHECK (IS_SYSTEM IN (0,1)),
  CONSTRAINT CK_ACCESS_ROLE_ACTIVE CHECK (ACTIVE IN (0,1))
);

-- 2. Permission dictionary. Authorization engine must use these codes consistently.
CREATE TABLE ACCESS_PERMISSION (
  ID NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY NOT NULL,
  CODE VARCHAR2(100 CHAR) NOT NULL,
  TITLE VARCHAR2(200 CHAR) NOT NULL,
  DESCRIPTION VARCHAR2(1000 CHAR),
  CONSTRAINT PK_ACCESS_PERMISSION PRIMARY KEY (ID),
  CONSTRAINT UQ_ACCESS_PERMISSION_CODE UNIQUE (CODE)
);

-- 3. Role-permission mapping. No per-user override in initial release.
CREATE TABLE ACCESS_ROLE_PERMISSION (
  ROLE_ID NUMBER NOT NULL,
  PERMISSION_ID NUMBER NOT NULL,
  CREATED_AT TIMESTAMP(6) DEFAULT SYSTIMESTAMP NOT NULL,
  CONSTRAINT PK_ACCESS_ROLE_PERMISSION PRIMARY KEY (ROLE_ID, PERMISSION_ID),
  CONSTRAINT FK_ARP_ROLE FOREIGN KEY (ROLE_ID) REFERENCES ACCESS_ROLE(ID),
  CONSTRAINT FK_ARP_PERMISSION FOREIGN KEY (PERMISSION_ID) REFERENCES ACCESS_PERMISSION(ID)
);
CREATE INDEX IX_ARP_PERMISSION ON ACCESS_ROLE_PERMISSION(PERMISSION_ID);

-- 4. Role assignment: each record is a role + its OWN scope.
-- NULL scope field = no restriction on that dimension; all NULL = global.
-- The application must enforce hierarchy consistency (scheme -> album -> unit etc.)
-- and administrator delegation rules before INSERT/UPDATE.
CREATE TABLE ACCESS_ASSIGNMENT (
  ID NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY NOT NULL,
  USER_ID NUMBER NOT NULL,
  ROLE_ID NUMBER NOT NULL,
  SCHEME_TYPE VARCHAR2(64 CHAR),
  UNIT_ID NUMBER,
  WORKSHOP_ID NUMBER,
  DEPARTMENT_ID NUMBER,
  ALBUM_ID NUMBER,
  SCHEME_ID NUMBER,
  ACTIVE NUMBER(1) DEFAULT 1 NOT NULL,
  CREATED_AT TIMESTAMP(6) DEFAULT SYSTIMESTAMP NOT NULL,
  CREATED_BY NUMBER,
  REVOKED_AT TIMESTAMP(6),
  REVOKED_BY NUMBER,
  CONSTRAINT PK_ACCESS_ASSIGNMENT PRIMARY KEY (ID),
  CONSTRAINT FK_AA_ROLE FOREIGN KEY (ROLE_ID) REFERENCES ACCESS_ROLE(ID),
  CONSTRAINT CK_AA_ACTIVE CHECK (ACTIVE IN (0,1)),
  CONSTRAINT CK_AA_REVOKE CHECK (ACTIVE = 1 OR REVOKED_AT IS NOT NULL)
);
CREATE INDEX IX_ACCESS_ASSIGNMENT_USER ON ACCESS_ASSIGNMENT(USER_ID,ACTIVE);
CREATE INDEX IX_AA_ROLE_ACTIVE ON ACCESS_ASSIGNMENT(ROLE_ID,ACTIVE);
CREATE INDEX IX_AA_SCOPE ON ACCESS_ASSIGNMENT(UNIT_ID,SCHEME_TYPE,ACTIVE);
CREATE INDEX IX_AA_SCHEME ON ACCESS_ASSIGNMENT(SCHEME_ID,ACTIVE);

-- Prevent duplicate ACTIVE assignments for identical role+scope; revoked rows remain in history.
-- Oracle unique indexes permit many NULLs, so NVL is used for nullable scope fields.
-- Reserved sentinel -1 must never be a valid scope ID.
CREATE UNIQUE INDEX UQ_AA_ACTIVE_SCOPE ON ACCESS_ASSIGNMENT (
  CASE WHEN ACTIVE=1 THEN USER_ID END,
  CASE WHEN ACTIVE=1 THEN ROLE_ID END,
  CASE WHEN ACTIVE=1 THEN NVL(SCHEME_TYPE,'~ALL~') END,
  CASE WHEN ACTIVE=1 THEN NVL(UNIT_ID,-1) END,
  CASE WHEN ACTIVE=1 THEN NVL(WORKSHOP_ID,-1) END,
  CASE WHEN ACTIVE=1 THEN NVL(DEPARTMENT_ID,-1) END,
  CASE WHEN ACTIVE=1 THEN NVL(ALBUM_ID,-1) END,
  CASE WHEN ACTIVE=1 THEN NVL(SCHEME_ID,-1) END
);
ALTER TABLE ACCESS_ASSIGNMENT ADD CONSTRAINT CK_AA_SCOPE_IDS CHECK (
  (UNIT_ID IS NULL OR UNIT_ID > 0) AND
  (WORKSHOP_ID IS NULL OR WORKSHOP_ID > 0) AND
  (DEPARTMENT_ID IS NULL OR DEPARTMENT_ID > 0) AND
  (ALBUM_ID IS NULL OR ALBUM_ID > 0) AND
  (SCHEME_ID IS NULL OR SCHEME_ID > 0) AND
  (SCHEME_TYPE IS NULL OR SCHEME_TYPE <> '~ALL~')
);

-- 5. Audit log; retain historical rows even if assignment is revoked.
CREATE TABLE ACCESS_AUDIT (
  ID NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY NOT NULL,
  ACTOR_USER_ID NUMBER NOT NULL,
  TARGET_USER_ID NUMBER NOT NULL,
  ACTION VARCHAR2(32 CHAR) NOT NULL,
  ASSIGNMENT_ID NUMBER,
  ROLE_CODE VARCHAR2(64 CHAR),
  DETAILS_JSON CLOB,
  CREATED_AT TIMESTAMP(6) DEFAULT SYSTIMESTAMP NOT NULL,
  CONSTRAINT PK_ACCESS_AUDIT PRIMARY KEY (ID),
  CONSTRAINT CK_ACCESS_AUDIT_ACTION CHECK (ACTION IN ('GRANT','REVOKE','UPDATE','BOOTSTRAP')),
  CONSTRAINT CK_ACCESS_AUDIT_JSON CHECK (DETAILS_JSON IS JSON)
);
CREATE INDEX IX_AUDIT_TARGET_TIME ON ACCESS_AUDIT(TARGET_USER_ID,CREATED_AT);
CREATE INDEX IX_AUDIT_ACTOR_TIME ON ACCESS_AUDIT(ACTOR_USER_ID,CREATED_AT);

-- 6. Optional organization structure. These tables are NOT needed by the current
-- admin UI, but support future many-to-many memberships and scoped delegation.
CREATE TABLE ACCESS_ORG_UNIT (
  ID NUMBER NOT NULL,
  CODE VARCHAR2(64 CHAR) NOT NULL,
  TITLE VARCHAR2(200 CHAR) NOT NULL,
  PARENT_ID NUMBER,
  UNIT_KIND VARCHAR2(32 CHAR) NOT NULL,
  ACTIVE NUMBER(1) DEFAULT 1 NOT NULL,
  CONSTRAINT PK_ACCESS_ORG_UNIT PRIMARY KEY (ID),
  CONSTRAINT UQ_ACCESS_ORG_CODE UNIQUE (CODE),
  CONSTRAINT FK_ACCESS_ORG_PARENT FOREIGN KEY (PARENT_ID) REFERENCES ACCESS_ORG_UNIT(ID),
  CONSTRAINT CK_ACCESS_ORG_ACTIVE CHECK (ACTIVE IN (0,1)),
  CONSTRAINT CK_ACCESS_ORG_KIND CHECK (UNIT_KIND IN ('UNIT','WORKSHOP','DEPARTMENT')),
  CONSTRAINT CK_ACCESS_ORG_SELF CHECK (PARENT_ID IS NULL OR PARENT_ID <> ID)
);
CREATE INDEX IX_ACCESS_ORG_PARENT ON ACCESS_ORG_UNIT(PARENT_ID);

CREATE TABLE ACCESS_USER_ORG (
  USER_ID NUMBER NOT NULL,
  ORG_UNIT_ID NUMBER NOT NULL,
  MEMBERSHIP_KIND VARCHAR2(32 CHAR) DEFAULT 'MEMBER' NOT NULL,
  ACTIVE NUMBER(1) DEFAULT 1 NOT NULL,
  CREATED_AT TIMESTAMP(6) DEFAULT SYSTIMESTAMP NOT NULL,
  CONSTRAINT PK_ACCESS_USER_ORG PRIMARY KEY (USER_ID,ORG_UNIT_ID,MEMBERSHIP_KIND),
  CONSTRAINT FK_ACCESS_USER_ORG FOREIGN KEY (ORG_UNIT_ID) REFERENCES ACCESS_ORG_UNIT(ID),
  CONSTRAINT CK_ACCESS_USER_ORG_ACTIVE CHECK (ACTIVE IN (0,1)),
  CONSTRAINT CK_ACCESS_USER_ORG_KIND CHECK (MEMBERSHIP_KIND IN ('MEMBER','MANAGER','ADMIN'))
);
CREATE INDEX IX_ACCESS_USER_ORG_UNIT ON ACCESS_USER_ORG(ORG_UNIT_ID,ACTIVE);

-- 7. Roles seed (idempotent)
MERGE INTO ACCESS_ROLE t USING (
  SELECT 'reader' CODE,'Перегляд' TITLE FROM DUAL UNION ALL
  SELECT 'editor','Редагування' FROM DUAL UNION ALL
  SELECT 'administrator','Адміністратор підрозділу' FROM DUAL UNION ALL
  SELECT 'super_administrator','Суперадміністратор' FROM DUAL
) s ON (t.CODE=s.CODE)
WHEN NOT MATCHED THEN INSERT(CODE,TITLE) VALUES(s.CODE,s.TITLE);

-- 8. Permission seed (idempotent)
MERGE INTO ACCESS_PERMISSION t USING (
 SELECT 'scheme.read' CODE,'Перегляд схем' TITLE FROM DUAL UNION ALL
 SELECT 'scheme.search','Пошук схем' FROM DUAL UNION ALL
 SELECT 'scheme.create','Створення схем' FROM DUAL UNION ALL
 SELECT 'scheme.update','Зміна схем' FROM DUAL UNION ALL
 SELECT 'scheme.delete','Видалення схем' FROM DUAL UNION ALL
 SELECT 'canvas.read','Перегляд полотна' FROM DUAL UNION ALL
 SELECT 'canvas.edit','Редагування полотна' FROM DUAL UNION ALL
 SELECT 'node.create','Створення вузлів' FROM DUAL UNION ALL
 SELECT 'node.update','Зміна вузлів' FROM DUAL UNION ALL
 SELECT 'node.delete','Видалення вузлів' FROM DUAL UNION ALL
 SELECT 'edge.create','Створення зв’язків' FROM DUAL UNION ALL
 SELECT 'edge.update','Зміна зв’язків' FROM DUAL UNION ALL
 SELECT 'edge.delete','Видалення зв’язків' FROM DUAL UNION ALL
 SELECT 'zvt.manage','Керування ЗВТ' FROM DUAL UNION ALL
 SELECT 'zvtBranch.manage','Керування гілками ЗВТ' FROM DUAL UNION ALL
 SELECT 'annotation.manage','Керування анотаціями' FROM DUAL UNION ALL
 SELECT 'specification.manage','Керування специфікаціями' FROM DUAL UNION ALL
 SELECT 'attachment.read','Перегляд вкладень' FROM DUAL UNION ALL
 SELECT 'attachment.manage','Керування вкладеннями' FROM DUAL UNION ALL
 SELECT 'notes.manage','Керування нотатками' FROM DUAL UNION ALL
 SELECT 'card.read','Перегляд інформаційної картки' FROM DUAL UNION ALL
 SELECT 'card.update','Редагування інформаційної картки' FROM DUAL UNION ALL
 SELECT 'access.users.read','Пошук користувачів для адміністрування' FROM DUAL UNION ALL
 SELECT 'access.assignments.read','Перегляд призначених ролей' FROM DUAL UNION ALL
 SELECT 'access.assignments.grant','Видача ролей' FROM DUAL UNION ALL
 SELECT 'access.assignments.revoke','Відкликання ролей' FROM DUAL UNION ALL
 SELECT 'access.audit.read','Перегляд аудиту' FROM DUAL
) s ON (t.CODE=s.CODE)
WHEN NOT MATCHED THEN INSERT(CODE,TITLE) VALUES(s.CODE,s.TITLE);

-- 9. Reader rights. Editor inherits reader rights by explicitly copying grants.
INSERT INTO ACCESS_ROLE_PERMISSION(ROLE_ID,PERMISSION_ID)
SELECT r.ID,p.ID FROM ACCESS_ROLE r CROSS JOIN ACCESS_PERMISSION p
WHERE r.CODE='reader' AND p.CODE IN (
 'scheme.read','scheme.search','canvas.read','attachment.read','card.read'
) AND NOT EXISTS (SELECT 1 FROM ACCESS_ROLE_PERMISSION rp WHERE rp.ROLE_ID=r.ID AND rp.PERMISSION_ID=p.ID);

INSERT INTO ACCESS_ROLE_PERMISSION(ROLE_ID,PERMISSION_ID)
SELECT r.ID,p.ID FROM ACCESS_ROLE r CROSS JOIN ACCESS_PERMISSION p
WHERE r.CODE='editor' AND p.CODE IN (
 'scheme.read','scheme.search','canvas.read','attachment.read','card.read',
 'scheme.create','scheme.update','scheme.delete','canvas.edit',
 'node.create','node.update','node.delete',
 'edge.create','edge.update','edge.delete',
 'zvt.manage','zvtBranch.manage','annotation.manage','specification.manage',
 'attachment.manage','notes.manage','card.update'
) AND NOT EXISTS (SELECT 1 FROM ACCESS_ROLE_PERMISSION rp WHERE rp.ROLE_ID=r.ID AND rp.PERMISSION_ID=p.ID);

INSERT INTO ACCESS_ROLE_PERMISSION(ROLE_ID,PERMISSION_ID)
SELECT r.ID,p.ID FROM ACCESS_ROLE r CROSS JOIN ACCESS_PERMISSION p
WHERE r.CODE='administrator' AND p.CODE IN (
 'access.users.read','access.assignments.read',
 'access.assignments.grant','access.assignments.revoke','access.audit.read'
) AND NOT EXISTS (SELECT 1 FROM ACCESS_ROLE_PERMISSION rp WHERE rp.ROLE_ID=r.ID AND rp.PERMISSION_ID=p.ID);

INSERT INTO ACCESS_ROLE_PERMISSION(ROLE_ID,PERMISSION_ID)
SELECT r.ID,p.ID FROM ACCESS_ROLE r CROSS JOIN ACCESS_PERMISSION p
WHERE r.CODE='super_administrator'
AND NOT EXISTS (SELECT 1 FROM ACCESS_ROLE_PERMISSION rp WHERE rp.ROLE_ID=r.ID AND rp.PERMISSION_ID=p.ID);

COMMIT;

-- Integration notes:
-- A) APP_USER and existing SCHEME/ALBUM/ORG tables remain external. Add FKs
--    only after verifying actual names, datatypes and migration quality.
-- B) The existing admin API's revoke SQL sets ACTIVE=0 only. Change it to:
--    UPDATE ACCESS_ASSIGNMENT SET ACTIVE=0,REVOKED_AT=SYSTIMESTAMP,
--      REVOKED_BY=:actorId WHERE ID=:id AND ACTIVE=1;
-- C) ACCESS_AUDIT.ACTION must be GRANT/REVOKE/UPDATE/BOOTSTRAP (as used by API).
-- D) Existing authorization code may use different permission codes. Align codes
--    before production; do not assume this seed is automatically compatible.
-- E) Authorization is enforced in Node.js API; Oracle tables alone do not secure it.
-- F) Oracle DDL commits implicitly; apply to a clean schema or use migrations.
