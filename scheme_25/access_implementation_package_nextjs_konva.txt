import { ROLE_PERMISSIONS } from './permissions';

function scopeMatches(scope = {}, resource = {}) {
  // Відсутність scope означає доступ до всіх ресурсів ЛИШЕ в межах
  // свідомо виданого глобального grant; у production глобальний grant
  // повинен видаватися виключно уповноваженим адміністратором.
  if (scope.schemeId != null && String(scope.schemeId) !== String(resource.schemeId)) return false;
  if (scope.albumId != null && String(scope.albumId) !== String(resource.albumId)) return false;
  if (scope.unitId != null && String(scope.unitId) !== String(resource.unitId)) return false;
  if (scope.workshopId != null && String(scope.workshopId) !== String(resource.workshopId)) return false;
  if (scope.departmentId != null && String(scope.departmentId) !== String(resource.departmentId)) return false;
  return true;
}

export function can(subject, permission, resource) {
  if (!subject || subject.status !== 'active' || !resource) return false;
  if (!resource.schemeId && permission !== 'access.manage') return false;
  const assignments = Array.isArray(subject.assignments) ? subject.assignments : [];
  return assignments.some((grant) => {
    if (grant.active === false) return false;
    const permissions = ROLE_PERMISSIONS[grant.role] || [];
    if (!permissions.includes('*') && !permissions.includes(permission)) return false;
    if (Array.isArray(grant.schemeTypes) && grant.schemeTypes.length > 0 &&
        !grant.schemeTypes.includes(resource.type)) return false;
    return scopeMatches(grant.scope || {}, resource);
  });
}

export function requirePermission(subject, permission, resource) {
  if (!can(subject, permission, resource)) {
    const error = new Error(`ACCESS_DENIED: ${permission}`);
    error.code = 'ACCESS_DENIED';
    throw error;
  }
}

// Застереження: це мінімальна allow-only модель. Для explicit deny,
// ієрархії підрозділів, делегування ролей, тимчасових прав потрібна
// окрема політика, узгоджена між frontend та backend.

export const useAccessStore = create((set, get) => ({
  subject: null,
  schemeResource: null,
  desiredMode: 'viewer',
  setSession: (subject) => set({ subject }),
  setSchemeResource: (schemeResource) => set({ schemeResource }),
  setDesiredMode: (desiredMode) => set({
    desiredMode: desiredMode === 'editor' ? 'editor' : 'viewer',
  }),
  clear: () => set({ subject: null, schemeResource: null, desiredMode: 'viewer' }),
  can: (permission, resource) => {
    const state = get();
    return can(state.subject, permission, resource || state.schemeResource);
  },
  getEffectiveMode: () => {
    const state = get();
    return state.desiredMode === 'editor' &&
      can(state.subject, 'scheme.edit', state.schemeResource)
      ? 'editor' : 'viewer';
  },
}));

import { useAccessStore } from './useAccessStore';
import { requirePermission } from './policy';

export function assertCanvasPermission(permission, resource = null) {
  const { subject, schemeResource } = useAccessStore.getState();
  requirePermission(subject, permission, resource || schemeResource);
}

// Зручна обгортка для існуючих методів; async/Promise не потрібні.
export function guarded(permission, action, getResource) {
  return (...args) => {
    const resource = getResource ? getResource(...args) : null;
    assertCanvasPermission(permission, resource);
    return action(...args);
  };
}

// НЕ ПЕРЕНОСЬТЕ саму перевірку тільки у Konva handlers.

export const useCanvasStore = create((set, get) => ({
  nodes: [], edges: [],
  addNode: (node) => {
    assertCanvasPermission(P.ELEMENT_CREATE);
    set((s) => ({ nodes: [...s.nodes, node] }));
  },
  moveNode: (id, point) => {
    assertCanvasPermission(P.ELEMENT_MOVE);
    set((s) => ({ nodes: s.nodes.map((n) =>
      n.id === id ? { ...n, x: point.x, y: point.y } : n
    ) }));
  },
  deleteNode: (id) => {
    assertCanvasPermission(P.ELEMENT_DELETE);
    set((s) => ({ nodes: s.nodes.filter((n) => n.id !== id) }));
  },
}));

  );

Наведено Express-подібний API. Потрібна адаптація до ваших route names,

// server/middleware/authorizeScheme.js
const { can } = require('../access/policy.cjs');

function authorizeScheme(permission) {
  return async (req, res, next) => {
    try {
      // req.user встановлює перевірений authentication middleware.
      // НІКОЛИ не приймайте userId/roles із req.body як доказ прав.
      if (!req.user?.id) return res.status(401).json({ error: 'UNAUTHENTICATED' });
      const schemeId = req.params.schemeId;
      const resource = await req.services.schemeRepository.getAccessMetadata(schemeId);
      if (!resource) return res.status(404).json({ error: 'NOT_FOUND' });
      const subject = await req.services.accessRepository.getSubject(req.user.id);
      if (!can(subject, permission, resource)) return res.status(403).json({ error: 'FORBIDDEN' });
      req.schemeResource = resource;
      return next();
    } catch (error) { return next(error); }
  };
}
module.exports = { authorizeScheme };

// server/routes/schemes.js
const express = require('express');
const { authorizeScheme } = require('../middleware/authorizeScheme');
const router = express.Router();
router.get('/:schemeId', authorizeScheme('scheme.view'), async (req, res, next) => {
  try {
    const data = await req.services.schemeRepository.getFull(req.params.schemeId);
    res.json(data);
  } catch (error) { next(error); }
});
router.put('/:schemeId', authorizeScheme('scheme.save'), async (req, res, next) => {
  try {
    // Додатково: validate payload, перевірити version/ETag,
    // обмежити розмір JSON і допустимі типи об'єктів.
    const result = await req.services.schemeRepository.saveTransaction({
      schemeId: req.params.schemeId,
      payload: req.body,
      actorId: req.user.id,
      expectedVersion: req.body.version,
    });
    res.json(result);
  } catch (error) { next(error); }
});
module.exports = router;

УВАГА: server/access/policy.cjs має реалізувати ТУ САМУ політику,
що й frontend/policy.js. Не варто механічно require() ES-модуль без
налаштованого транспілятора. Можна винести чистий engine в shared package
з відповідною збіркою або підтримувати тести на однакових fixtures.

ПЕРЕВІРЯТИ ОКРЕМО:
- GET scheme, list/search schemes, cards, attachments, download, export;
- POST/PUT/PATCH/DELETE nodes, edges, ZVT, branches, notes, annotations;
- POST/PUT scheme JSON та batch sync;
- адміністративні endpoint-и (з урахуванням scope адміністратора).

Якщо API приймає ВЕСЬ JSON схеми, SCHEME_SAVE дозволяє повний запис
лише в межах відповідного ресурсу. Якщо у майбутньому з'являться
часткові права на типи елементів, потрібен server-side diff/валідація
змін або окремі endpoint-и з точнішими permissions.

Захист від IDOR: metadata ресурсу брати з БД за schemeId, а не з
payload клієнта. При перенесенні схеми між підрозділами перевіряти
право як на вихідний, так і на цільовий scope.

DB транзакція: схема JSON + nodes/edges/ZVT/branches/attachments/
інші нормалізовані сутності мають записуватися атомарно через ваш
існуючий dbmanager з одним connection/transaction context.

  CONSTRAINT PK_ACCESS_ROLE_PERMISSION PRIMARY KEY (ROLE_ID, PERMISSION_ID)
);

CREATE TABLE ACCESS_ASSIGNMENT (
  ID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  USER_ID NUMBER NOT NULL,
  ROLE_ID NUMBER NOT NULL REFERENCES ACCESS_ROLE(ID),
  SCHEME_TYPE VARCHAR2(64),
  UNIT_ID NUMBER,
  WORKSHOP_ID NUMBER,
  DEPARTMENT_ID NUMBER,
  ALBUM_ID NUMBER,
  SCHEME_ID NUMBER,
  ACTIVE NUMBER(1) DEFAULT 1 NOT NULL CHECK (ACTIVE IN (0,1)),
  CREATED_AT TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
  CREATED_BY NUMBER
);
CREATE INDEX IX_ACCESS_ASSIGNMENT_USER ON ACCESS_ASSIGNMENT(USER_ID, ACTIVE);
CREATE INDEX IX_ACCESS_ASSIGNMENT_SCOPE ON ACCESS_ASSIGNMENT(SCHEME_ID, UNIT_ID, WORKSHOP_ID, DEPARTMENT_ID);

-- Уточнити FK до існуючих таблиць користувачів/схем/підрозділів.
-- NULL у scope означає wildcard лише для адміністративно затверджених
-- assignments; глобальні права не можна видавати за замовчуванням.
-- Для множини scheme types можна використовувати окрему таблицю
-- ACCESS_ASSIGNMENT_SCHEME_TYPE (ASSIGNMENT_ID, SCHEME_TYPE).
-- Для журналювання додати ACCESS_AUDIT із actor, target, action,
-- timestamp, before/after (за правилами безпеки даних).

ОБОВ'ЯЗКОВІ НЕГАТИВНІ ТЕСТИ:
1. Reader відкриває ?mode=editor -> actualMode viewer.
2. Reader викликає controller.deleteNode напряму -> ACCESS_DENIED.
3. Reader викликає store.deleteNode напряму -> ACCESS_DENIED.
4. Reader тисне Delete / Ctrl+V / Ctrl+Z -> документ не змінюється.
5. Reader намагається PUT /schemes/:id -> HTTP 403.
6. Editor W2 редагує схему W1 -> HTTP 403 та блокування UI.
7. Editor W2 читає схему U1 (reader grant) -> HTTP 200, edit 403.
8. Неавторизований користувач -> 401.
9. Схема не існує -> 404.
10. Роль змінено на сервері -> наступна операція враховує нові права.
11. Дві одночасні зміни -> version conflict 409.
12. Помилка в транзакції -> JSON та таблиці не розходяться.
13. Невідома мутаційна команда -> UNREGISTERED_COMMAND.
14. Перегляд вкладень reader дозволений, видалення заборонене.
15. Каскадне видалення branch+ZVT не обходить перевірку.

14. ПІДСУМОК: ЩО САМЕ ЗМІНЮВАТИ У ВАШИХ ФАЙЛАХ
============================================================
canvasController.js:
  + import assertCanvasPermission / guarded
  + COMMAND_PERMISSIONS
  + перевірка перед кожною мутаційною командою
  = поточні алгоритми переміщення/геометрії залишаються

useCanvasStore.js:
  + assertCanvasPermission перед кожним публічним mutator
  + окремий внутрішній шлях server hydration
  = поточна структура nodes/edges/zvt/zvtBranch/... залишається

Konva components:
  + draggable={canMove}
  + Transformer доступний лише editor
  + onDragEnd -> controller; rollback на відмову
  = поточна графіка не переписується

MUI toolbar/context menus:
  + disabled/hidden для команд без permission
  + editor/viewer switch із effective mode
  = читач може select, zoom, pan, переглядати картки

Node.js API:
  + authenticate + authorizeScheme middleware
  + metadata ресурсу з Oracle
  + транзакційне збереження
  = жодне клієнтське обмеження не вважається захистом API

Важлива межа готовності: наведені модулі policy, store, guard і зразки
контролера можна переносити в проєкт як основу. Але повністю автоматичний
copy-paste без адаптації неможливий без реальних файлів canvasController,
useCanvasStore, route handlers, схеми БД та auth middleware.
