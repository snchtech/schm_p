1. ЩО ПЕРЕВІРЕНО В ПОПЕРЕДНІХ ПАКЕТАХ

Порівняно oracle21_access_roles_schema.sql та admin_roles_implementation_nextjs_konva.txt.
Попередня модель мала ACCESS_ROLE, ACCESS_PERMISSION, ACCESS_ROLE_PERMISSION,
ACCESS_ASSIGNMENT, ACCESS_AUDIT, ACCESS_ORG_UNIT, ACCESS_USER_ORG.
Виявлені проблеми, які виправляє v2:
  1) Немає однозначного SCOPE_TYPE: комбінація nullable UNIT_ID / WORKSHOP_ID /
     DEPARTMENT_ID / ALBUM_ID / SCHEME_ID допускає неоднозначні призначення.
  2) Немає окремого DIVISION і ясної ієрархії DIVISION -> UNIT.
  3) SCHEME_TYPE — довільний рядок без зовнішнього ключа.
  4) Немає VALID_FROM / VALID_TO для тимчасових дозволів.
  5) Немає надійної моделі власності схеми і перевірки вкладеності scope.
  6) API адміністратора припускає APP_USER.UNIT_ID і старі колонки scope.
  7) Попередній seed для super_administrator НЕ надавав дозволів (можливо,
     це бажана модель з bypass, але її треба явно визначити).
  8) У попередній моделі є схема 'premises'; у новій прикладах — 'building'.
     НЕ перейменовувати наявні типи без перевірки фактичних кодів.

Цей пакет — цільова модель для НОВОЇ інсталяції або планової міграції.
НЕ запускати CREATE TABLE поверх чинних таблиць. Не запускати без бекапу.
Oracle DDL виконує неявні COMMIT. Порядок створення наведений нижче.

2. ПРИНЦИПИ

- Кожен ACCESS_ASSIGNMENT = одна роль + один тип схеми (або всі) + ОДНА
  область дії (GLOBAL / DIVISION / UNIT / ALBUM / SCHEME).
- Дозволи від різних призначень об'єднуються, але scope від одного запису
  НІКОЛИ не переноситься на інший.
- SCOPE_ID NULL тільки при GLOBAL; для решти SCOPE_ID > 0.
- SCHEME_TYPE_ID NULL означає «усі типи».
- Автор схеми НЕ є автоматично її єдиним редактором.
- Права доступу визначаються на сервері з актуальних даних Oracle.
- Відсутність призначення = заборона (deny by default).
- У таблицях SCHEME/ALBUM/APP_USER використовуйте фактичні назви свого
  проєкту. Нижче створено мінімальні довідники/зв'язки без нав'язування
  структури існуючих схем і альбомів.

CREATE TABLE ACCESS_SCHEME_TYPE (

  CONSTRAINT UQ_AST_CODE UNIQUE (CODE),
  CONSTRAINT CK_AST_ACTIVE CHECK (ACTIVE IN (0,1))

-- 3.2. Ролі
CREATE TABLE ACCESS_ROLE (
  ID NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY,
  CODE VARCHAR2(64 CHAR) NOT NULL,
  TITLE VARCHAR2(200 CHAR) NOT NULL,
  ACTIVE NUMBER(1) DEFAULT 1 NOT NULL,
  IS_SYSTEM NUMBER(1) DEFAULT 1 NOT NULL,
  CONSTRAINT PK_AR PRIMARY KEY (ID),
  CONSTRAINT UQ_AR_CODE UNIQUE (CODE),
  CONSTRAINT CK_AR_ACTIVE CHECK (ACTIVE IN (0,1)),
  CONSTRAINT CK_AR_SYSTEM CHECK (IS_SYSTEM IN (0,1))
);

-- 3.3. Дозволи
CREATE TABLE ACCESS_PERMISSION (
  ID NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY,
  CODE VARCHAR2(100 CHAR) NOT NULL,
  TITLE VARCHAR2(200 CHAR) NOT NULL,
  CONSTRAINT PK_AP PRIMARY KEY (ID),
  CONSTRAINT UQ_AP_CODE UNIQUE (CODE)
);

CREATE TABLE ACCESS_ROLE_PERMISSION (
  ROLE_ID NUMBER NOT NULL,
  PERMISSION_ID NUMBER NOT NULL,
  CONSTRAINT PK_ARP PRIMARY KEY (ROLE_ID, PERMISSION_ID),
  CONSTRAINT FK_ARP_ROLE FOREIGN KEY (ROLE_ID) REFERENCES ACCESS_ROLE(ID),
  CONSTRAINT FK_ARP_PERMISSION FOREIGN KEY (PERMISSION_ID) REFERENCES ACCESS_PERMISSION(ID)
);
CREATE INDEX IX_ARP_PERMISSION ON ACCESS_ROLE_PERMISSION(PERMISSION_ID);

-- 3.4. Організаційна структура (універсальне дерево).
-- Якщо довідник підрозділів уже існує, краще адаптувати FK до нього.
CREATE TABLE ACCESS_ORG_UNIT (
  ID NUMBER NOT NULL,
  CODE VARCHAR2(64 CHAR) NOT NULL,
  TITLE VARCHAR2(200 CHAR) NOT NULL,
  PARENT_ID NUMBER,
  ORG_KIND VARCHAR2(20 CHAR) NOT NULL,
  ACTIVE NUMBER(1) DEFAULT 1 NOT NULL,
  CONSTRAINT PK_AOU PRIMARY KEY (ID),
  CONSTRAINT UQ_AOU_CODE UNIQUE (CODE),
  CONSTRAINT FK_AOU_PARENT FOREIGN KEY (PARENT_ID) REFERENCES ACCESS_ORG_UNIT(ID),
  CONSTRAINT CK_AOU_KIND CHECK (ORG_KIND IN ('DIVISION','UNIT')),
  CONSTRAINT CK_AOU_ACTIVE CHECK (ACTIVE IN (0,1)),
  CONSTRAINT CK_AOU_SELF CHECK (PARENT_ID IS NULL OR PARENT_ID <> ID)
);
CREATE INDEX IX_AOU_PARENT ON ACCESS_ORG_UNIT(PARENT_ID);

-- 3.5. Належність користувача до організаційних одиниць.
-- APP_USER є існуючою таблицею: USER_ID і USERNAME треба звірити.
CREATE TABLE ACCESS_USER_ORG (
  USER_ID NUMBER NOT NULL,
  ORG_UNIT_ID NUMBER NOT NULL,
  MEMBERSHIP_KIND VARCHAR2(20 CHAR) DEFAULT 'MEMBER' NOT NULL,
  ACTIVE NUMBER(1) DEFAULT 1 NOT NULL,
  CONSTRAINT PK_AUO PRIMARY KEY (USER_ID,ORG_UNIT_ID,MEMBERSHIP_KIND),
  CONSTRAINT FK_AUO_ORG FOREIGN KEY (ORG_UNIT_ID) REFERENCES ACCESS_ORG_UNIT(ID),
  CONSTRAINT CK_AUO_KIND CHECK (MEMBERSHIP_KIND IN ('MEMBER','MANAGER','ADMIN')),
  CONSTRAINT CK_AUO_ACTIVE CHECK (ACTIVE IN (0,1))
);
CREATE INDEX IX_AUO_ORG ON ACCESS_USER_ORG(ORG_UNIT_ID,ACTIVE);

-- 3.6. Призначення: рівно одна область дії в одному рядку.
-- SCOPE_ID може посилатися на різні бізнес-таблиці, тому поліморфний FK
-- неможливий: існування ресурсу перевіряє API у транзакції.
CREATE TABLE ACCESS_ASSIGNMENT (
  ID NUMBER GENERATED BY DEFAULT ON NULL AS IDENTITY,
  USER_ID NUMBER NOT NULL,
  ROLE_ID NUMBER NOT NULL,
  SCHEME_TYPE_ID NUMBER,
  SCOPE_TYPE VARCHAR2(16 CHAR) DEFAULT 'GLOBAL' NOT NULL,
  SCOPE_ID NUMBER,
  ACTIVE NUMBER(1) DEFAULT 1 NOT NULL,
  VALID_FROM TIMESTAMP(6) DEFAULT SYSTIMESTAMP NOT NULL,
  VALID_TO TIMESTAMP(6),
  CREATED_AT TIMESTAMP(6) DEFAULT SYSTIMESTAMP NOT NULL,
  CREATED_BY NUMBER,
  REVOKED_AT TIMESTAMP(6),
  REVOKED_BY NUMBER,
  CONSTRAINT PK_AA PRIMARY KEY (ID),
  CONSTRAINT FK_AA_ROLE FOREIGN KEY (ROLE_ID) REFERENCES ACCESS_ROLE(ID),
  CONSTRAINT FK_AA_TYPE FOREIGN KEY (SCHEME_TYPE_ID) REFERENCES ACCESS_SCHEME_TYPE(ID),
  CONSTRAINT CK_AA_ACTIVE CHECK (ACTIVE IN (0,1)),
  CONSTRAINT CK_AA_SCOPE CHECK (SCOPE_TYPE IN ('GLOBAL','DIVISION','UNIT','ALBUM','SCHEME')),
  CONSTRAINT CK_AA_SCOPE_ID CHECK (
    (SCOPE_TYPE = 'GLOBAL' AND SCOPE_ID IS NULL) OR
    (SCOPE_TYPE <> 'GLOBAL' AND SCOPE_ID > 0)
  ),
  CONSTRAINT CK_AA_VALID CHECK (VALID_TO IS NULL OR VALID_TO > VALID_FROM),
  CONSTRAINT CK_AA_REVOKED CHECK (
    (ACTIVE = 1 AND REVOKED_AT IS NULL AND REVOKED_BY IS NULL) OR
    (ACTIVE = 0 AND REVOKED_AT IS NOT NULL)
  )
);
CREATE INDEX IX_AA_USER_ACTIVE ON ACCESS_ASSIGNMENT(USER_ID,ACTIVE,VALID_TO);
CREATE INDEX IX_AA_SCOPE ON ACCESS_ASSIGNMENT(SCOPE_TYPE,SCOPE_ID,ACTIVE);
CREATE INDEX IX_AA_ROLE ON ACCESS_ASSIGNMENT(ROLE_ID,ACTIVE);
CREATE INDEX IX_AA_TYPE ON ACCESS_ASSIGNMENT(SCHEME_TYPE_ID);

-- Унікальність АКТИВНИХ призначень; відкликані записи залишаються в історії.
-- NULL для SCHEME_TYPE_ID представляємо як -1 (ID довідника > 0).
CREATE UNIQUE INDEX UQ_AA_ACTIVE ON ACCESS_ASSIGNMENT (
  CASE WHEN ACTIVE = 1 THEN USER_ID END,
  CASE WHEN ACTIVE = 1 THEN ROLE_ID END,
  CASE WHEN ACTIVE = 1 THEN NVL(SCHEME_TYPE_ID,-1) END,
  CASE WHEN ACTIVE = 1 THEN SCOPE_TYPE END,
  CASE WHEN ACTIVE = 1 THEN NVL(SCOPE_ID,-1) END
);
ALTER TABLE ACCESS_SCHEME_TYPE ADD CONSTRAINT CK_AST_ID CHECK (ID > 0);

-- 3.7. Аудит адміністративних дій.
CREATE TABLE ACCESS_AUDIT (

  ACTOR_USER_ID NUMBER NOT NULL,

  CONSTRAINT CK_AUD_ACTION CHECK (ACTION IN ('GRANT','REVOKE','UPDATE','BOOTSTRAP')),
  CONSTRAINT CK_AUD_JSON CHECK (DETAILS_JSON IS JSON)

CREATE INDEX IX_AUD_TARGET ON ACCESS_AUDIT(TARGET_USER_ID,CREATED_AT);
CREATE INDEX IX_AUD_ACTOR ON ACCESS_AUDIT(ACTOR_USER_ID,CREATED_AT);

6. SQL: ПЕРЕВІРКА ДОЗВОЛУ НА КОНКРЕТНУ СХЕМУ

-- Вхідні значення мають бути ЗАВАНТАЖЕНІ СЕРВЕРОМ із БД:
-- :scheme_id, :scheme_type_id, :album_id, :division_id, :unit_id.
-- :permission_code — серверний whitelist, не довіряти довільному вводу.
-- Перевірка DIVISION/UNIT тут припускає, що власність схеми збережена
-- однозначно. Якщо використовується ієрархічний org tree, DIVISION_ID
-- отримуйте на сервері через ancestry, а не з клієнта.
SELECT CASE WHEN EXISTS (
  SELECT 1
  FROM ACCESS_ASSIGNMENT a
  JOIN ACCESS_ROLE r ON r.ID = a.ROLE_ID AND r.ACTIVE = 1
  JOIN ACCESS_ROLE_PERMISSION rp ON rp.ROLE_ID = r.ID
  JOIN ACCESS_PERMISSION p ON p.ID = rp.PERMISSION_ID
  WHERE a.USER_ID = :user_id
    AND a.ACTIVE = 1
    AND a.REVOKED_AT IS NULL
    AND a.VALID_FROM <= SYSTIMESTAMP
    AND (a.VALID_TO IS NULL OR a.VALID_TO > SYSTIMESTAMP)
    AND p.CODE = :permission_code
    AND (a.SCHEME_TYPE_ID IS NULL OR a.SCHEME_TYPE_ID = :scheme_type_id)
    AND (
      a.SCOPE_TYPE = 'GLOBAL'
      OR (a.SCOPE_TYPE = 'DIVISION' AND a.SCOPE_ID = :division_id)
      OR (a.SCOPE_TYPE = 'UNIT' AND a.SCOPE_ID = :unit_id)
      OR (a.SCOPE_TYPE = 'ALBUM' AND a.SCOPE_ID = :album_id)
      OR (a.SCOPE_TYPE = 'SCHEME' AND a.SCOPE_ID = :scheme_id)
    )
) THEN 1 ELSE 0 END AS ALLOWED FROM DUAL;

-- Зауваження: це запит для існуючої схеми, а не для створення нової.
-- Для scheme.create потрібна перевірка target album/unit і узгодженості
-- типу; для схеми, що має декілька власників, окрема бізнес-політика.

7. ПЕРЕВІРКА КОНСИСТЕНТНОСТІ / ДІАГНОСТИКА

-- Активні призначення
SELECT a.ID,a.USER_ID,r.CODE ROLE_CODE,t.CODE SCHEME_TYPE,
       a.SCOPE_TYPE,a.SCOPE_ID,a.VALID_FROM,a.VALID_TO
FROM ACCESS_ASSIGNMENT a
JOIN ACCESS_ROLE r ON r.ID=a.ROLE_ID
LEFT JOIN ACCESS_SCHEME_TYPE t ON t.ID=a.SCHEME_TYPE_ID
WHERE a.ACTIVE=1 ORDER BY a.USER_ID,a.ID;

-- Права ролей
SELECT r.CODE ROLE_CODE,p.CODE PERMISSION_CODE
FROM ACCESS_ROLE_PERMISSION rp
JOIN ACCESS_ROLE r ON r.ID=rp.ROLE_ID
JOIN ACCESS_PERMISSION p ON p.ID=rp.PERMISSION_ID
ORDER BY r.CODE,p.CODE;

-- Сумнівні призначення: ресурс SCOPE_ID не можна валідувати одним FK.
-- API повинен перевіряти існування та доступність ресурсу на INSERT,
-- UPDATE і під час виконання авторизації. Планова перевірка orphan scopes
-- виконується JOIN з реальними SCHEME / ALBUM / ORG таблицями.

8. МІГРАЦІЯ ІЗ ПОПЕРЕДНЬОЇ СХЕМИ — НЕ ВИКОНУВАТИ АВТОМАТИЧНО

1) Зробіть бекап ACCESS_* та інвентаризацію існуючих таблиць.
2) З'ясуйте реальні назви: користувачі, альбоми, схеми, типи схем,
   підрозділи, unit; звірте PK, типи та схему-власника Oracle.
3) Перевірте всі старі ACCESS_ASSIGNMENT:
   SELECT ID,USER_ID,ROLE_ID,SCHEME_TYPE,UNIT_ID,WORKSHOP_ID,
          DEPARTMENT_ID,ALBUM_ID,SCHEME_ID FROM ACCESS_ASSIGNMENT;
4) Перетворіть старі колонки на ОДИН SCOPE_TYPE/SCOPE_ID. Якщо заповнено
   декілька колонок — не конвертуйте автоматично: це перетин умов,
   який не можна безпечно замінити одним scope. Потрібне ручне рішення
   або розширення моделі на composite scopes.
5) Відобразіть старий SCHEME_TYPE на ACCESS_SCHEME_TYPE.ID. Перевірте,
   чи 'premises' означає будівлі, а не окремий тип.
6) Впровадьте нові таблиці спочатку як ACCESS_*_V2 або в тестовій схемі;
   перенесіть дані, звірте кількість та семантику призначень.
7) Переробіть API admin roles: замінити SCHEME_TYPE,UNIT_ID,
   WORKSHOP_ID,DEPARTMENT_ID,ALBUM_ID,SCHEME_ID на
   SCHEME_TYPE_ID,SCOPE_TYPE,SCOPE_ID; адаптувати валідацію scope.
8) Оновіть фронт MUI (вибір одного scope type + scope target),
   can()/useCan і server authorization. Перевіряйте дію кожного
   призначення незалежно, не об'єднуйте глобальний reader із editor.
9) Перевірте перехід за міжсхемними посиланнями; на кожному переході
   окремо перевіряйте scheme.read цільової схеми.
10) Після тестування переключіть API і тільки потім прибирайте стару схему.

9. ПЕРЕВІРКИ БЕЗПЕКИ, ЯКІ SQL НЕ МОЖЕ ЗАМІНИТИ

- user_id для дій беріть з автентифікованої сесії, не з request body.
- Адміністратор підрозділу може видавати лише дозволи в межах власного
  scope і тільки ролі, які йому дозволено делегувати. Забороніть
  самопризначення super_administrator та ескалацію через GLOBAL.
- Додавання прав, відкликання та ACCESS_AUDIT робіть в одній транзакції.
- Для схеми: перевіряйте scheme.read / scheme.update сервером при кожному
  GET/PUT/DELETE; mode=editor у URL не є правом.
- Перевіряйте права для експорту, вкладень, search і API посилань.
- Не використовуйте CREATED_BY як критерій права редагування.
- Забезпечте консистентність DIVISION/UNIT і відсутність циклів дерева
  організації в API або тригерах; CHECK не контролює цикли.
- Для SCOPE_TYPE='DIVISION'/'UNIT' перевіряйте ORG_KIND цільового ID.
- Для ALBUM/SCHEME перевіряйте існування та належність при видачі прав.
- Якщо потрібно приховувати деякі схеми від глобального reader, додайте
  класифікацію/політику винятків окремо; поточна модель — allow-only.
- Вирішіть, чи глобальний super_administrator повинен мати доступ до
  всього контенту: в цьому seed він має всі права, але scope призначення
  все одно обмежує його, якщо не GLOBAL.

10. ПІДСУМОК

Нова схема підтримує:
  A) reader тільки equipment, тільки premises або обидва;
  B) editor свого DIVISION/UNIT без обмеження авторством;
  C) reader GLOBAL + editor DIVISION/UNIT;
  D) глобальну навігацію за зв'язками між схемами;
  E) editor конкретного альбому чи схеми;
  F) тимчасові ролі через VALID_FROM/VALID_TO;
  G) адміністратора підрозділу без автоматичних прав редагування.

УВАГА: це цільова v2, а НЕ drop-in заміна попереднього admin API.
Наявний API очікує старі колонки ACCESS_ASSIGNMENT; його потрібно
адаптувати за кроками міграції. Вказані APP_USER / SCHEME / ALBUM / ORG
не перевірені проти реальної структури вашої БД — FK до них свідомо
не додано, щоб не вигадувати схему ваших таблиць.
