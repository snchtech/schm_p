ГОТОВИЙ ПАКЕТ: АДМІНІСТРУВАННЯ РОЛЕЙ
============================================================================
Next.js App Router (JavaScript) + MUI + Node.js Express API + Oracle 21
Версія 1.0. Сумісний із попередньою моделлю ACCESS_ROLE / ACCESS_ASSIGNMENT.

МОЖЛИВОСТІ
- Пошук користувача за user_id або username (точний user_id, prefix username).
- Перегляд поточних призначень із областю дії (scope).
- Додавання і відкликання окремих призначень ролей.
- Ролі reader, editor, administrator, super_administrator.
- Область дії: unitId, workshopId, departmentId, albumId, schemeId, schemeType.
- Адміністратор керує лише підлеглим unitId, лише reader/editor.
- Super administrator керує усіма користувачами і ролями.
- Транзакційне створення/відкликання, аудит, серверна авторизація.

ВАЖЛИВО ПЕРЕД ЗАПУСКОМ
1. Приклади вважають, що автентифікація вже є. Обов'язково підключіть
   ваш реальний session middleware: req.auth = {userId, username}.
   НЕ приймайте userId/role з HTTP headers, query або body як особу актора.
2. APP_USER(USER_ID, USERNAME, UNIT_ID, ACTIVE) — АДАПТЕР до вашої таблиці
   користувачів. Замініть лише SQL в userRepository.js, якщо структура інша.
3. У цьому пакеті адміністратор має ОДИН підрозділ unitId з APP_USER.UNIT_ID.
   Для багатьох підрозділів додайте таблицю admin memberships та перевірку.
4. Всі присвоєння administrator/super_administrator робить тільки super_admin.
   Видача super_administrator через UI вимкнена; bootstrap вручну.
5. Довільні NULL у scope = глобальний доступ. Тому звичайний admin не може
   створити призначення без unitId. Змішані області різних unit заборонені.
6. База прав для reader/editor залишається з попереднього пакета. Цей пакет
   створює ACCESS_ASSIGNMENT, а не підміняє permission engine.

CREATE TABLE ACCESS_ROLE (
 ID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
 CODE VARCHAR2(64) NOT NULL UNIQUE,
 TITLE VARCHAR2(200) NOT NULL
);
CREATE TABLE ACCESS_ASSIGNMENT (
 ID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
 USER_ID NUMBER NOT NULL,
 ROLE_ID NUMBER NOT NULL REFERENCES ACCESS_ROLE(ID),
 SCHEME_TYPE VARCHAR2(64),
 UNIT_ID NUMBER,
 WORKSHOP_ID NUMBER,
 DEPARTMENT_ID NUMBER,
 ALBUM_ID NUMBER,
 SCHEME_ID NUMBER,
 ACTIVE NUMBER(1) DEFAULT 1 NOT NULL CHECK (ACTIVE IN (0,1)),
 CREATED_AT TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
 CREATED_BY NUMBER
);
CREATE INDEX IX_ACCESS_ASSIGNMENT_USER ON ACCESS_ASSIGNMENT(USER_ID,ACTIVE);
CREATE TABLE ACCESS_AUDIT (
 ID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
 ACTOR_USER_ID NUMBER NOT NULL,
 TARGET_USER_ID NUMBER NOT NULL,
 ACTION VARCHAR2(32) NOT NULL,
 ASSIGNMENT_ID NUMBER,
 ROLE_CODE VARCHAR2(64),
 DETAILS_JSON CLOB CHECK (DETAILS_JSON IS JSON),
 CREATED_AT TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL
);
MERGE INTO ACCESS_ROLE r USING
 (SELECT 'reader' code,'Перегляд' title FROM dual UNION ALL
  SELECT 'editor','Редагування' FROM dual UNION ALL
  SELECT 'administrator','Адміністратор' FROM dual UNION ALL

 ON (r.CODE=s.CODE)
 WHEN NOT MATCHED THEN INSERT(CODE,TITLE) VALUES(s.CODE,s.TITLE);

// ПІДКЛЮЧИТИ ПІСЛЯ реального middleware сесії/токена.
// Приклад інтеграції: app.use(existingSessionAuth); app.use('/api/admin/access', ...)
export function requireAuth(req,res,next) {
  const id = Number(req.auth?.userId);
  if (!Number.isSafeInteger(id) || id <= 0) {
    return res.status(401).json({error:'UNAUTHENTICATED'});
  }
  req.auth.userId = id;
  next();
}
// НЕ використовуйте req.headers['x-user-id'] як автентифікацію.

// Замініть APP_USER та назви колонок на ваші фактичні.
export async function findUsers(conn, search) {
  const q = String(search || '').trim();
  if (!q) return [];
  let sql, binds;
  if (/^[0-9]+$/.test(q)) {
    sql = `SELECT USER_ID AS "userId", USERNAME AS "username",
      UNIT_ID AS "unitId", ACTIVE AS "active"
      FROM APP_USER WHERE USER_ID=:id AND ACTIVE=1`;
    binds = {id:Number(q)};
  } else {
    sql = `SELECT USER_ID AS "userId", USERNAME AS "username",
      UNIT_ID AS "unitId", ACTIVE AS "active"
      FROM APP_USER WHERE LOWER(USERNAME) LIKE :prefix AND ACTIVE=1
      ORDER BY USERNAME FETCH FIRST 30 ROWS ONLY`;
    // INSTR avoids SQL LIKE wildcard injection for '%' and '_'.
    sql = `SELECT USER_ID AS "userId", USERNAME AS "username",
      UNIT_ID AS "unitId", ACTIVE AS "active"
      FROM APP_USER WHERE INSTR(LOWER(USERNAME), :prefix)=1 AND ACTIVE=1
      ORDER BY USERNAME FETCH FIRST 30 ROWS ONLY`;
    binds = {prefix:q.toLowerCase()};
  }
  return (await conn.execute(sql,binds,{outFormat:4002})).rows;
}
export async function findUserById(conn,id) {
  const r=await conn.execute(`SELECT USER_ID AS "userId", USERNAME AS "username",
    UNIT_ID AS "unitId", ACTIVE AS "active" FROM APP_USER
    WHERE USER_ID=:id`,{id},{outFormat:4002});
  return r.rows[0] || null;
}

import {oracledb} from '../db/oracle.js';
export async function rolesForUser(conn,userId) {
 const r=await conn.execute(`SELECT r.CODE AS "code", a.UNIT_ID AS "unitId"
   FROM ACCESS_ASSIGNMENT a JOIN ACCESS_ROLE r ON r.ID=a.ROLE_ID
   WHERE a.USER_ID=:userId AND a.ACTIVE=1`,{userId},
   {outFormat:oracledb.OUT_FORMAT_OBJECT});
 return r.rows;
}
export async function assignmentsForUser(conn,userId) {
 const r=await conn.execute(`SELECT a.ID AS "id", r.CODE AS "role",
  a.SCHEME_TYPE AS "schemeType", a.UNIT_ID AS "unitId",
  a.WORKSHOP_ID AS "workshopId", a.DEPARTMENT_ID AS "departmentId",
  a.ALBUM_ID AS "albumId", a.SCHEME_ID AS "schemeId",
  a.CREATED_AT AS "createdAt"
  FROM ACCESS_ASSIGNMENT a JOIN ACCESS_ROLE r ON r.ID=a.ROLE_ID
  WHERE a.USER_ID=:userId AND a.ACTIVE=1 ORDER BY a.ID DESC`,
  {userId},{outFormat:oracledb.OUT_FORMAT_OBJECT});
 return r.rows;
}
export async function getRole(conn,code) {
 const r=await conn.execute('SELECT ID AS "id", CODE AS "code" FROM ACCESS_ROLE WHERE CODE=:code',
  {code},{outFormat:oracledb.OUT_FORMAT_OBJECT});
 return r.rows[0]||null;
}
export async function getAssignmentForUpdate(conn,id) {
 const r=await conn.execute(`SELECT a.ID AS "id", a.USER_ID AS "userId",
  r.CODE AS "role", a.UNIT_ID AS "unitId", a.ACTIVE AS "active"
  FROM ACCESS_ASSIGNMENT a JOIN ACCESS_ROLE r ON r.ID=a.ROLE_ID
  WHERE a.ID=:id FOR UPDATE`,{id},{outFormat:oracledb.OUT_FORMAT_OBJECT});
 return r.rows[0]||null;
}
export async function insertAssignment(conn,{userId,roleId,scope,actorId}) {
 const result=await conn.execute(`INSERT INTO ACCESS_ASSIGNMENT
  (USER_ID,ROLE_ID,SCHEME_TYPE,UNIT_ID,WORKSHOP_ID,DEPARTMENT_ID,
   ALBUM_ID,SCHEME_ID,ACTIVE,CREATED_BY)
  VALUES (:userId,:roleId,:schemeType,:unitId,:workshopId,:departmentId,
   :albumId,:schemeId,1,:actorId) RETURNING ID INTO :id`,
  {userId,roleId,actorId,
   schemeType:scope.schemeType,unitId:scope.unitId,workshopId:scope.workshopId,
   departmentId:scope.departmentId,albumId:scope.albumId,schemeId:scope.schemeId,
   id:{dir:oracledb.BIND_OUT,type:oracledb.NUMBER}},
  {autoCommit:false});
 return result.outBinds.id[0];
}
export async function audit(conn,{actorId,targetId,action,assignmentId,role,details}) {
 await conn.execute(`INSERT INTO ACCESS_AUDIT
  (ACTOR_USER_ID,TARGET_USER_ID,ACTION,ASSIGNMENT_ID,ROLE_CODE,DETAILS_JSON)
  VALUES(:actorId,:targetId,:action,:assignmentId,:role,:details)`,
  {actorId,targetId,action,assignmentId,role,
   details:JSON.stringify(details||{})},{autoCommit:false});
}

import {connection,oracledb} from '../db/oracle.js';
import {findUsers,findUserById} from '../repositories/userRepository.js';
import {rolesForUser,assignmentsForUser,getRole,getAssignmentForUpdate,
  insertAssignment,audit} from '../repositories/accessRepository.js';

const ROLES=['reader','editor','administrator','super_administrator'];
const SCHEME_TYPES=['equipment','premises']; // підставте власні коди
const fail=(status,code)=>{const e=new Error(code);e.status=status;return e;};
const positiveOrNull=v=>{
 if(v===''||v===null||v===undefined)return null;
 const n=Number(v);
 if(!Number.isSafeInteger(n)||n<=0)throw fail(400,'INVALID_SCOPE_ID');
 return n;
};
const parseScope=raw=>{
 const s=raw||{};
 const keys=['unitId','workshopId','departmentId','albumId','schemeId','schemeType'];
 if(Object.keys(s).some(k=>!keys.includes(k)))throw fail(400,'UNKNOWN_SCOPE_FIELD');
 const scope={};
 for(const k of keys.filter(k=>k!=='schemeType'))scope[k]=positiveOrNull(s[k]);
 scope.schemeType=s.schemeType||null;
 if(scope.schemeType&&!SCHEME_TYPES.includes(scope.schemeType))throw fail(400,'INVALID_SCHEME_TYPE');
 return scope;
};
const isSuper=roles=>roles.some(x=>x.code==='super_administrator');
const isAdmin=roles=>roles.some(x=>x.code==='administrator');

// ВАЖЛИВО: звичайний admin має власний unitId у APP_USER.
// Не дозволяємо admin передавати довільний workshopId/albumId/schemeId,
// доки не реалізовано серверну перевірку їх належності його unitId.
function assertActor(actor,roles,target,role,scope,operation) {
 if(!isSuper(roles)&&!isAdmin(roles))throw fail(403,'ADMIN_REQUIRED');
 if(isSuper(roles))return;
 if(!actor.unitId||Number(actor.unitId)!==Number(target.unitId))
   throw fail(403,'TARGET_OUTSIDE_UNIT');
 if(!['reader','editor'].includes(role))throw fail(403,'ROLE_NOT_DELEGABLE');
 if(scope.unitId!==Number(actor.unitId))throw fail(403,'SCOPE_OUTSIDE_UNIT');
 if(['workshopId','departmentId','albumId','schemeId'].some(k=>scope[k]!=null))
   throw fail(403,'NARROW_SCOPE_REQUIRES_HIERARCHY_VALIDATION');
 if(Number(actor.userId)===Number(target.userId))throw fail(403,'SELF_CHANGE_FORBIDDEN');
}
async function getContext(conn,actorId) {
 const actor=await findUserById(conn,actorId);
 if(!actor||Number(actor.active)!==1)throw fail(403,'ACTOR_INACTIVE');
 const roles=await rolesForUser(conn,actorId);
 if(!isSuper(roles)&&!isAdmin(roles))throw fail(403,'ADMIN_REQUIRED');
 return {actor,roles};
}
async function withConnection(fn) {
 const conn=await connection();
 try{return await fn(conn);}finally{await conn.close();}
}
export async function getMe(actorId) {
 return withConnection(async conn=>{
  const {actor,roles}=await getContext(conn,actorId);
  return {userId:actor.userId,unitId:actor.unitId,
   superAdmin:isSuper(roles),roles:roles.map(r=>r.code)};
 });
}
export async function searchUsers(actorId,q) {
 return withConnection(async conn=>{
  const {actor,roles}=await getContext(conn,actorId);
  const users=await findUsers(conn,q);
  return isSuper(roles)?users:users.filter(u=>Number(u.unitId)===Number(actor.unitId));
 });
}
export async function getAssignments(actorId,targetId) {
 return withConnection(async conn=>{
  const {actor,roles}=await getContext(conn,actorId);
  const target=await findUserById(conn,targetId);
  if(!target||Number(target.active)!==1)throw fail(404,'USER_NOT_FOUND');
  if(!isSuper(roles)&&Number(target.unitId)!==Number(actor.unitId))throw fail(403,'TARGET_OUTSIDE_UNIT');
  const list=await assignmentsForUser(conn,targetId);
  return isSuper(roles)?list:list.filter(a=>['reader','editor'].includes(a.role)
    && Number(a.unitId)===Number(actor.unitId));
 });
}
export async function grant(actorId,targetId,input) {
 return withConnection(async conn=>{
  try {
   const {actor,roles}=await getContext(conn,actorId);
   const target=await findUserById(conn,targetId);
   if(!target||Number(target.active)!==1)throw fail(404,'USER_NOT_FOUND');
   const role=String(input?.role||'');
   if(!ROLES.includes(role))throw fail(400,'INVALID_ROLE');
   const scope=parseScope(input?.scope);
   if(role==='super_administrator')throw fail(403,'SUPER_ADMIN_BOOTSTRAP_ONLY');
   // Для адміністратора дозволено лише глобальний scope і тільки super_admin.
   if(role==='administrator'&&Object.values(scope).some(v=>v!=null))
     throw fail(400,'ADMIN_ROLE_MUST_BE_GLOBAL');
   assertActor(actor,roles,target,role,scope,'grant');
   if(Number(target.userId)===Number(actor.userId))throw fail(403,'SELF_CHANGE_FORBIDDEN');
   const roleRow=await getRole(conn,role);
   if(!roleRow)throw fail(400,'ROLE_NOT_CONFIGURED');
   // Серіалізуємо зміни призначень для одного користувача.
   await conn.execute('SELECT USER_ID FROM APP_USER WHERE USER_ID=:id FOR UPDATE',
     {id:targetId});
   const list=await assignmentsForUser(conn,targetId);
   const duplicate=list.some(a=>a.role===role&&
    ['unitId','workshopId','departmentId','albumId','schemeId','schemeType']
     .every(k=>(a[k]??null)===(scope[k]??null)));
   if(duplicate)throw fail(409,'ASSIGNMENT_ALREADY_EXISTS');
   const id=await insertAssignment(conn,{userId:targetId,roleId:roleRow.id,scope,actorId});
   await audit(conn,{actorId,targetId,action:'GRANT',assignmentId:id,role,details:scope});
   await conn.commit();return {id};
  } catch(e){await conn.rollback();throw e;}
 });
}
export async function revoke(actorId,assignmentId) {
 return withConnection(async conn=>{
  try {
   const {actor,roles}=await getContext(conn,actorId);
   const assignment=await getAssignmentForUpdate(conn,assignmentId);
   if(!assignment||Number(assignment.active)!==1)throw fail(404,'ASSIGNMENT_NOT_FOUND');
   const target=await findUserById(conn,assignment.userId);
   if(!target)throw fail(404,'USER_NOT_FOUND');
   assertActor(actor,roles,target,assignment.role,
     {unitId:assignment.unitId},'revoke');
   if(Number(target.userId)===Number(actor.userId))throw fail(403,'SELF_CHANGE_FORBIDDEN');
   if(assignment.role==='super_administrator')
     throw fail(403,'SUPER_ADMIN_BOOTSTRAP_ONLY');
   await conn.execute('UPDATE ACCESS_ASSIGNMENT SET ACTIVE=0 WHERE ID=:id',
     {id:assignmentId},{autoCommit:false});
   await audit(conn,{actorId,targetId:assignment.userId,action:'REVOKE',
     assignmentId,role:assignment.role,details:{}});
   await conn.commit();return {ok:true};
  }catch(e){await conn.rollback();throw e;}
 });
}

 const showError=e=>{setError(e.message||'Помилка');setSuccess('');};
 useEffect(()=>{accessApi.me().then(setMe).catch(showError).finally(()=>setLoading(false));},[]);
 const find=async()=>{
  setBusy(true);setError('');setSelected(null);setAssignments([]);
  try{setUsers(await accessApi.search(search.trim()));}catch(e){showError(e);}
  finally{setBusy(false);}
 };
 const choose=async user=>{
  setSelected(user);setError('');setSuccess('');
  try{setAssignments(await accessApi.list(user.userId));}catch(e){showError(e);}
  setScope({...initialScope,unitId:me?.superAdmin?'':String(me?.unitId||'')});
 };
 const refresh=async()=>{if(selected)setAssignments(await accessApi.list(selected.userId));};
 const save=async()=>{
  if(!selected)return;
  setBusy(true);setError('');setSuccess('');
  try{
   const payload={role,scope:Object.fromEntries(Object.entries(scope).map(
    ([k,v])=>[k,k==='schemeType'?(v||null):idOrNull(v)]))};
   await accessApi.grant(selected.userId,payload);
   await refresh();setSuccess('Роль призначено');
  }catch(e){showError(e);}finally{setBusy(false);}
 };
 const remove=async a=>{
  if(!window.confirm(`Відкликати ${a.role} (ID ${a.id})?`))return;
  setBusy(true);setError('');setSuccess('');
  try{await accessApi.revoke(a.id);await refresh();setSuccess('Призначення відкликано');}
  catch(e){showError(e);}finally{setBusy(false);}
 };
 if(loading)return <Box sx={{p:4}}><CircularProgress/></Box>;
 if(!me)return <Alert severity="error">Немає доступу до адміністрування.</Alert>;
 const allowedRoles=me.superAdmin?roleOptions:roleOptions.filter(r=>
  ['reader','editor'].includes(r.value));
 const canEditScope=me.superAdmin;
 return <Box sx={{maxWidth:1200,mx:'auto',p:{xs:2,md:4}}>
  <Typography variant="h4" sx={{mb:1}}>Адміністрування доступів</Typography>
  <Typography color="text.secondary" sx={{mb:3}}>
   {me.superAdmin?'Суперадміністратор: усі користувачі':'Адміністратор підрозділу: '+me.unitId}
  </Typography>
  {error&&<Alert severity="error" sx={{mb:2}} onClose={()=>setError('')}>{error}</Alert>}
  {success&&<Alert severity="success" sx={{mb:2}}>{success}</Alert>}
  <Paper sx={{p:2,mb:3}}>
   <Typography variant="h6" sx={{mb:2}}>1. Знайти користувача</Typography>
   <Stack direction={{xs:'column',sm:'row'}} spacing={2}>
    <TextField fullWidth label="user_id або username" value={search}
     onChange={e=>setSearch(e.target.value)}
     onKeyDown={e=>{if(e.key==='Enter')find();}}/>
    <Button variant="contained" disabled={busy||search.trim().length<2} onClick={find}>
     Знайти</Button>
   </Stack>
   <Table size="small" sx={{mt:2}}><TableHead><TableRow>
    <TableCell>User ID</TableCell><TableCell>Username</TableCell>
    <TableCell>Підрозділ</TableCell><TableCell/>
   </TableRow></TableHead><TableBody>
    {users.map(u=><TableRow key={u.userId} selected={selected?.userId===u.userId}>
     <TableCell>{u.userId}</TableCell><TableCell>{u.username}</TableCell>
     <TableCell>{u.unitId??'—'}</TableCell>
     <TableCell><Button onClick={()=>choose(u)}>Вибрати</Button></TableCell>
    </TableRow>)}
   </TableBody></Table>
  </Paper>
  {selected&&<>
   <Paper sx={{p:2,mb:3}}>
    <Typography variant="h6">2. Поточні призначення: {selected.username} (#{selected.userId})</Typography>
    <Table size="small" sx={{mt:2}}><TableHead><TableRow>
     <TableCell>ID</TableCell><TableCell>Роль</TableCell><TableCell>Область</TableCell><TableCell/>
    </TableRow></TableHead><TableBody>
     {assignments.map(a=><TableRow key={a.id}>
      <TableCell>{a.id}</TableCell><TableCell><Chip size="small" label={a.role}/></TableCell>
      <TableCell>{['unitId','workshopId','departmentId','albumId','schemeId','schemeType']
       .filter(k=>a[k]!=null).map(k=>`${k}: ${a[k]}`).join('; ')||'Глобально'}</TableCell>
      <TableCell><Button color="error" disabled={busy||a.role==='super_administrator'}
       onClick={()=>remove(a)}>Відкликати</Button></TableCell>
     </TableRow>)}
     {!assignments.length&&<TableRow><TableCell colSpan={4}>Призначень немає</TableCell></TableRow>}
    </TableBody></Table>
   </Paper>
   <Paper sx={{p:2}}>
    <Typography variant="h6" sx={{mb:2}}>3. Видати роль</Typography>
    <Stack spacing={2}>
     <FormControl fullWidth><InputLabel id="role-label">Роль</InputLabel>
      <Select labelId="role-label" value={role} label="Роль" onChange={e=>{
       setRole(e.target.value);if(e.target.value==='administrator')setScope(initialScope);
      }}>
       {allowedRoles.map(r=><MenuItem key={r.value} value={r.value}>{r.label}</MenuItem>)}
      </Select></FormControl>
     <Divider/>
     <Typography variant="subtitle2">Область дії (порожнє поле = без обмеження)</Typography>
     <Stack direction={{xs:'column',md:'row'}} spacing={2} flexWrap="wrap">
      {['unitId','workshopId','departmentId','albumId','schemeId'].map(k=><TextField
       key={k} label={k} type="number" value={scope[k]}
       disabled={!canEditScope||role==='administrator'}
       onChange={e=>setScope(prev=>({...prev,[k]:e.target.value}))}
       inputProps={{min:1}} sx={{minWidth:160}}/>)}
      <FormControl sx={{minWidth:180}} disabled={role==='administrator'}>
       <InputLabel id="type-label">Тип схеми</InputLabel>
       <Select labelId="type-label" label="Тип схеми" value={scope.schemeType}
        onChange={e=>setScope(prev=>({...prev,schemeType:e.target.value}))}>
        <MenuItem value="">Усі</MenuItem><MenuItem value="equipment">Обладнання</MenuItem>
        <MenuItem value="premises">Приміщення</MenuItem>
       </Select></FormControl>
     </Stack>
     {!me.superAdmin&&<Alert severity="info">Доступ обмежено вашим підрозділом.
      Для складніших областей зверніться до суперадміністратора.</Alert>}
     <Button variant="contained" onClick={save} disabled={busy}>
      {busy?'Збереження...':'Видати роль'}</Button>
    </Stack>
   </Paper>
  </>}
 </Box>;
}

Після login або зміни ролей отримуйте assignments для поточного користувача
через ОКРЕМИЙ endpoint /api/access/me (не адміністративний).
Не викликайте /api/admin/access/users/:id/assignments для звичайного reader.

Перетворення ACCESS_ASSIGNMENT -> grants для попереднього engine:
function assignmentToGrant(a) {
 return {
   role:a.role,
   scope:{
     ...(a.unitId!=null?{unitId:String(a.unitId)}:{}),
     ...(a.workshopId!=null?{workshopId:String(a.workshopId)}:{}),
     ...(a.departmentId!=null?{departmentId:String(a.departmentId)}:{}),
     ...(a.albumId!=null?{albumId:String(a.albumId)}:{}),
     ...(a.schemeId!=null?{schemeId:String(a.schemeId)}:{})
   },
   schemeTypes:a.schemeType?[a.schemeType]:[]
 };
}

Після grant/revoke: інвалідовуйте кеш прав користувача на сервері.
Для вже відкритої сесії: перевірка версії прав або refetch при 403.
Не вважайте оновлення React/Zustand достатнім для серверної безпеки.

ПЕРЕВІРКИ:
- Reader/editor: GET /api/admin/access/me -> 403.
- Administrator unit 10: бачить лише user unit 10; unit 20 -> 403.
- Administrator unit 10: не може видати administrator або super_admin.
- Administrator unit 10: не може видати unitId=NULL або unitId=20.
- Administrator unit 10: не може видати workshopId, поки немає hierarchy check.
- Super admin: може видати reader/editor з різними scope.
- Super admin: може видати administrator з глобальним scope.
- Жоден користувач не може змінити власні призначення через UI/API.
- Дублікат активного призначення -> 409.
- Revoke не видаляє запис, а переводить ACTIVE=0 та пише AUDIT.
- Прямий POST без cookie/token -> 401.
- Підміна req.body.userId не змінює actorId.
- Після відкликання editor API збереження схеми -> 403.
