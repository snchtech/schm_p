-- ORACLE DATABASE 21c | SCHEME EDITOR | INSTALL SCRIPT v1.0
-- Next.js + Konva: JSON snapshots, relational search projections, cross-scheme links
-- IMPORTANT: Execute in a TEST schema first, using SQLcl / SQL Developer (Run Script F5).
-- Existing tables are NOT dropped or altered.
-- PREREQUISITES:
--   SCHEME(ID), SCHEME_VERSION(ID, SCHEME_ID), ALBUM_VERSION(ID) already exist.
--   All referenced IDs are assumed VARCHAR2(36) (same byte/char semantics).
--   Verify types using USER_TAB_COLUMNS before execution; adjust FK columns if necessary.
--   Existing SCHEME_VERSION must already reference SCHEME via SCHEME_ID.
--   No commits are embedded; Oracle DDL implicitly commits.
--   Run this ONCE in the desired schema. No destructive rollback is included.
--   All identifiers below are unquoted uppercase Oracle identifiers.
--
-- Recommended: treat JSON as the source of truth. Rebuild projection tables in one
-- transaction with the document update in the application service.
-- NOTE: Oracle 21c supports JSON in CLOB with IS JSON constraint.
-- Avoid SecureFiles compression licensing assumptions.

-- 01. Versioned JSON snapshots
CREATE TABLE SCHEME_DOCUMENT (
  SCHEME_VERSION_ID VARCHAR2(36) NOT NULL,
  SCHEMA_VERSION NUMBER(6) DEFAULT 1 NOT NULL,
  DOCUMENT_JSON CLOB NOT NULL,
  CONTENT_SHA256 VARCHAR2(64) NOT NULL,
  DOCUMENT_BYTES NUMBER(15),
  REVISION_NO NUMBER(12) DEFAULT 1 NOT NULL,
  CREATED_AT TIMESTAMP(6) WITH TIME ZONE DEFAULT SYSTIMESTAMP NOT NULL,
  UPDATED_AT TIMESTAMP(6) WITH TIME ZONE DEFAULT SYSTIMESTAMP NOT NULL,
  CONSTRAINT PK_SCH_DOC PRIMARY KEY (SCHEME_VERSION_ID),
  CONSTRAINT FK_SCH_DOC_VER FOREIGN KEY (SCHEME_VERSION_ID)
    REFERENCES SCHEME_VERSION(ID),
  CONSTRAINT CK_SCH_DOC_JSON CHECK (DOCUMENT_JSON IS JSON STRICT),
  CONSTRAINT CK_SCH_DOC_SCHEMA CHECK (SCHEMA_VERSION >= 1),
  CONSTRAINT CK_SCH_DOC_REV CHECK (REVISION_NO >= 1),
  CONSTRAINT CK_SCH_DOC_HASH CHECK (LENGTH(CONTENT_SHA256) = 64)
) LOB (DOCUMENT_JSON) STORE AS SECUREFILE;
COMMENT ON TABLE SCHEME_DOCUMENT IS 'Full versioned Konva document JSON; canonical visual state';

-- 02. Version-specific searchable object projections
CREATE TABLE SCHEME_OBJECT (
  SCHEME_VERSION_ID VARCHAR2(36) NOT NULL,
  OBJECT_ID VARCHAR2(36) NOT NULL,
  OBJECT_TYPE VARCHAR2(40) NOT NULL,
  OBJECT_SUBTYPE VARCHAR2(80),
  LABEL VARCHAR2(500),
  EXTERNAL_OBJECT_ID VARCHAR2(128),
  UNIT_ID VARCHAR2(36),
  SEARCHABLE_TEXT VARCHAR2(2000),
  PARENT_OBJECT_ID VARCHAR2(36),
  IS_DELETED NUMBER(1) DEFAULT 0 NOT NULL,
  CONSTRAINT PK_SCH_OBJ PRIMARY KEY (SCHEME_VERSION_ID, OBJECT_ID),
  CONSTRAINT FK_SCH_OBJ_VER FOREIGN KEY (SCHEME_VERSION_ID)
    REFERENCES SCHEME_VERSION(ID),
  CONSTRAINT FK_SCH_OBJ_PARENT FOREIGN KEY (SCHEME_VERSION_ID, PARENT_OBJECT_ID)
    REFERENCES SCHEME_OBJECT(SCHEME_VERSION_ID, OBJECT_ID)
    DEFERRABLE INITIALLY DEFERRED,
  CONSTRAINT CK_SCH_OBJ_DEL CHECK (IS_DELETED IN (0,1)),
  CONSTRAINT CK_SCH_OBJ_SELF CHECK (PARENT_OBJECT_ID IS NULL OR PARENT_OBJECT_ID <> OBJECT_ID)
);
CREATE INDEX IX_SCH_OBJ_EXT ON SCHEME_OBJECT(EXTERNAL_OBJECT_ID, OBJECT_TYPE);
CREATE INDEX IX_SCH_OBJ_TYPE ON SCHEME_OBJECT(SCHEME_VERSION_ID, OBJECT_TYPE);
CREATE INDEX IX_SCH_OBJ_UNIT ON SCHEME_OBJECT(UNIT_ID, SCHEME_VERSION_ID);
CREATE INDEX IX_SCH_OBJ_PARENT ON SCHEME_OBJECT(SCHEME_VERSION_ID, PARENT_OBJECT_ID);

-- 03. Intra-scheme graph connections, including node-edge, branch-ZVT, annotations
CREATE TABLE SCHEME_OBJECT_LINK (
  LINK_ID VARCHAR2(36) NOT NULL,
  SCHEME_VERSION_ID VARCHAR2(36) NOT NULL,
  SOURCE_OBJECT_ID VARCHAR2(36) NOT NULL,
  TARGET_OBJECT_ID VARCHAR2(36) NOT NULL,
  LINK_TYPE VARCHAR2(40) NOT NULL,
  SOURCE_ANCHOR VARCHAR2(80),
  TARGET_ANCHOR VARCHAR2(80),
  SEGMENT_ID VARCHAR2(100),
  CONSTRAINT PK_SCH_OBJ_LINK PRIMARY KEY (LINK_ID),
  CONSTRAINT FK_SCH_LINK_SRC FOREIGN KEY (SCHEME_VERSION_ID, SOURCE_OBJECT_ID)
    REFERENCES SCHEME_OBJECT(SCHEME_VERSION_ID, OBJECT_ID)
    DEFERRABLE INITIALLY DEFERRED,
  CONSTRAINT FK_SCH_LINK_DST FOREIGN KEY (SCHEME_VERSION_ID, TARGET_OBJECT_ID)
    REFERENCES SCHEME_OBJECT(SCHEME_VERSION_ID, OBJECT_ID)
    DEFERRABLE INITIALLY DEFERRED
);
CREATE INDEX IX_SCH_LINK_SRC ON SCHEME_OBJECT_LINK(SCHEME_VERSION_ID, SOURCE_OBJECT_ID);
CREATE INDEX IX_SCH_LINK_DST ON SCHEME_OBJECT_LINK(SCHEME_VERSION_ID, TARGET_OBJECT_ID);
CREATE INDEX IX_SCH_LINK_KIND ON SCHEME_OBJECT_LINK(LINK_TYPE, SCHEME_VERSION_ID);

-- 04. Stable object identity independent of version
CREATE TABLE SCHEME_OBJECT_IDENTITY (
  SCHEME_ID VARCHAR2(36) NOT NULL,
  OBJECT_ID VARCHAR2(36) NOT NULL,
  FIRST_VERSION_ID VARCHAR2(36),
  CREATED_AT TIMESTAMP(6) WITH TIME ZONE DEFAULT SYSTIMESTAMP NOT NULL,
  CONSTRAINT PK_SCH_OBJ_IDENTITY PRIMARY KEY (SCHEME_ID, OBJECT_ID),
  CONSTRAINT FK_SCH_IDENT_SCHEME FOREIGN KEY (SCHEME_ID) REFERENCES SCHEME(ID),
  CONSTRAINT FK_SCH_IDENT_VER FOREIGN KEY (FIRST_VERSION_ID) REFERENCES SCHEME_VERSION(ID)
);
-- Application must validate FIRST_VERSION_ID belongs to SCHEME_ID.
-- Never recycle OBJECT_ID for a different logical object in the same scheme.

-- 05. Cross-scheme navigation and semantic relationships
CREATE TABLE SCHEME_CROSS_LINK (
  ID VARCHAR2(36) NOT NULL,
  SOURCE_SCHEME_ID VARCHAR2(36) NOT NULL,
  SOURCE_OBJECT_ID VARCHAR2(36) NOT NULL,
  TARGET_SCHEME_ID VARCHAR2(36) NOT NULL,
  TARGET_OBJECT_ID VARCHAR2(36),
  LINK_KIND VARCHAR2(40) NOT NULL,
  CAPTION VARCHAR2(500),
  IS_ACTIVE NUMBER(1) DEFAULT 1 NOT NULL,
  CREATED_AT TIMESTAMP(6) WITH TIME ZONE DEFAULT SYSTIMESTAMP NOT NULL,
  CONSTRAINT PK_SCH_CROSS_LINK PRIMARY KEY (ID),
  CONSTRAINT FK_SCH_XSRC FOREIGN KEY (SOURCE_SCHEME_ID, SOURCE_OBJECT_ID)
    REFERENCES SCHEME_OBJECT_IDENTITY(SCHEME_ID, OBJECT_ID),
  CONSTRAINT FK_SCH_XTARGET_SCHEME FOREIGN KEY (TARGET_SCHEME_ID)
    REFERENCES SCHEME(ID),
  CONSTRAINT FK_SCH_XTARGET_OBJECT FOREIGN KEY (TARGET_SCHEME_ID, TARGET_OBJECT_ID)
    REFERENCES SCHEME_OBJECT_IDENTITY(SCHEME_ID, OBJECT_ID),
  CONSTRAINT CK_SCH_XACTIVE CHECK (IS_ACTIVE IN (0,1))
);
CREATE INDEX IX_SCH_XSRC ON SCHEME_CROSS_LINK(SOURCE_SCHEME_ID, SOURCE_OBJECT_ID);
CREATE INDEX IX_SCH_XTARGET ON SCHEME_CROSS_LINK(TARGET_SCHEME_ID, TARGET_OBJECT_ID);
CREATE INDEX IX_SCH_XKIND ON SCHEME_CROSS_LINK(LINK_KIND, IS_ACTIVE);
-- The backend generates URLs, e.g. /albums/:albumId/schemes/:schemeId?objectId=:id
-- Do NOT store an absolute URL as a canonical relationship.

-- 06. Searchable business attributes ONLY (not Konva styling)
CREATE TABLE SCHEME_OBJECT_ATTRIBUTE (
  SCHEME_VERSION_ID VARCHAR2(36) NOT NULL,
  OBJECT_ID VARCHAR2(36) NOT NULL,
  ATTRIBUTE_CODE VARCHAR2(100) NOT NULL,
  VALUE_TEXT VARCHAR2(2000),
  VALUE_NUMBER NUMBER,
  VALUE_DATE TIMESTAMP(6),
  CONSTRAINT PK_SCH_OBJ_ATTR PRIMARY KEY
    (SCHEME_VERSION_ID, OBJECT_ID, ATTRIBUTE_CODE),
  CONSTRAINT FK_SCH_ATTR_OBJ FOREIGN KEY (SCHEME_VERSION_ID, OBJECT_ID)
    REFERENCES SCHEME_OBJECT(SCHEME_VERSION_ID, OBJECT_ID),
  CONSTRAINT CK_SCH_ATTR_ONE CHECK (
    (CASE WHEN VALUE_TEXT IS NOT NULL THEN 1 ELSE 0 END) +
    (CASE WHEN VALUE_NUMBER IS NOT NULL THEN 1 ELSE 0 END) +
    (CASE WHEN VALUE_DATE IS NOT NULL THEN 1 ELSE 0 END) <= 1
  )
);
CREATE INDEX IX_SCH_ATTR_TEXT ON SCHEME_OBJECT_ATTRIBUTE(ATTRIBUTE_CODE, VALUE_TEXT);
CREATE INDEX IX_SCH_ATTR_NUM ON SCHEME_OBJECT_ATTRIBUTE(ATTRIBUTE_CODE, VALUE_NUMBER);

-- 07. File metadata (physical bytes remain in filesystem / object storage)
CREATE TABLE FILE_ASSET (
  ID VARCHAR2(36) NOT NULL,
  ORIGINAL_NAME VARCHAR2(500) NOT NULL,
  MIME_TYPE VARCHAR2(150),
  BYTE_SIZE NUMBER(15),
  STORAGE_PROVIDER VARCHAR2(30) NOT NULL,
  STORAGE_KEY VARCHAR2(1000) NOT NULL,
  SHA256 VARCHAR2(64),
  CREATED_AT TIMESTAMP(6) WITH TIME ZONE DEFAULT SYSTIMESTAMP NOT NULL,
  CREATED_BY VARCHAR2(36),
  CONSTRAINT PK_FILE_ASSET PRIMARY KEY (ID),
  CONSTRAINT CK_FILE_ASSET_SIZE CHECK (BYTE_SIZE IS NULL OR BYTE_SIZE >= 0),
  CONSTRAINT CK_FILE_ASSET_STORAGE CHECK (STORAGE_PROVIDER IN ('filesystem','s3','blob'))
);
CREATE INDEX IX_FILE_ASSET_SHA ON FILE_ASSET(SHA256);
-- MIME validation, malware scanning and access authorization are backend responsibilities.

CREATE TABLE SCHEME_ASSET_REF (
  SCHEME_VERSION_ID VARCHAR2(36) NOT NULL,
  OBJECT_ID VARCHAR2(36) NOT NULL,
  ASSET_ID VARCHAR2(36) NOT NULL,
  USAGE_TYPE VARCHAR2(30) NOT NULL,
  CONSTRAINT PK_SCH_ASSET_REF PRIMARY KEY
    (SCHEME_VERSION_ID, OBJECT_ID, ASSET_ID),
  CONSTRAINT FK_SCH_ASSET_OBJ FOREIGN KEY (SCHEME_VERSION_ID, OBJECT_ID)
    REFERENCES SCHEME_OBJECT(SCHEME_VERSION_ID, OBJECT_ID),
  CONSTRAINT FK_SCH_ASSET_FILE FOREIGN KEY (ASSET_ID) REFERENCES FILE_ASSET(ID),
  CONSTRAINT CK_SCH_ASSET_USE CHECK (USAGE_TYPE IN ('attachment','image','underlay','symbol','other'))
);
CREATE INDEX IX_SCH_ASSET_FILE ON SCHEME_ASSET_REF(ASSET_ID);

-- 14. Validation and operational notes
-- * Do not run CREATE TABLE on existing names; inspect USER_TABLES first.
-- * FK type/length/collation must match existing referenced key types.
-- * Deferrable self-FK permits inserting parents and children in arbitrary order.
-- * Full-text search can be added later via Oracle Text, not CLOB LIKE.
-- * All user/role/unit access rules must be enforced in API, not only Konva.
-- * Use UUID generation in application; avoid reusing IDs for new objects.
-- * Oracle VARCHAR2 uses bytes by default depending on session NLS_LENGTH_SEMANTICS.
-- * A single CLOB is preferred over chunking; chunking is not implemented here.
-- * JSON constraints validate syntax, not business schema or cross-reference consistency.
-- * For multi-million row datasets, benchmark before partitioning/index expansion.
-- * Before production, verify backup, migration rollback, and concurrency tests.
