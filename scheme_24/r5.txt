У твоїй архітектурі Next.js (frontend) → Node.js API → Oracle 21c я рекомендую виконувати збереження JSON Snapshot і синхронізацію індексованих таблиць через один endpoint, а не створювати окремі API-виклики для `DOC_SNAPSHOT`, `DOC_OBJECT` та `DOC_OBJECT_LINK`.

## 1. Які endpoint створити

| HTTP  | Endpoint                                      | Призначення                               |
| ----- | --------------------------------------------- | ----------------------------------------- |
| `PUT` | `/api/documents/:versionId/snapshot`          | Зберегти JSON та індексовані дані         |
| `GET` | `/api/documents/:versionId/snapshot`          | Завантажити повну схему                   |
| `GET` | `/api/documents/:versionId/objects`           | Отримати список об'єктів                  |
| `GET` | `/api/documents/:versionId/objects/:objectId` | Отримати інформацію про конкретний об'єкт |
| `GET` | `/api/documents/:versionId/links`             | Отримати зв'язки між об'єктами            |

Для пошуку обладнання по всіх схемах згодом можна додати окремий endpoint `GET /api/objects/search`.

## 3. Route — documentRoutes.js

```
import express from "express";

import {
  saveDocument,
  getDocumentSnapshot,
  getDocumentObjects,
  getDocumentObject,
} from "../controllers/documentController.js";

const router = express.Router();

router.put("/:versionId/snapshot", saveDocument);

router.get("/:versionId/snapshot", getDocumentSnapshot);

router.get("/:versionId/objects", getDocumentObjects);

router.get("/:versionId/objects/:objectId", getDocumentObject);

export default router;
```

Підключення до `app.js`:

```
import documentRoutes from "./routes/documentRoutes.js";

// Після middleware автентифікації
app.use("/api/documents", documentRoutes);
```

## 4. Controller — documentController.js

Контролер повинен приймати HTTP-запит і передавати його в сервіс.

```
import {
  saveDocumentVersion
} from "../services/documentSaveService.js";

export async function saveDocument(req, res) {
  try {
    const { versionId } = req.params;
    const { snapshot, saveRevision } = req.body;

    const result = await saveDocumentVersion({
      versionId,
      snapshot,
      saveRevision,
      userId: req.user.id,
    });

    return res.json({
      success: true,
      ...result,
    });
  } catch (error) {
    console.error("Save document error:", error);

    return res.status(error.statusCode || 500).json({
      success: false,
      message: error.statusCode
        ? error.message
        : "Помилка збереження документа",
    });
  }
}
```

Тут передбачається, що middleware автентифікації вже встановив `req.user`. Якщо його немає, потрібно підключити наявний механізм авторизації.

## 5. Service — documentSaveService.js

Саме тут має відбуватися вся логіка збереження.

```
export async function saveDocumentVersion({
  versionId,
  snapshot,
  saveRevision,
  userId,
}) {
  const connection = await getOracleConnection();

  try {
    // 1. Перевірити доступ користувача
    // 2. Заблокувати DOC_VERSION FOR UPDATE
    // 3. Переконатися, що STATUS_CODE = 'DRAFT'
    // 4. Перевірити SAVE_REVISION
    // 5. Зберегти DOC_SNAPSHOT
    // 6. Синхронізувати DOC_OBJECT (NODE, EDGE ...)
    // 7. Синхронізувати DOC_OBJECT_LINK
    // 8. Синхронізувати DOC_OBJECT_ATTRIBUTE

    await connection.commit();

    return {
      versionId,
      // Повернути фактичний новий SAVE_REVISION
    };
  } catch (error) {
    await connection.rollback();
    throw error;
  } finally {
    await connection.close();
  }
}
```

Це каркас, а не готовий до запуску сервіс: функцію `getOracleConnection()` та операції запису потрібно реалізувати відповідно до твого підключення до Oracle.

Критично: якщо твій `OracleDBManager.query(dbName, sql, params)` відкриває окреме з'єднання для кожного запиту, його не можна використовувати для цих операцій без модифікації. Усі SQL-запити збереження мають виконуватися через одне Oracle-з'єднання та одну транзакцію.

## 6. Як це викликати з Next.js

У кнопці «Зберегти» виконується тільки один HTTP-запит:

```
import { useCanvasStore } from "@/store/useCanvasStore";
import { buildSnapshot } from "@/lib/schemeSnapshot";

async function saveScheme(versionId, saveRevision) {
  const state = useCanvasStore.getState();

  const snapshot = buildSnapshot(state);

  const response = await fetch(
    `${process.env.NEXT_PUBLIC_API_URL}/api/documents/${encodeURIComponent(versionId)}/snapshot`,
    {
      method: "PUT",
      headers: {
        "Content-Type": "application/json",
      },
      credentials: "include",
      body: JSON.stringify({
        snapshot,
        saveRevision,
      }),
    }
  );

  if (!response.ok) {
    throw new Error("Не вдалося зберегти схему");
  }

  return response.json();
}
```

Отже, на рівні API один виклик з Next.js, а всередині Node.js — декілька SQL-операцій в одній транзакції.

Для читання навпаки: повну схему завантажуємо з `DOC_SNAPSHOT`, а окремі дані для пошуку, інформаційних карток, фільтрації та аналітики отримуємо з індексованих таблиць.

Для твого проєкту я рекомендую почати саме з `PUT /api/documents/:versionId/snapshot` і `GET /api/documents/:versionId/snapshot`. Коли ці два endpoint працюватимуть, розширимо API пошуком об'єктів та зв'язків, не змінюючи механізм збереження.
