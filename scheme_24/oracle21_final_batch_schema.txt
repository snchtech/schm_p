-- Execute as the application schema owner, in SQL Developer (Run Script/F5).
-- Prerequisites: CREATE TABLE, CREATE INDEX; existing ETLS_USERS/dict_unit are external and NOT FK-linked here.
-- Existing ALBUM / ALBUM_VERSION / DOC_RESOURCE / DOC_VERSION names retained from screenshots.
-- Architectural choice: DOC_OBJECT stores ONLY CURRENT working state, keyed by DOC_RESOURCE_ID;
-- official versions are immutable JSON snapshots, NOT object-by-object revisions.
-- Removed: DOC_OBJECT_IDENTITY, OBJECT_REVISION, DOC_VERSION_OBJECT (not needed for batch save).
-- No DOC_CHANGE_LOG or DOC_EDIT_SESSION. DOC_OBJECT_LINK is retained for endpoint relationships.
-- IMPORTANT: use ONE Oracle transaction for updating objects, links, attributes and current snapshot.
-- IMPORTANT: this is CREATE script, not idempotent migration; existing tables cause ORA-00955.

-- 04. CURRENT OBJECTS: ONE ROW PER OBJECT IN A DOCUMENT

-- 05. CURRENT JSON SNAPSHOT + IMMUTABLE OFFICIAL VERSION JSON
CREATE TABLE DOC_SNAPSHOT_CURRENT (
 DOC_RESOURCE_ID VARCHAR2(36) PRIMARY KEY,

);

CREATE TABLE DOC_SNAPSHOT_HISTORY (
 DOC_VERSION_ID VARCHAR2(36) PRIMARY KEY,
 SNAPSHOT_JSON CLOB NOT NULL,
 SOURCE_SAVE_REVISION NUMBER(12) NOT NULL,
 CREATED_AT TIMESTAMP(6) WITH TIME ZONE DEFAULT SYSTIMESTAMP NOT NULL,
 CREATED_BY VARCHAR2(36) NOT NULL,
 CONSTRAINT FK_SNAPSHOT_HISTORY_VER FOREIGN KEY(DOC_VERSION_ID) REFERENCES DOC_VERSION(ID),
 CONSTRAINT CK_SNAPSHOT_HISTORY_JSON CHECK(SNAPSHOT_JSON IS JSON)
);
-- Enforce immutable history via privileges / API; see trigger below.
CREATE OR REPLACE TRIGGER TRG_DOC_SNAPSHOT_HISTORY_IMMUTABLE
BEFORE UPDATE OR DELETE ON DOC_SNAPSHOT_HISTORY
BEGIN
 RAISE_APPLICATION_ERROR(-20051,'Historical snapshot is immutable');
END;
/

-- 10. NOTES / DESIGN DECISIONS
-- (a) DOC_RESOURCE belongs to an ALBUM_VERSION (as in screenshots). If albums are republished,
--     determine whether the document itself must move or whether membership needs a join table.
-- (b) Current snapshot JSON must include nodes, edges, zvt, branches, specification, canvas metadata,
--     and schemaVersion. Current object table stores the same objects for search and relationships.
-- (c) Specification can be stored in snapshot JSON now. Add a relational specification table only
--     if server-side reporting/filtering of specification rows is needed.
-- (d) Object coordinates/geometry remain in OBJECT_JSON; EDGE_FUNCTION_ID references versioned code key.
-- (e) Never DELETE old official JSON snapshots in ordinary saves.
-- (f) Logical deletion IS_DELETED does not automatically remove dependent links: batch API must do so.
-- (g) The screenshot schema had DOC_OBJECT keyed by DOC_VERSION_ID and DOC_OBJECT_LINK likewise;
--     this script INTENTIONALLY replaces those with DOC_RESOURCE_ID for mutable working state.
-- (h) Oracle does not support TEXT data type; replaced with VARCHAR2/CLOB.
-- (i) Existing screenshots contained CREATE INDEX on undefined DOC/DOCUMENT tables; removed.
-- (j) User/department/unit FKs omitted because external table definitions were not provided.
-- (k) Snapshot is a CLOB JSON in Oracle 21c; later consider native JSON for high-volume workloads.
-- (l) The PK / indexes make this usable, but test with realistic workload and inspect query plans.
-- (m) Cycles in DOC_RESOURCE hierarchy or DOC_OBJECT parent chains require API-side validation.
-- (n) DOC_SNAPSHOT_HISTORY trigger blocks updates/deletes, including manual maintenance;
--     administrators may need a controlled migration procedure.
-- (o) Oracle DDL auto-commits; use a fresh schema for testing before applying to production.
