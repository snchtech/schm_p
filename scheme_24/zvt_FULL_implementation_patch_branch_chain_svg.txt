ВАЖЛИВО ПРО СТАТУС
Цей документ містить повні самодостатні реалізації ядра геометрії, моделі,
операцій додавання/видалення/перестановки, приклад Konva-шару, SVG-експорту,
Zustand-slice, міграції та автоматичних тестів. Він НЕ є готовим патчем
безпосередньо до вашого репозиторію: назви/формати node, edge, існуючого
ZvtSymbol, root-mark renderer, прав доступу та MUI-форм треба зіставити
з вашим фактичним кодом. Не копіюйте старі однойменні реалізації разом із
новими: це спричинить подвійне зміщення та конфлікти імпортів.

1. ЧОМУ ПОПЕРЕДНІЙ TXT БУВ НЕДОСТАТНІМ
Попередній файл містив 291 рядок і переважно інтеграційні фрагменти.
Не було завершеного store slice, повного спільного resolver для node/edge,
повноцінного SVG renderer, міграції й виконуваних тестів. Тому вважати
його повним готовим кодом було б неправильно. Нижче — значно ширший пакет.

2. ПОВЕДІНКА ТА ІНВАРІАНТИ
branch.attachedTo: обов'язково {type:'node'|'edge', id, side?/segmentId?/t?}.
zvt.branchId: обов'язково; пряме zvt.attachedTo не використовується.
route.version=3. Для node route.coordinateMode='translation';
для edge route.coordinateMode='edge-local'.
route.vertices — усі точки ПІСЛЯ root у локальних координатах;
остання з них є БАЗОВОЮ terminal-точкою.
connection.offsetAlongBranch — відстань від ПОЧАТКУ terminal segment.
connection.side — сторона КРУЖЕЧКА, з якої виходить stub:
  top => кружечок нижче tap;
  bottom => кружечок вище tap;
  left => кружечок праворуч tap;
  right => кружечок ліворуч tap.
stubLength — довжина від branch tap до поверхні кола, а не до центру.
radius — радіус кола.
tap, contact, center, trunk обчислюються одним resolver.

6. DELETE LIFECYCLE
delete branch -> видалити branch + всі дочірні ЗВТ + runtime.
delete ZVT -> видалити лише ЗВТ; trunk derived до останнього tap.
delete останнього ZVT -> видалити також branch.
Порожній branch дозволений тільки ДО додавання першого ЗВТ.
Перед видаленням branch з дочірніми ЗВТ показати MUI confirm dialog.

7. КРИТИЧНІ ІНТЕГРАЦІЙНІ АДАПТЕРИ
Node у branchChain.js за замовчуванням:
  {id,x,y,width,height}, x/y = CENTER у world coordinates.
Якщо ваш node.x/y — лівий верхній кут або node має rotation,
адаптуйте resolveRoot() під фактичні anchors / getClientRect().
Edge:
  {id,segments:[{id,start:{x,y},end:{x,y}}]}.
Якщо у вас інший формат edge, змініть лише getEdgeSegment().
Якщо edge polyline, зберігайте реальний segmentId, не використовуйте
fallback до першого сегмента при зміні геометрії.
Для повороту edge route у локальному базисі tangent/normal
обертається разом із edge; за потреби вибору сторони виходу
нормаль має зберігати стабільний знак відносно сегмента.
В SVG використовуйте ті самі координати world/diagram.

8. РЕДАГУВАННЯ BENDS
Ваші існуючі STEP 12 insertDogLeg/shiftSegmentParallel можна

// src/lib/zvt/branchChain.js
// Pure JavaScript. Shared by Konva and SVG. Coordinates are diagram/world coordinates.
export const SIDES = Object.freeze({
  top:{x:0,y:-1}, bottom:{x:0,y:1}, left:{x:-1,y:0}, right:{x:1,y:0}
});
const EPS=1e-7;
const add=(a,b)=>({x:a.x+b.x,y:a.y+b.y});
const sub=(a,b)=>({x:a.x-b.x,y:a.y-b.y});
const mul=(a,n)=>({x:a.x*n,y:a.y*n});
const dot=(a,b)=>a.x*b.x+a.y*b.y;
const length=a=>Math.hypot(a.x,a.y);
const unit=a=>{const n=length(a);if(n<EPS)throw Error("Zero-length segment");return mul(a,1/n)};
const finite=n=>typeof n==="number"&&Number.isFinite(n);
const assertPoint=p=>{if(!p||!finite(p.x)||!finite(p.y))throw Error("Invalid point");return p};
const clamp=(n,a,b)=>Math.max(a,Math.min(b,n));
const uid=()=>globalThis.crypto?.randomUUID?.()??`zvt-${Date.now()}-${Math.random().toString(36).slice(2)}`;
export const pt=(x,y)=>({x,y});
export const sideVector=side=>{if(!SIDES[side])throw Error(`Unknown connection side ${side}`);return SIDES[side]};
export const segmentFrame=(a,b)=>{
  const tangent=unit(sub(b,a));
  return {tangent,normal:{x:-tangent.y,y:tangent.x}};
};
export function getEdgeSegment(edge,segmentId){
  // ADAPTER: replace edge.points if your project stores coordinates elsewhere.
  const segments=edge.segments??edge.geometry?.segmentsMeta;
  const seg=segments?.find(s=>s.id===segmentId)??segments?.[0];
  if(seg){
    if(seg.start&&seg.end)return {a:assertPoint(seg.start),b:assertPoint(seg.end)};
    if([seg.x1,seg.y1,seg.x2,seg.y2].every(finite))
      return {a:pt(seg.x1,seg.y1),b:pt(seg.x2,seg.y2)};
  }
  const points=edge.points??edge.geometry?.points;
  if(Array.isArray(points)&&points.length>=4)
    return {a:pt(points[0],points[1]),b:pt(points[2],points[3])};
  if(edge.start&&edge.end)return {a:assertPoint(edge.start),b:assertPoint(edge.end)};
  throw Error(`Cannot resolve edge segment: ${edge.id}`);
}
export function resolveRoot(branch,nodesById,edgesById){
  const a=branch.attachedTo;
  if(!a?.id||!["node","edge"].includes(a.type))throw Error(`branch ${branch.id} missing attachedTo`);
  if(a.type==="node"){
    const node=nodesById[a.id];if(!node)throw Error(`Missing node ${a.id}`);
    // NODE ADAPTER: position is center, width/height are full dimensions.
    // For custom shapes use node.anchors[side] in WORLD coordinates.
    const side=a.side??"right";
    const anchor=node.anchors?.[side];
    if(anchor)return {point:assertPoint(anchor),frame:{tangent:pt(1,0),normal:pt(0,1)}};
    const x=node.x??node.position?.x,y=node.y??node.position?.y;
    if(!finite(x)||!finite(y))throw Error("Node position adapter required");
    const w=node.width??node.geometry?.width??0,h=node.height??node.geometry?.height??0;
    const offsets={left:pt(-w/2,0),right:pt(w/2,0),top:pt(0,-h/2),bottom:pt(0,h/2)};
    if(!offsets[side])throw Error(`Invalid node side ${side}`);
    const local=a.localOffset??pt(0,0);
    return {point:add(pt(x,y),add(offsets[side],local)),frame:{tangent:pt(1,0),normal:pt(0,1)}};
  }
  const edge=edgesById[a.id];if(!edge)throw Error(`Missing edge ${a.id}`);
  const {a:p0,b:p1}=getEdgeSegment(edge,a.segmentId);
  const frame=segmentFrame(p0,p1);
  const t=clamp(a.t??0.5,0,1);
  return {point:add(p0,mul(sub(p1,p0),t)),frame};
}
export function rootAtEdgePointer(edge,segmentId,pointer){
  const {a,b}=getEdgeSegment(edge,segmentId);
  const d=sub(b,a);
  return clamp(dot(sub(pointer,a),d)/dot(d,d),0,1);
}
export function worldToLocal(root,frame,world,mode){
  const d=sub(world,root);
  return mode==="edge-local" ? {u:dot(d,frame.tangent),v:dot(d,frame.normal)}
    : {dx:d.x,dy:d.y};
}
export function localToWorld(root,frame,p,mode){
  return mode==="edge-local"
    ? add(root,add(mul(frame.tangent,p.u),mul(frame.normal,p.v)))
    : add(root,pt(p.dx,p.dy));
}
export function resolveBaseRoute(branch,rootResult){
  const route=branch.route;
  if(route?.version!==3)throw Error(`Migrate branch ${branch.id} route to version 3`);
  const {point:root,frame}=rootResult;
  const rel=route.vertices??[];
  const points=[root,...rel.map(p=>localToWorld(root,frame,p,route.coordinateMode))];
  if(points.length<2)throw Error(`branch ${branch.id} needs terminal point`);
  return points;
}
export function persistEditedRoute(branch,worldPoints,rootResult){
  if(worldPoints.length<2)throw Error("Invalid route");
  const mode=branch.route.coordinateMode;
  return {...branch,route:{...branch.route,
    vertices:worldPoints.slice(1).map(p=>worldToLocal(rootResult.point,rootResult.frame,p,mode))}};
}
export function createBranch({id=uid(),attachedTo,route,parameter="F"}){
  if(!attachedTo?.id||!["node","edge"].includes(attachedTo.type))throw Error("attachedTo required");
  if(!route?.vertices?.length)throw Error("route.vertices required");
  return {id,kind:"zvtBranch",attachedTo:{...attachedTo},data:{parameter},
    route:{version:3,coordinateMode:attachedTo.type==="edge"?"edge-local":"translation",
      memberSpacing:90,...route}};
}
export function createZvt({id=uid(),branchId,connection={},data={},style={},integration={}}){
  if(!branchId)throw Error("branchId required");
  const side=connection.side??"bottom";sideVector(side);
  const stubLength=connection.stubLength??22,radius=style.radius??22;
  if(!finite(stubLength)||stubLength<0||!finite(radius)||radius<=0)throw Error("Invalid ZVT dimensions");
  return {id,kind:"zvt",branchId,
    connection:{side,stubLength,offsetAlongBranch:connection.offsetAlongBranch??0},
    data:{...data},style:{radius,...style},integration:{...integration}};
}
export function axisOf(basePoints){
  const origin=basePoints.at(-2),baseEnd=basePoints.at(-1);
  const delta=sub(baseEnd,origin),baseLength=length(delta);
  if(baseLength<EPS)throw Error("Terminal segment must be nonzero");
  return {origin,direction:mul(delta,1/baseLength),baseLength};
}
export const pointOnAxis=(axis,offset)=>add(axis.origin,mul(axis.direction,offset));
export function membersOf(branchId,zvts){
  return zvts.filter(z=>z.branchId===branchId).slice().sort((a,b)=>
    a.connection.offsetAlongBranch-b.connection.offsetAlongBranch||
    String(a.id).localeCompare(String(b.id)));
}
export function memberGeometry(zvt,axis){
  const side=sideVector(zvt.connection.side);
  const offset=zvt.connection.offsetAlongBranch;
  const tap=pointOnAxis(axis,offset);
  const contact=sub(tap,mul(side,zvt.connection.stubLength));
  const center=sub(contact,mul(side,zvt.style.radius));
  return {zvt,tap,contact,center,stub:[tap.x,tap.y,contact.x,contact.y]};
}
export function resolveBranchScene(branch,basePoints,zvts){
  const axis=axisOf(basePoints),members=membersOf(branch.id,zvts);
  const last=members.at(-1);
  const terminalOffset=last?last.connection.offsetAlongBranch:axis.baseLength;
  const trunk=[...basePoints.slice(0,-1),pointOnAxis(axis,terminalOffset)];
  return {branch,axis,trunk,members:members.map(z=>memberGeometry(z,axis))};
}
export function nextOffset(branch,basePoints,zvts){
  const axis=axisOf(basePoints),members=membersOf(branch.id,zvts);
  if(!members.length)return axis.baseLength;
  return Math.max(axis.baseLength,members.at(-1).connection.offsetAlongBranch)
    +Math.max(1,branch.route.memberSpacing??90);
}
export function appendZvt(state,branchId,options,basePoints){
  const branch=state.zvtBranches.find(b=>b.id===branchId);
  if(!branch)throw Error("Branch not found");
  const zvt=createZvt({...options,branchId,connection:{...options.connection,
    offsetAlongBranch:nextOffset(branch,basePoints,state.zvts)}});
  return {...state,zvts:[...state.zvts,zvt]};
}
export function tapFromCenter(center,zvt){
  const s=sideVector(zvt.connection.side);
  return add(center,mul(s,zvt.connection.stubLength+zvt.style.radius));
}
export function offsetFromCenter(center,zvt,axis,minOffset=0){
  const tap=tapFromCenter(center,zvt);
  return Math.max(minOffset,dot(sub(tap,axis.origin),axis.direction));
}
export function centerAtOffset(zvt,axis,offset){
  return memberGeometry({...zvt,connection:{...zvt.connection,offsetAlongBranch:offset}},axis).center;
}
export function reorderZvt(state,branchId,zvtId,targetIndex){
  const ordered=membersOf(branchId,state.zvts);
  const source=ordered.findIndex(z=>z.id===zvtId);
  if(source<0)throw Error("ZVT not in branch");
  const offsets=ordered.map(z=>z.connection.offsetAlongBranch);
  const [moved]=ordered.splice(source,1);
  ordered.splice(clamp(targetIndex,0,ordered.length),0,moved);
  const map=new Map(ordered.map((z,i)=>[z.id,offsets[i]]));
  return {...state,zvts:state.zvts.map(z=>map.has(z.id)?
    {...z,connection:{...z.connection,offsetAlongBranch:map.get(z.id)}}:z)};
}
export function moveZvt(state,zvtId,offset){
  if(!finite(offset))throw Error("Invalid offset");
  return {...state,zvts:state.zvts.map(z=>z.id===zvtId?
    {...z,connection:{...z.connection,offsetAlongBranch:offset}}:z)};
}
export function deleteZvt(state,zvtId){
  const target=state.zvts.find(z=>z.id===zvtId);if(!target)return state;
  const zvts=state.zvts.filter(z=>z.id!==zvtId);
  const hasRemaining=zvts.some(z=>z.branchId===target.branchId);
  const zvtBranches=hasRemaining?state.zvtBranches:
    state.zvtBranches.filter(b=>b.id!==target.branchId);
  const runtime={...(state.zvtRuntimeById??{})};delete runtime[zvtId];
  return {...state,zvts,zvtBranches,zvtRuntimeById:runtime};
}
export function deleteBranch(state,branchId){
  const removed=new Set(state.zvts.filter(z=>z.branchId===branchId).map(z=>z.id));
  const runtime={...(state.zvtRuntimeById??{})};
  for(const id of removed)delete runtime[id];
  return {...state,zvtBranches:state.zvtBranches.filter(b=>b.id!==branchId),
    zvts:state.zvts.filter(z=>!removed.has(z.id)),zvtRuntimeById:runtime};
}
export function translateNode(state,nodeId,dx,dy){
  // Demo adapter. In actual app update your canonical node store instead.
  return {...state,nodes:state.nodes.map(n=>n.id===nodeId?
    {...n,x:n.x+dx,y:n.y+dy}:n)};
}
export function resolveAllScenes(state){
  const nodesById=Object.fromEntries((state.nodes??[]).map(n=>[n.id,n]));
  const edgesById=Object.fromEntries((state.edges??[]).map(e=>[e.id,e]));
  return state.zvtBranches.map(branch=>{
    const root=resolveRoot(branch,nodesById,edgesById);
    const base=resolveBaseRoute(branch,root);
    return {...resolveBranchScene(branch,base,state.zvts),root};
  });
}
export function validateGraph(state){
  const branches=new Map(state.zvtBranches.map(b=>[b.id,b]));
  for(const b of state.zvtBranches){
    if(!b.attachedTo?.id)throw Error(`branch ${b.id} has no attachedTo`);
    if(b.route?.version!==3)throw Error(`branch ${b.id} has legacy route`);
  }
  for(const z of state.zvts){
    if(!branches.has(z.branchId))throw Error(`Orphan ZVT ${z.id}`);
    sideVector(z.connection.side);
  }
  return true;
}

// Run ONCE on load after node/edge geometry is available; back up JSON first.
import {
 resolveRoot,worldToLocal,getEdgeSegment,rootAtEdgePointer
} from "./branchChain.js";
export function migrateBranchRouteV3(branch,nodesById,edgesById){
 if(branch.route?.version===3)return branch;
 const a={...branch.attachedTo};
 if(a.type==="edge"){
  const edge=edgesById[a.id];
  if(!edge)throw Error(`Cannot migrate missing edge ${a.id}`);
  const oldPoint=a.preferredPoint;
  a.t=oldPoint?rootAtEdgePointer(edge,a.segmentId,oldPoint):(a.t??0.5);
 }
 const updated={...branch,attachedTo:a};
 const root=resolveRoot(updated,nodesById,edgesById);
 const old=branch.route??{};
 // Legacy buildBaseBranchRoute may use route.type/lengths rather than terminalPoint.
 // For those records, call your EXISTING route builder first to produce oldPoints.
 const oldPoints=old.absolutePoints??
   (old.terminalPoint?
     [root.point,...(old.manualVertices??[]),old.terminalPoint]:
     null);
 if(!oldPoints||oldPoints.length<2)
   throw Error(`Cannot migrate ${branch.id}: supply legacy absolutePoints from old route builder`);
 const mode=a.type==="edge"?"edge-local":"translation";
 return {...updated,route:{...old,version:3,coordinateMode:mode,
   vertices:oldPoints.slice(1).map(point=>worldToLocal(root.point,root.frame,point,mode)),
   memberSpacing:old.memberSpacing??90}};
}
export function migrateDiagramV3(document){
 const nodesById=Object.fromEntries((document.nodes??[]).map(n=>[n.id,n]));
 const edgesById=Object.fromEntries((document.edges??[]).map(e=>[e.id,e]));
 return {...document,zvtBranches:(document.zvtBranches??[]).map(b=>
   migrateBranchRouteV3(b,nodesById,edgesById))};
}
